-- 🛠️ Утилиты
local function Create(className, props)
    local inst = Instance.new(className)
    for k, v in pairs(props or {}) do
        inst[k] = v
    end
    return inst
end

local function Corner(radius, parent)
    return Create("UICorner", {
        CornerRadius = UDim.new(0, radius or 8),
        Parent = parent
    })
end

local function Stroke(color, thickness, transparency, parent)
    return Create("UIStroke", {
        Color = color or Theme.Accent,
        Thickness = thickness or 1,
        Transparency = transparency or 0.5,
        Parent = parent
    })
end

local function Tween(obj, time, props)
    local tween = TweenService:Create(obj, TweenInfo.new(time or 0.2, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), props)
    tween:Play()
    return tween
end

-- 🖱️ Перетаскивание
local function MakeDraggable(frame, dragArea)
    dragArea = dragArea or frame
    local dragging, dragStart, startPos

    dragArea.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            dragStart = input.Position
            startPos = frame.Position
            input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then
                    dragging = false
                end
            end)
        end
    end)

    UserInputService.InputChanged:Connect(function(input)
        if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
            local delta = input.Position - dragStart
            frame.Position = UDim2.new(
                startPos.X.Scale, startPos.X.Offset + delta.X,
                startPos.Y.Scale, startPos.Y.Offset + delta.Y
            )
        end
    end)
end

    -- 📄 Создать вкладку
    function Window:Tab(tabName, icon)
        local tabButton = Create("TextButton", {
            Name = tabName .. "Tab",
            Size = UDim2.new(1, 0, 0, 38),
            BackgroundColor3 = Theme.Secondary,
            BorderSizePixel = 0,
            Font = Enum.Font.GothamMedium,
            Text = "  " .. (icon and icon .. "  " or "") .. tabName,
            TextColor3 = Theme.SubText,
            TextSize = 13,
            TextXAlignment = Enum.TextXAlignment.Left,
            Parent = sidebar
        })
        Corner(8, tabButton)

        local page = Create("ScrollingFrame", {
            Name = tabName .. "Page",
            Size = UDim2.new(1, 0, 1, 0),
            BackgroundTransparency = 1,
            BorderSizePixel = 0,
            ScrollBarThickness = 4,
            ScrollBarImageColor3 = Theme.Accent,
            CanvasSize = UDim2.new(0, 0, 0, 0),
            AutomaticCanvasSize = Enum.AutomaticSize.Y,
            Visible = false,
            Parent = pagesContainer
        })

        Create("UIListLayout", {
            Padding = UDim.new(0, 8),
            SortOrder = Enum.SortOrder.LayoutOrder,
            Parent = page
        })

        Create("UIPadding", {
            PaddingRight = UDim.new(0, 8),
            Parent = page
        })

        local tab = {
            Button = tabButton,
            Page = page,
            Name = tabName
        }

        -- Активация вкладки
        local function activate()
            for _, t in pairs(Window.Tabs) do
                t.Page.Visible = false
                Tween(t.Button, 0.15, {BackgroundColor3 = Theme.Secondary, TextColor3 = Theme.SubText})
            end
            page.Visible = true
            Tween(tabButton, 0.15, {BackgroundColor3 = Theme.Accent, TextColor3 = Theme.Text})
            Window.ActiveTab = tab
        end

        tabButton.MouseButton1Click:Connect(activate)

        -- Hover эффект
        tabButton.MouseEnter:Connect(function()
            if Window.ActiveTab ~= tab then
                Tween(tabButton, 0.15, {BackgroundColor3 = Theme.Tertiary, TextColor3 = Theme.Text})
            end
        end)
        tabButton.MouseLeave:Connect(function()
            if Window.ActiveTab ~= tab then
                Tween(tabButton, 0.15, {BackgroundColor3 = Theme.Secondary, TextColor3 = Theme.SubText})
            end
        end)

        table.insert(Window.Tabs, tab)
        if not Window.ActiveTab then activate() end

        -- 📌 Методы вкладки
        local TabAPI = {}

        function TabAPI:Button(text, callback)
            local btn = Create("TextButton", {
                Size = UDim2.new(1, 0, 0, 36),

                Font = Enum.Font.GothamMedium,

            btn.MouseEnter:Connect(function()
                Tween(btn, 0.15, {BackgroundColor3 = Theme.Accent})

            btn.MouseLeave:Connect(function()
                Tween(btn, 0.15, {BackgroundColor3 = Theme.Secondary})
            end)
            btn.MouseButton1Click:Connect(function()
                Tween(btn, 0.08, {BackgroundColor3 = Theme.AccentDark})
                task.wait(0.08)
                Tween(btn, 0.15, {BackgroundColor3 = Theme.Accent})
                if callback then callback() end
            end)
            return btn
        end

        -- Тумблер
        function TabAPI:Toggle(text, default, callback)
            local state = default or false

            local frame = Create("TextButton", {
                Size = UDim2.new(1, 0, 0, 36),
                BackgroundColor3 = Theme.Secondary,
                BorderSizePixel = 0,
                Text = "",
                Parent = page
            })
            Corner(8, frame)

            local label = Create("TextLabel", {
                Size = UDim2.new(1, -60, 1, 0),
                Position = UDim2.new(0, 12, 0, 0),
                BackgroundTransparency = 1,
                Font = Enum.Font.GothamMedium,
                Text = text,
                TextColor3 = Theme.Text,
                TextSize = 14,
                TextXAlignment = Enum.TextXAlignment.Left,
                Parent = frame
            })

            local switchBg = Create("Frame", {
                Size = UDim2.new(0, 42, 0, 22),
                Position = UDim2.new(1, -52, 0.5, -11),
                BackgroundColor3 = state and Theme.Accent or Theme.Tertiary,
                BorderSizePixel = 0,
                Parent = frame
            })
            Corner(11, switchBg)

            local knob = Create("Frame", {
                Size = UDim2.new(0, 18, 0, 18),
                Position = state and UDim2.new(1, -20, 0.5, -9) or UDim2.new(0, 2, 0.5, -9),
                BackgroundColor3 = Theme.Text,
                BorderSizePixel = 0,
                Parent = switchBg
            })
            Corner(9, knob)

            local function update()
                Tween(switchBg, 0.2, {BackgroundColor3 = state and Theme.Accent or Theme.Tertiary})
                Tween(knob, 0.2, {Position = state and UDim2.new(1, -20, 0.5, -9) or UDim2.new(0, 2, 0.5, -9)})
                if callback then callback(state) end
            end

            frame.MouseButton1Click:Connect(function()
                state = not state
                update()
            end)

            return {
                Set = function(v) state = v; update() end,
                Get = function() return state end
            }
        end

        -- Слайдер
        function TabAPI:Slider(text, min, max, default, callback)
            min, max = min or 0, max or 100
            local value = default or min

            local frame = Create("Frame", {
                Size = UDim2.new(1, 0, 0, 50),
                BackgroundColor3 = Theme.Secondary,
                BorderSizePixel = 0,
                Parent = page
            })
            Corner(8, frame)

            local label = Create("TextLabel", {
                Size = UDim2.new(0.7, 0, 0, 20),
                Position = UDim2.new(0, 12, 0, 6),
                BackgroundTransparency = 1,
                Font = Enum.Font.GothamMedium,
                Text = text,
                TextColor3 = Theme.Text,
                TextSize = 13,
                TextXAlignment = Enum.TextXAlignment.Left,
                Parent = frame
            })

            local valueLabel = Create("TextLabel", {
                Size = UDim2.new(0.3, -12, 0, 20),
                Position = UDim2.new(0.7, 0, 0, 6),
                BackgroundTransparency = 1,
                Font = Enum.Font.GothamBold,
                Text = tostring(value),
                TextColor3 = Theme.Accent,
                TextSize = 13,
                TextXAlignment = Enum.TextXAlignment.Right,
                Parent = frame
            })

            local barBg = Create("Frame", {
                Size = UDim2.new(1, -24, 0, 6),
                Position = UDim2.new(0, 12, 0, 34),
                BackgroundColor3 = Theme.Tertiary,
                BorderSizePixel = 0,
                Parent = frame
            })
            Corner(3, barBg)

            local fill = Create("Frame", {
                Size = UDim2.new((value - min) / (max - min), 0, 1, 0),
                BackgroundColor3 = Theme.Accent,
                BorderSizePixel = 0,
                Parent = barBg
            })
            Corner(3, fill)

            local dragging = false

            local function setValueFromX(x)
                local rel = math.clamp((x - barBg.AbsolutePosition.X) / barBg.AbsoluteSize.X, 0, 1)
                value = math.floor(min + (max - min) * rel + 0.5)
                fill.Size = UDim2.new(rel, 0, 1, 0)
                valueLabel.Text = tostring(value)
                if callback then callback(value) end
            end

            barBg.InputBegan:Connect(function(input)
                if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
                    dragging = true
                    setValueFromX(input.Position.X)
                end
            end)
            UserInputService.InputChanged:Connect(function(input)
                if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
                    setValueFromX(input.Position.X)
                end
            end)
            UserInputService.InputEnded:Connect(function(input)
                if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
                    dragging = false
                end
            end)

            return {
                Set = function(v)
                    value = math.clamp(v, min, max)
                    local rel = (value - min) / (max - min)
                    fill.Size = UDim2.new(rel, 0, 1, 0)
                    valueLabel.Text = tostring(value)
                    if callback then callback(value) end
                end,
                Get = function() return value end
            }
        end

                    label.Text = text .. ": " .. opt

            btn.MouseButton1Click:Connect(function()
                open = not open
                local targetHeight = open and (#options * 32 + 8) or 0
                Tween(list, 0.2, {Size = UDim2.new(1, 0, 0, targetHeight)})
                Tween(frame, 0.2, {Size = UDim2.new(1, 0, 0, 36 + targetHeight)})
                arrow.Text = open and "▲" or "▼"
            end)

            return frame
        end

        return TabAPI
    end

end

-- 🔔 Система уведомлений
function BeautifulUI:Notify(title, text, duration)
    duration = duration or 3

    local gui = Create("ScreenGui", {
        Name = "BeautifulUINotify",
        ResetOnSpawn = false,
        ZIndexBehavior = Enum.ZIndexBehavior.Sibling,
        Parent = (gethui and gethui()) or CoreGui
    })

    local container = Create("Frame", {
        Size = UDim2.new(0, 320, 1, 0),
        Position = UDim2.new(1, -340, 0, 0),
        BackgroundTransparency = 1,
        Parent = gui
    })
    Create("UIListLayout", {
        Padding = UDim.new(0, 10),
        VerticalAlignment = Enum.VerticalAlignment.Bottom,
        SortOrder = Enum.SortOrder.LayoutOrder,
        Parent = container
    })
    Create("UIPadding", {
        PaddingBottom = UDim.new(0, 20),
        Parent = container
    })

    local notify = Create("Frame", {
        Size = UDim2.new(1, 0, 0, 80),
        Position = UDim2.new(1, 340, 0, 0),
        BackgroundColor3 = Theme.Secondary,
        BorderSizePixel = 0,
        Parent = container
    })
    Corner(10, notify)
    Stroke(Theme.Accent, 1, 0.5, notify)

    -- Полоса слева
    local bar = Create("Frame", {
        Size = UDim2.new(0, 4, 1, 0),
        BackgroundColor3 = Theme.Accent,
        BorderSizePixel = 0,
        Parent = notify
    })
    Corner(2, bar)

    Create("TextLabel", {
        Size = UDim2.new(1, -24, 0, 20),
        Position = UDim2.new(0, 16, 0, 12),
        BackgroundTransparency = 1,
        Font = Enum.Font.GothamBold,
        Text = title or "Уведомление",
        TextColor3 = Theme.Text,
        TextSize = 14,
        TextXAlignment = Enum.TextXAlignment.Left,
        Parent = notify
    })

    Create("TextLabel", {
        Size = UDim2.new(1, -24, 0, 36),
        Position = UDim2.new(0, 16, 0, 34),
        BackgroundTransparency = 1,
        Font = Enum.Font.Gotham,
        Text = text or "",
        TextColor3 = Theme.SubText,
        TextSize = 12,
        TextWrapped = true,
        TextXAlignment = Enum.TextXAlignment.Left,
        TextYAlignment = Enum.TextYAlignment.Top,
        Parent = notify
    })

    -- Анимация появления
    Tween(notify, 0.3, {Position = UDim2.new(0, 0, 0, 0)})

    -- Анимация исчезновения
    task.delay(duration, function()
        Tween(notify, 0.3, {Position = UDim2.new(1, 340, 0, 0)})
        task.wait(0.35)
        gui:Destroy()
    end)
end

-- 🎯 Горячая клавиша для скрытия/показа
function BeautifulUI:BindToggle(window, key)
    key = key or Enum.KeyCode.RightShift
    local visible = true

    UserInputService.InputBegan:Connect(function(input, gpe)
        if gpe then return end
        if input.KeyCode == key then
            visible = not visible
            window.Main.Visible = visible
        end
    end)
end

return BeautifulUI
