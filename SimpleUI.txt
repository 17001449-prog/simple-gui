function BeautifulUI:SetTheme(name)
    if Themes[name] then
        Theme = Themes[name]

        return true
    end
    return false
end

local function Create(className, props)
    local inst = Instance.new(className)
    for k, v in pairs(props or {}) do inst[k] = v end
    return inst
end

local function Corner(radius, parent)
    return Create("UICorner", {CornerRadius = UDim.new(0, radius or 8), Parent = parent})
end

local function Stroke(color, thickness, transparency, parent)
    return Create("UIStroke", {Color = color or Theme.Accent, Thickness = thickness or 1, Transparency = transparency or 0.5, Parent = parent})
end

local function Tween(obj, time, props)
    local tw = TweenService:Create(obj, TweenInfo.new(time or 0.2, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), props)
    tw:Play()
    return tw
end

local function Gradient(c1, c2, rot, parent)
    return Create("UIGradient", {Color = ColorSequence.new(c1, c2), Rotation = rot or 90, Parent = parent})
end

local function MakeDraggable(frame, dragArea)
    dragArea = dragArea or frame
    local dragging, dragStart, startPos
    dragArea.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            dragStart = input.Position
            startPos = frame.Position
            input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then dragging = false end
            end)
        end
    end)
    UserInputService.InputChanged:Connect(function(input)
        if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
            local delta = input.Position - dragStart
            frame.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
        end
    end)
end

local ConfigFlags = {}
local ConfigPath = "BeautifulUI_Config.json"

function BeautifulUI:SetConfigPath(p) ConfigPath = p end

function BeautifulUI:SaveConfig()
    local data = {}
    for k, v in pairs(ConfigFlags) do
        if v.Get then data[k] = v:Get() end
    end
    local ok, encoded = pcall(HttpService.JSONEncode, HttpService, data)
    if ok then
        pcall(function()
            if writefile then writefile(ConfigPath, encoded) end
        end)
        BeautifulUI:Notify("Конфиг", "Сохранено ✅", 2, "Success")
    end
end

function BeautifulUI:LoadConfig()
    local ok, content = pcall(function()
        if readfile and isfile and isfile(ConfigPath) then return readfile(ConfigPath) end
    end)
    if not ok or not content then
        BeautifulUI:Notify("Конфиг", "Файл не найден", 2, "Warning")
        return
    end
    local ok2, data = pcall(HttpService.JSONDecode, HttpService, content)
    if not ok2 then return end
    for k, v in pairs(data) do
        if ConfigFlags[k] then ConfigFlags[k]:Set(v) end
    end
    BeautifulUI:Notify("Конфиг", "Загружено ✅", 2, "Success")
end

function BeautifulUI:DeleteConfig()
    pcall(function()
        if delfile and isfile and isfile(ConfigPath) then delfile(ConfigPath) end
    end)
end

local NotifyGui, NotifyContainer

local function EnsureNotifyGui()
    if NotifyGui and NotifyGui.Parent then return end
    NotifyGui = Create("ScreenGui", { Name = "BeautifulUINotify", ResetOnSpawn = false, ZIndexBehavior = Enum.ZIndexBehavior.Sibling, Parent = (gethui and gethui()) or CoreGui })
    NotifyContainer = Create("Frame", { Size = UDim2.new(0, 340, 1, 0), Position = UDim2.new(1, -360, 0, 0), BackgroundTransparency = 1, Parent = NotifyGui })
    Create("UIListLayout", { Padding = UDim.new(0, 10), VerticalAlignment = Enum.VerticalAlignment.Bottom, SortOrder = Enum.SortOrder.LayoutOrder, Parent = NotifyContainer })
    Create("UIPadding", { PaddingBottom = UDim.new(0, 20), Parent = NotifyContainer })
end

local NotifyCount = 0
function BeautifulUI:Notify(title, text, duration, notifType, buttons)
    EnsureNotifyGui()
    duration = duration or 3
    notifType = notifType or "Info"
    NotifyCount = NotifyCount + 1

    local colors = { Info = Theme.Info, Success = Theme.Success, Warning = Theme.Warning, Danger = Theme.Danger, Error = Theme.Danger }
    local icons = { Info = "ℹ️", Success = "✅", Warning = "⚠️", Danger = "❌", Error = "❌" }
    local accentColor = colors[notifType] or Theme.Accent
    local icon = icons[notifType] or "🔔"
    local hasButtons = buttons and #buttons > 0
    local height = hasButtons and 125 or 85

    local notify = Create("Frame", {
        Size = UDim2.new(1, 0, 0, height),
        Position = UDim2.new(1, 360, 0, 0),
        BackgroundColor3 = Theme.Secondary,
        BorderSizePixel = 0,
        LayoutOrder = NotifyCount,
        Parent = NotifyContainer
    })
    Corner(10, notify)
    Stroke(accentColor, 1.5, 0.3, notify)

    local bar = Create("Frame", { Size = UDim2.new(0, 4, 1, 0), BackgroundColor3 = accentColor, BorderSizePixel = 0, Parent = notify })
    Corner(2, bar)

    Create("TextLabel", { Size = UDim2.new(0, 24, 0, 24), Position = UDim2.new(0, 14, 0, 12), BackgroundTransparency = 1, Font = Enum.Font.GothamBold, Text = icon, TextColor3 = accentColor, TextSize = 16, TextXAlignment = Enum.TextXAlignment.Left, Parent = notify })
    Create("TextLabel", { Size = UDim2.new(1, -50, 0, 20), Position = UDim2.new(0, 40, 0, 12), BackgroundTransparency = 1, Font = Enum.Font.GothamBold, Text = title or "Уведомление", TextColor3 = Theme.Text, TextSize = 14, TextXAlignment = Enum.TextXAlignment.Left, Parent = notify })
    Create("TextLabel", { Size = UDim2.new(1, -50, 0, 40), Position = UDim2.new(0, 40, 0, 34), BackgroundTransparency = 1, Font = Enum.Font.Gotham, Text = text or "", TextColor3 = Theme.SubText, TextSize = 12, TextWrapped = true, TextXAlignment = Enum.TextXAlignment.Left, TextYAlignment = Enum.TextYAlignment.Top, Parent = notify })

    -- Кнопки
    if hasButtons then
        local btnHolder = Create("Frame", {
            Size = UDim2.new(1, -20, 0, 30),
            Position = UDim2.new(0, 10, 1, -40),
            BackgroundTransparency = 1,
            Parent = notify
        })
        Create("UIListLayout", { FillDirection = Enum.FillDirection.Horizontal, Padding = UDim.new(0, 6), HorizontalAlignment = Enum.HorizontalAlignment.Right, SortOrder = Enum.SortOrder.LayoutOrder, Parent = btnHolder })

        for i, btnData in ipairs(buttons) do
            local btnColor = btnData.Color or Theme.Accent
            local btn = Create("TextButton", {
                Size = UDim2.new(0, 100, 1, 0),
                BackgroundColor3 = btnColor,
                BorderSizePixel = 0,
                Font = Enum.Font.GothamBold,
                Text = btnData.Text or "OK",
                TextColor3 = Theme.Text,
                TextSize = 12,
                LayoutOrder = i,
                Parent = btnHolder
            })
            Corner(6, btn)
            btn.MouseButton1Click:Connect(function()
                if btnData.Callback then pcall(btnData.Callback) end
                Tween(notify, 0.3, {Position = UDim2.new(1, 360, 0, 0)})
                task.wait(0.35)
                notify:Destroy()
            end)
        end
    else
        local progressBar = Create("Frame", { Size = UDim2.new(1, -20, 0, 2), Position = UDim2.new(0, 10, 1, -6), BackgroundColor3 = Theme.Tertiary, BorderSizePixel = 0, Parent = notify })
        Corner(1, progressBar)
        local progressFill = Create("Frame", { Size = UDim2.new(1, 0, 1, 0), BackgroundColor3 = accentColor, BorderSizePixel = 0, Parent = progressBar })
        Corner(1, progressFill)
        Tween(progressFill, duration, {Size = UDim2.new(0, 0, 1, 0)})

        task.delay(duration, function()
            Tween(notify, 0.3, {Position = UDim2.new(1, 360, 0, 0)})
            task.wait(0.35)
            notify:Destroy()
        end)
    end

    Tween(notify, 0.3, {Position = UDim2.new(0, 0, 0, 0)})
end

function BeautifulUI:CreateWindow(config)
    config = config or {}
    local title = config.Title or "Beautiful UI"
    local subtitle = config.Subtitle or "v3.0"
    local size = config.Size or UDim2.new(0, IsMobile and 480 or 700, 0, IsMobile and 340 or 500)
    local toggleKey = config.ToggleKey or Enum.KeyCode.RightShift
    local initialTheme = config.Theme or "Purple"
    local windowIcon = config.Icon or "⚡"

    if Themes[initialTheme] then Theme = Themes[initialTheme] end

    local gui = Create("ScreenGui", {
        Name = "BeautifulUI_" .. tick(),
        ResetOnSpawn = false,
        ZIndexBehavior = Enum.ZIndexBehavior.Sibling,
        Parent = (gethui and gethui()) or CoreGui
    })

    -- 📌 Иконка вызова окна (появляется когда окно свёрнуто)
    local iconButton = Create("TextButton", {
        Name = "IconButton",
        Size = UDim2.new(0, 44, 0, 44),
        Position = UDim2.new(0.5, -22, 0, 12),
        BackgroundColor3 = Theme.Accent,
        BorderSizePixel = 0,
        Font = Enum.Font.GothamBold,
        Text = windowIcon,
        TextColor3 = Theme.Text,
        TextSize = 22,
        Visible = false,
        Parent = gui
    })
    Corner(12, iconButton)
    local iconStroke = Stroke(Theme.Text, 2, 0.4, iconButton)

    -- Пульсация иконки
    task.spawn(function()
        while iconButton.Parent do
            if iconButton.Visible then
                Tween(iconStroke, 0.8, {Transparency = 0.7})
                task.wait(0.8)
                Tween(iconStroke, 0.8, {Transparency = 0.3})
                task.wait(0.8)
            else
                task.wait(0.5)
            end
        end
    end)

    -- Анимация появления иконки
    local function ShowIcon()
        iconButton.Position = UDim2.new(0.5, -22, 0, -60)
        iconButton.Visible = true
        Tween(iconButton, 0.4, {Position = UDim2.new(0.5, -22, 0, 12)})
    end

    local function HideIcon()
        Tween(iconButton, 0.3, {Position = UDim2.new(0.5, -22, 0, -60)})
        task.wait(0.3)
        iconButton.Visible = false
    end

    -- Основное окно
    local main = Create("Frame", {
        Name = "Main",
        Size = UDim2.new(0, 0, 0, 0),
        Position = UDim2.new(0.5, -size.X.Offset/2, 0.5, -size.Y.Offset/2),
        BackgroundColor3 = Theme.Background,
        BorderSizePixel = 0,
        Parent = gui
    })
    Corner(14, main)
    Stroke(Theme.Accent, 1.5, 0.4, main)
    Tween(main, 0.3, {Size = size})

    Gradient(Theme.Gradient1, Theme.Gradient2, 90, main)

    -- Топбар
    local topBar = Create("Frame", { Name = "TopBar", Size = UDim2.new(1, 0, 0, 55), BackgroundTransparency = 1, Parent = main })
    Create("TextLabel", { Size = UDim2.new(0.7, 0, 0, 24), Position = UDim2.new(0, 20, 0, 8), BackgroundTransparency = 1, Font = Enum.Font.GothamBold, Text = title, TextColor3 = Theme.Text, TextSize = 20, TextXAlignment = Enum.TextXAlignment.Left, Parent = topBar })
    Create("TextLabel", { Size = UDim2.new(0.7, 0, 0, 16), Position = UDim2.new(0, 20, 0, 32), BackgroundTransparency = 1, Font = Enum.Font.Gotham, Text = subtitle, TextColor3 = Theme.SubText, TextSize = 11, TextXAlignment = Enum.TextXAlignment.Left, Parent = topBar })

    -- Кнопки управления
    local minimizeBtn = Create("TextButton", { Size = UDim2.new(0, 30, 0, 30), Position = UDim2.new(1, -78, 0, 12), BackgroundColor3 = Theme.Tertiary, BorderSizePixel = 0, Font = Enum.Font.GothamBold, Text = "—", TextColor3 = Theme.SubText, TextSize = 14, Parent = topBar })
    Corner(8, minimizeBtn)

    local collapseBtn = Create("TextButton", { Size = UDim2.new(0, 30, 0, 30), Position = UDim2.new(1, -114, 0, 12), BackgroundColor3 = Theme.Tertiary, BorderSizePixel = 0, Font = Enum.Font.GothamBold, Text = "▣", TextColor3 = Theme.SubText, TextSize = 14, Parent = topBar })
    Corner(8, collapseBtn)

    local closeBtn = Create("TextButton", { Size = UDim2.new(0, 30, 0, 30), Position = UDim2.new(1, -42, 0, 12), BackgroundColor3 = Theme.Tertiary, BorderSizePixel = 0, Font = Enum.Font.GothamBold, Text = "✕", TextColor3 = Theme.SubText, TextSize = 14, Parent = topBar })
    Corner(8, closeBtn)

    -- Hover-эффекты
    minimizeBtn.MouseEnter:Connect(function() Tween(minimizeBtn, 0.15, {BackgroundColor3 = Theme.Warning, TextColor3 = Theme.Text}) end)
    minimizeBtn.MouseLeave:Connect(function() Tween(minimizeBtn, 0.15, {BackgroundColor3 = Theme.Tertiary, TextColor3 = Theme.SubText}) end)
    collapseBtn.MouseEnter:Connect(function() Tween(collapseBtn, 0.15, {BackgroundColor3 = Theme.Accent, TextColor3 = Theme.Text}) end)
    collapseBtn.MouseLeave:Connect(function() Tween(collapseBtn, 0.15, {BackgroundColor3 = Theme.Tertiary, TextColor3 = Theme.SubText}) end)
    closeBtn.MouseEnter:Connect(function() Tween(closeBtn, 0.15, {BackgroundColor3 = Theme.Danger, TextColor3 = Theme.Text}) end)
    closeBtn.MouseLeave:Connect(function() Tween(closeBtn, 0.15, {BackgroundColor3 = Theme.Tertiary, TextColor3 = Theme.SubText}) end)

    closeBtn.MouseButton1Click:Connect(function()
        Tween(main, 0.25, {Size = UDim2.new(0, 0, 0, 0), Position = UDim2.new(0.5, 0, 0.5, 0)})
        task.wait(0.26)
        gui:Destroy()
    end)

    -- 📌 Сворачивание в иконку
    collapseBtn.MouseButton1Click:Connect(function()
        Tween(main, 0.3, {Size = UDim2.new(0, 0, 0, 0), Position = UDim2.new(0.5, 0, 0.5, 0)})
        task.wait(0.31)
        main.Visible = false
        ShowIcon()
    end)

    iconButton.MouseButton1Click:Connect(function()
        HideIcon()
        task.wait(0.05)
        main.Visible = true
        main.Position = UDim2.new(0.5, -size.X.Offset/2, 0.5, -size.Y.Offset/2)
        Tween(main, 0.3, {Size = size})
    end)

    -- Минимизация (в полоску)
    local minimized = false
    local prevSize = size
    minimizeBtn.MouseButton1Click:Connect(function()
        minimized = not minimized
        if minimized then
            prevSize = main.Size
            Tween(main, 0.25, {Size = UDim2.new(prevSize.X.Scale, prevSize.X.Offset, 0, 55)})
        else
            Tween(main, 0.25, {Size = prevSize})
        end
    end)

    MakeDraggable(main, topBar)

    Create("Frame", { Size = UDim2.new(1, -40, 0, 1), Position = UDim2.new(0, 20, 0, 55), BackgroundColor3 = Theme.Tertiary, BorderSizePixel = 0, Parent = main })

    local content = Create("Frame", { Name = "Content", Size = UDim2.new(1, -20, 1, -75), Position = UDim2.new(0, 10, 0, 65), BackgroundTransparency = 1, Parent = main })

    Corner(10, sidebar)
    Create("UIListLayout", { Padding = UDim.new(0, 6), SortOrder = Enum.SortOrder.LayoutOrder, HorizontalAlignment = Enum.HorizontalAlignment.Center, Parent = sidebar })
    Create("UIPadding", { PaddingTop = UDim.new(0, 8), PaddingLeft = UDim.new(0, 8), PaddingRight = UDim.new(0, 8), Parent = sidebar })

    -- Глобальный поиск вкладок
    local searchBox = Create("TextBox", {
        Size = UDim2.new(1, 0, 0, 30),
        BackgroundColor3 = Theme.Tertiary,
        BorderSizePixel = 0,
        Font = Enum.Font.Gotham,
        Text = "",
        PlaceholderText = "🔍 Поиск...",
        PlaceholderColor3 = Theme.SubText,
        TextColor3 = Theme.Text,
        TextSize = 12,
        ClearTextOnFocus = false,
        LayoutOrder = -1,
        Parent = sidebar
    })
    Corner(6, searchBox)
    Create("UIPadding", { PaddingLeft = UDim.new(0, 8), Parent = searchBox })

    local pagesContainer = Create("Frame", { Name = "Pages", Size = UDim2.new(1, -160, 1, 0), Position = UDim2.new(0, 160, 0, 0), BackgroundTransparency = 1, Parent = content })

    local Window = {
        Gui = gui, Main = main, Sidebar = sidebar, PagesContainer = pagesContainer,
        Tabs = {}, ActiveTab = nil, ToggleKey = toggleKey
    }

    searchBox:GetPropertyChangedSignal("Text"):Connect(function()
        local q = searchBox.Text:lower()
        for _, t in pairs(Window.Tabs) do
            local match = t.Name:lower():find(q, 1, true)
            t.Button.Visible = (q == "") or (match ~= nil)
        end
    end)

    -- ═══ ВКЛАДКА ═══
    function Window:Tab(tabName, icon)

            Name = tabName .. "Tab", Size = UDim2.new(1, 0, 0, 36),

            Text = "  " .. (icon and icon .. "  " or "") .. tabName,
            TextColor3 = Theme.SubText, TextSize = 13,
            TextXAlignment = Enum.TextXAlignment.Left, Parent = sidebar
        })
        Corner(8, tabButton)

        local page = Create("ScrollingFrame", {

        function TabAPI:_mkButton(parent, text, callback, style, icon)
            style = style or "Primary"
            local bgColor, hoverColor, textColor = Theme.Secondary, Theme.Accent, Theme.Text

        function TabAPI:_mkSlider(parent, text, min, max, default, callback, flag)
            min, max = min or 0, max or 100
            local value = default or min
            local frame = Create("Frame", { Size = UDim2.new(1, 0, 0, 50), BackgroundColor3 = Theme.Secondary, BorderSizePixel = 0, Parent = parent })
            Corner(8, frame)
            Create("TextLabel", { Size = UDim2.new(0.7, 0, 0, 20), Position = UDim2.new(0, 12, 0, 6), BackgroundTransparency = 1, Font = Enum.Font.GothamMedium, Text = text, TextColor3 = Theme.Text, TextSize = 13, TextXAlignment = Enum.TextXAlignment.Left, Parent = frame })
            local valueLabel = Create("TextLabel", { Size = UDim2.new(0.3, -12, 0, 20), Position = UDim2.new(0.7, 0, 0, 6), BackgroundTransparency = 1, Font = Enum.Font.GothamBold, Text = tostring(value), TextColor3 = Theme.Accent, TextSize = 13, TextXAlignment = Enum.TextXAlignment.Right, Parent = frame })
            local barBg = Create("Frame", { Size = UDim2.new(1, -24, 0, 6), Position = UDim2.new(0, 12, 0, 34), BackgroundColor3 = Theme.Tertiary, BorderSizePixel = 0, Parent = frame })
            Corner(3, barBg)
            local fill = Create("Frame", { Size = UDim2.new((value - min) / (max - min), 0, 1, 0), BackgroundColor3 = Theme.Accent, BorderSizePixel = 0, Parent = barBg })
            Corner(3, fill)

            local dragging = false
            local function setValueFromX(x)
                local rel = math.clamp((x - barBg.AbsolutePosition.X) / barBg.AbsoluteSize.X, 0, 1)
                value = math.floor(min + (max - min) * rel + 0.5)
                fill.Size = UDim2.new(rel, 0, 1, 0)
                valueLabel.Text = tostring(value)
                if callback then pcall(callback, value) end
            end

            barBg.InputBegan:Connect(function(input)
                if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
                    dragging = true; setValueFromX(input.Position.X)
                end
            end)
            UserInputService.InputChanged:Connect(function(input)
                if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then setValueFromX(input.Position.X) end
            end)
            UserInputService.InputEnded:Connect(function(input)
                if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then dragging = false end
            end)

            local api = { Type = "Slider",
                Set = function(v) value = math.clamp(v, min, max); local rel = (value - min) / (max - min); fill.Size = UDim2.new(rel, 0, 1, 0); valueLabel.Text = tostring(value); if callback then pcall(callback, value) end end,
                Get = function() return value end }
            if flag then ConfigFlags[flag] = api end
            return api
        end

            Corner(6, canvas)

            -- Сетка
            for i = 1, 4 do
                Create("Frame", {
                    Size = UDim2.new(1, 0, 0, 1),
                    Position = UDim2.new(0, 0, i/5, 0),
                    BackgroundColor3 = Theme.Secondary,
                    BackgroundTransparency = 0.5,
                    BorderSizePixel = 0,
                    Parent = canvas
                })
            end

            local data = {}
            local maxPoints = 50

            -- Для Line/Bar
            local barsContainer = Create("Frame", {
                Size = UDim2.new(1, 0, 1, 0),
                BackgroundTransparency = 1,
                Parent = canvas
            })
            Create("UIListLayout", {
                FillDirection = Enum.FillDirection.Horizontal,
                HorizontalAlignment = Enum.HorizontalAlignment.Left,
                VerticalAlignment = Enum.VerticalAlignment.Bottom,
                SortOrder = Enum.SortOrder.LayoutOrder,
                Padding = UDim.new(0, 2),
                Parent = barsContainer
            })
            Create("UIPadding", { PaddingLeft = UDim.new(0, 4), PaddingRight = UDim.new(0, 4), PaddingBottom = UDim.new(0, 4), Parent = barsContainer })

            -- Pie container
            local pieContainer = Create("Frame", {
                Size = UDim2.new(1, 0, 1, 0),
                BackgroundTransparency = 1,
                Visible = (graphType == "Pie"),
                Parent = canvas
            })

            barsContainer.Visible = (graphType ~= "Pie")

            local function redraw()
                -- Очищаем Line (только Bars остаются)
                for _, c in ipairs(barsContainer:GetChildren()) do
                    if c:IsA("Frame") then c:Destroy() end
                end

                if graphType == "Bar" or graphType == "Line" then
                    for i, v in ipairs(data) do
                        local barFrame = Create("Frame", {
                            Size = UDim2.new(0, 0, v / 100, 0),
                            BackgroundColor3 = Theme.Accent,
                            BorderSizePixel = 0,
                            LayoutOrder = i,
                            Parent = barsContainer
                        })
                        Corner(2, barFrame)
                        local targetWidth = math.floor((canvas.AbsoluteSize.X - 12) / math.max(#data, 1)) - 3
                        Tween(barFrame, 0.2, {Size = UDim2.new(0, math.max(targetWidth, 3), v / 100, 0)})
                    end
                elseif graphType == "Pie" then
                    for _, c in ipairs(pieContainer:GetChildren()) do c:Destroy() end
                    if #data == 0 then return end
                    local total = 0
                    for _, v in ipairs(data) do total = total + v end
                    if total == 0 then return end

                    local size = math.min(canvas.AbsoluteSize.X, canvas.AbsoluteSize.Y) - 20
                    local frameSize = UDim2.new(0, size, 0, size)
                    local startAngle = -90
                    local palette = {Theme.Accent, Theme.Success, Theme.Warning, Theme.Danger, Theme.Info, Color3.fromRGB(180,100,255)}

                    for i, v in ipairs(data) do
                        local pct = v / total
                        local angle = pct * 360
                        local slice = Create("Frame", {
                            Size = frameSize,
                            Position = UDim2.new(0.5, -size/2, 0.5, -size/2),
                            BackgroundColor3 = palette[((i-1) % #palette) + 1],
                            BorderSizePixel = 0,
                            BackgroundTransparency = 0,
                            Parent = pieContainer
                        })
                        Corner(size/2, slice)
                        -- Имитация pie через ZIndex и повороты
                        slice.ZIndex = 10 - i
                        slice.Rotation = startAngle
                        slice.Size = UDim2.new(0, size, 0, size * pct)
                        slice.Position = UDim2.new(0.5, -size/2, 0.5, -size/2)
                        slice.AnchorPoint = Vector2.new(0.5, 0.5)
                        slice.ClipsDescendants = true
                        startAngle = startAngle + angle
                    end
                end
            end

            -- Обновление при изменении размера
            canvas:GetPropertyChangedSignal("AbsoluteSize"):Connect(function() task.wait(0.05); redraw() end)

            return {
                -- Добавить точку
                AddPoint = function(value)
                    value = math.clamp(value, 0, 100)
                    table.insert(data, value)
                    if #data > maxPoints then table.remove(data, 1) end
                    redraw()
                end,
                -- Установить массив
                SetData = function(arr)
                    data = {}
                    for _, v in ipairs(arr) do table.insert(data, math.clamp(v, 0, 100)) end
                    redraw()
                end,
                GetData = function() return data end,
                Clear = function() data = {}; redraw() end,
                -- Добавлять случайные точки для тестов
                AutoUpdate = function(interval, min, max)
                    min, max = min or 0, max or 100
                    task.spawn(function()
                        while canvas.Parent do
                            table.insert(data, math.random(min, max))
                            if #data > maxPoints then table.remove(data, 1) end
                            redraw()
                            task.wait(interval or 1)
                        end
                    end)
                end
            }
        end

        -- 🗂️ Под-вкладки (вложенные страницы)
        function TabAPI:SubTab(name, icon)
            local header = Create("TextButton", {
                Size = UDim2.new(1, 0, 0, 32),
                BackgroundColor3 = Theme.Tertiary,
                BorderSizePixel = 0,
                Font = Enum.Font.GothamMedium,
                Text = "  " .. (icon and icon .. "  " or "") .. name,
                TextColor3 = Theme.Text,
                TextSize = 13,
                TextXAlignment = Enum.TextXAlignment.Left,
                Parent = page
            })
            Corner(6, header)

            local subPage = Create("Frame", {
                Size = UDim2.new(1, 0, 0, 0),
                BackgroundTransparency = 1,
                ClipsDescendants = true,
                Parent = page
            })
            Create("UIListLayout", { Padding = UDim.new(0, 6), SortOrder = Enum.SortOrder.LayoutOrder, Parent = subPage })
            Create("UIPadding", { PaddingLeft = UDim.new(0, 10), PaddingRight = UDim.new(0, 6), PaddingTop = UDim.new(0, 6), PaddingBottom = UDim.new(0, 6), Parent = subPage })

            local open = true
            local function updateHeight()
                local layout = subPage:FindFirstChildOfClass("UIListLayout")
                local h = layout.AbsoluteContentSize.Y + 12
                Tween(subPage, 0.25, {Size = UDim2.new(1, 0, 0, open and h or 0)})
            end

            header.MouseButton1Click:Connect(function()
                open = not open
                updateHeight()
            end)

            subPage.ChildAdded:Connect(function() task.wait(0.05); updateHeight() end)
            subPage.ChildRemoved:Connect(function() task.wait(0.05); updateHeight() end)
            task.defer(updateHeight)

            -- Возвращаем API такой же как TabAPI, но с контейнером subPage
            local SubAPI = {}
            local builders = {"_mkButton", "_mkToggle", "_mkSlider", "_mkInput", "_mkLabel", "_mkDropdown"}
            for _, bName in ipairs(builders) do
                SubAPI[bName] = function(self, ...) return TabAPI[bName](self, subPage, ...) end
            end

            SubAPI.Button = function(self, text, cb, style, icon) return TabAPI:_mkButton(subPage, text, cb, style, icon) end
            SubAPI.Toggle = function(self, text, def, cb, flag) return TabAPI:_mkToggle(subPage, text, def, cb, flag) end
            SubAPI.Slider = function(self, text, mi, ma, de, cb, flag) return TabAPI:_mkSlider(subPage, text, mi, ma, de, cb, flag) end
            SubAPI.Input = function(self, text, ph, cb, flag) return TabAPI:_mkInput(subPage, text, ph, cb, flag) end
            SubAPI.Label = function(self, text, color) return TabAPI:_mkLabel(subPage, text, color) end
            SubAPI.Dropdown = function(self, text, opts, cb, flag) return TabAPI:_mkDropdown(subPage, text, opts, cb, flag) end
            SubAPI.ImageButton = function(self, text, imgId, cb)
                local btn = Create("TextButton", { Size = UDim2.new(1, 0, 0, 50), BackgroundColor3 = Theme.Secondary, BorderSizePixel = 0, Text = "", Parent = subPage })
                Corner(6, btn)
                Create("ImageLabel", { Size = UDim2.new(0, 36, 0, 36), Position = UDim2.new(0, 8, 0.5, -18), BackgroundTransparency = 1, Image = imgId, Parent = btn })
                Create("TextLabel", { Size = UDim2.new(1, -60, 1, 0), Position = UDim2.new(0, 54, 0, 0), BackgroundTransparency = 1, Font = Enum.Font.Gotham, Text = text, TextColor3 = Theme.Text, TextSize = 13, TextXAlignment = Enum.TextXAlignment.Left, Parent = btn })
                btn.MouseButton1Click:Connect(function() if cb then pcall(cb) end end)
                return btn
            end

            return SubAPI
        end

        -- 🔍 Локальный поиск по элементам внутри вкладки
        function TabAPI:SearchBar()
            local searchFrame = Create("Frame", {
                Size = UDim2.new(1, 0, 0, 34),
                BackgroundColor3 = Theme.Tertiary,
                BorderSizePixel = 0,
                Parent = page
            })
            Corner(6, searchFrame)
            local box = Create("TextBox", {
                Size = UDim2.new(1, -40, 1, 0),
                Position = UDim2.new(0, 34, 0, 0),
                BackgroundTransparency = 1,
                Font = Enum.Font.Gotham,
                Text = "",
                PlaceholderText = "Поиск элементов...",
                PlaceholderColor3 = Theme.SubText,
                TextColor3 = Theme.Text,
                TextSize = 12,
                TextXAlignment = Enum.TextXAlignment.Left,
                ClearTextOnFocus = false,
                Parent = searchFrame
            })
            Create("TextLabel", {
                Size = UDim2.new(0, 30, 1, 0),
                Position = UDim2.new(0, 6, 0, 0),
                BackgroundTransparency = 1,
                Font = Enum.Font.GothamBold,
                Text = "🔍",
                TextColor3 = Theme.SubText,
                TextSize = 14,
                Parent = searchFrame
            })

            local tracked = {}
            local function trackElement(el)
                if el and el:IsA("GuiObject") then table.insert(tracked, el) end
                return el
            end

            -- Автоматически отслеживаем новые элементы
            page.ChildAdded:Connect(function(child)
                if child ~= searchFrame then trackElement(child) end
            end)

            box:GetPropertyChangedSignal("Text"):Connect(function()
                local q = box.Text:lower()
                for _, el in ipairs(tracked) do
                    if el.Parent then
                        local labelText = ""
                        if el:IsA("TextButton") or el:IsA("TextLabel") then labelText = el.Text
                        else
                            local lbl = el:FindFirstChildOfClass("TextLabel") or el:FindFirstChildOfClass("TextButton")
                            if lbl then labelText = lbl.Text end
                        end
                        el.Visible = (q == "") or (labelText:lower():find(q, 1, true) ~= nil)
                    end
                end
            end)

            return { Frame = searchFrame, Box = box }
        end

        return TabAPI
    end

    function Window:Notify(t, txt, dur, tp, btns)
        BeautifulUI:Notify(t, txt, dur, tp, btns)
    end

    -- Горячая клавиша
    local visible = true
    UserInputService.InputBegan:Connect(function(input, gpe)
        if gpe then return end
        if input.KeyCode == toggleKey then
            visible = not visible
            if not visible then
                Tween(main, 0.25, {Size = UDim2.new(0, 0, 0, 0), Position = UDim2.new(0.5, 0, 0.5, 0)})
                task.wait(0.26)
                main.Visible = false
                ShowIcon()
            else
                HideIcon()
                task.wait(0.05)
                main.Visible = true
                main.Position = UDim2.new(0.5, -size.X.Offset/2, 0.5, -size.Y.Offset/2)
                Tween(main, 0.3, {Size = size})
            end
        end
    end)

    return Window
