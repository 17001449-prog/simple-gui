-- 🅰️ Созданные объекты с шрифтом запоминаем: так потом можно применить свой шрифт
-- ко всему интерфейсу разом (включая элементы, созданные до смены шрифта).
local FontRegistry = {}
local HasFontFace = nil -- определяется один раз (FontFace существует не во всех средах)

local function Create(className, props)
    props = props or {}
    local inst = Instance.new(className)
    -- если свой шрифт уже активен, новый элемент сразу создаём через FontFace,
    -- иначе Roblox при присваивании Font сбросит наш FontFace
    local wantFace = (props.FontFace ~= nil) or (FontConfig.Active and not props.Font)
    for k, v in pairs(props) do
        if k ~= "FontFace" then inst[k] = v end
    end
    if wantFace then
        if HasFontFace == nil then
            HasFontFace = pcall(function() local t = Instance.new("TextLabel"); t.FontFace = Font.new("rbxasset://fonts/families/GothamSSm.json") end)
        end
        if HasFontFace then inst.FontFace = props.FontFace end
    end
    if inst:IsA("TextLabel") or inst:IsA("TextButton") or inst:IsA("TextBox") then
        -- снимок состояния ДО применения своего шрифта
        local fontName = inst.Font and tostring(inst.Font.Name) or ""
        FontRegistry[inst] = {
            original = inst.Font,
            baseSize = inst.TextSize,
            bold = fontName:find("Bold") ~= nil or fontName:find("Black") ~= nil or fontName:find("Semibold") ~= nil or fontName:find("SemiBold") ~= nil,
            mono = fontName == "Code",
        }
        -- шрифт уже выбран → новый элемент получает его сразу, а не после
        -- следующего SetCustomFont (иначе свежие элементы остаются старыми)
        if FontConfig.Active and applyFontTo then
            pcall(applyFontTo, inst)
        end
    end
    return inst
end

local function Corner(radius, parent)
    return Create("UICorner", {CornerRadius = UDim.new(0, radius or 8), Parent = parent})
end

local function Stroke(color, thickness, transparency, parent)
    return Create("UIStroke", {Color = C(color, Theme.Accent), Thickness = thickness or 1, Transparency = transparency or 0.5, Parent = parent})
end

local function Tween(obj, time, props)
    local tw = TweenService:Create(obj, TweenInfo.new(time or 0.2, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), props)
    tw:Play()
    return tw
end

local function Gradient(c1, c2, rot, parent)
    return Create("UIGradient", {Color = ColorSequence.new(C(c1), C(c2)), Rotation = rot or 90, Parent = parent})
end

-- Мягкая тень под окном (без внешних ассетов)
local function Shadow(size, parent, strong)
    local sh = Create("ImageLabel", {
        Name = "Shadow", Size = size, Position = UDim2.new(0.5, 0, 0.5, 0),
        AnchorPoint = Vector2.new(0.5, 0.5), BackgroundTransparency = 1,
        Image = "rbxassetid://1316045217", ImageColor3 = Color3.new(0, 0, 0),
        ImageTransparency = strong and 0.6 or 0.75, ScaleType = Enum.ScaleType.Slice,
        SliceCenter = Rect.new(10, 10, 118, 118), ZIndex = 0, Parent = parent,
    })
    return sh
end

-- Callback-обёртка, чтобы не ронять UI при ошибке в пользовательском коде
local function Safe(fn, ...)
    if not fn then return end
    local ok, err = pcall(fn, ...)
    if not ok then warn("[BeautifulUI] callback error:", err) end
end

-- 🖱️ ЕДИНЫЙ МЕНЕДЖЕР ПЕРЕТАСКИВАНИЯ (слайдеры, палитры цвета)
-- ⚡ Было: 3 подключения UserInputService на каждый элемент.

local DragTargets = {}
local DragManagerReady = false

local function ClearDeadDrags()
    local dead = {}
    for obj in pairs(DragTargets) do
        if not obj.Parent then table.insert(dead, obj) end
    end
    for _, obj in ipairs(dead) do DragTargets[obj] = nil end
end

local function InitDragManager()
    if DragManagerReady then return end
    DragManagerReady = true
    UserInputService.InputChanged:Connect(function(input)
        local t = input.UserInputType
        if t ~= Enum.UserInputType.MouseMovement and t ~= Enum.UserInputType.Touch then return end
        ClearDeadDrags()
        for obj, handler in pairs(DragTargets) do
            Safe(handler, input.Position.X, input.Position.Y)
        end
    end)
    UserInputService.InputEnded:Connect(function(input)
        local t = input.UserInputType
        if t == Enum.UserInputType.MouseButton1 or t == Enum.UserInputType.Touch then
            for obj in pairs(DragTargets) do DragTargets[obj] = nil end
        end
    end)
end

-- Перетаскивание с «захватом» курсора: TrackDrag(element, function(x, y) ... end)
local function TrackDrag(obj, onMove)
    InitDragManager()
    obj.InputBegan:Connect(function(input)
        local t = input.UserInputType
        if t == Enum.UserInputType.MouseButton1 or t == Enum.UserInputType.Touch then
            DragTargets[obj] = onMove
            Safe(onMove, input.Position.X, input.Position.Y)
        end
    end)
end

-- Универсальный «распаковщик» аргументов для el:Set(x) (вызов через двоеточие):
-- в этом случае первым аргументом приходит сам объект-элемент.
local function patchSet(api, impl)
    api.Set = function(a, b)
        if a == api then return impl(b) end
        return impl(a, b)
    end
    return api
end

local KeybindEntries = {}
local KeybindManagerReady = false

local function InitKeybindManager()
    if KeybindManagerReady then return end
    KeybindManagerReady = true
    UserInputService.InputBegan:Connect(function(input, gpe)
        if gpe then return end
        if input.UserInputType ~= Enum.UserInputType.Keyboard then return end
        local focused = UserInputService:GetFocusedTextBox()
        local dead = {}
        for obj, entry in pairs(KeybindEntries) do
            if not obj.Parent then
                table.insert(dead, obj)
            else
                Safe(entry, input, focused)
            end
        end
        for _, obj in ipairs(dead) do KeybindEntries[obj] = nil end
    end)
end

local function RegisterKeybind(obj, handler)
    InitKeybindManager()
    KeybindEntries[obj] = handler
end

-- ✨ Мигание элемента (подсветка при переходе из поиска/палитры)
local function FlashElement(obj, color, times)
    if not obj or not obj.Parent then return end
    local fl = Instance.new("UIStroke")
    fl.Color = color or Theme.Accent
    fl.Thickness = 2
    fl.Transparency = 0
    fl.Parent = obj
    task.spawn(function()
        for i = 1, (times or 2) do
            Tween(fl, 0.18, {Transparency = 1})
            task.wait(0.2)
            Tween(fl, 0.18, {Transparency = 0})
            task.wait(0.2)
        end
        Tween(fl, 0.25, {Transparency = 1})
        task.wait(0.3)
        if fl.Parent then fl:Destroy() end
    end)
end
BeautifulUI.FlashElement = FlashElement

-- ⌨️ Глобальный реестр горячих клавиш (для тоста-подсказки)
local Hotkeys = {}

-- Регистрирует клавишу, которой управляет пользовательский скрипт.
-- showToast = true добавляет её в памятку по Ctrl+.
function BeautifulUI:RegisterHotkey(key, label, showToast)
    table.insert(Hotkeys, {Key = key, Label = label or "—", Show = showToast ~= false})
end

local function KeyDisplay(key)
    if typeof(key) == "EnumItem" then return key.Name end
    return tostring(key)
end

function BeautifulUI:ShowHotkeys()
    local self_ = self
    if type(self_) ~= "table" then self_ = BeautifulUI end
    local lines = {
        "Ctrl+K — палитра команд",
        "Ctrl+. — эта памятка",
        "Ctrl+Shift+R — сброс настроек",
        "Ваша кнопка окна — показать/скрыть интерфейс",
    }
    for _, h in ipairs(Hotkeys) do
        if h.Show then
            table.insert(lines, KeyDisplay(h.Key) .. " — " .. h.Label)
        end
    end
    local text = table.concat(lines, "\n")
    pcall(function() self_:Notify("Горячие клавиши", text, 8, "Info") end)
    return text
end

-- ═══════════════════════════════════════════════
-- ⌨️ ВСТРОЕННЫЕ ГОРЯЧИЕ КЛАВИШИ
-- Ctrl+K — палитра команд, Ctrl+. — памятка, Ctrl+Shift+R — сброс настроек
-- ═══════════════════════════════════════════════
local BuiltinHotkeysReady = false
local function InitBuiltinHotkeys()
    if BuiltinHotkeysReady then return end
    BuiltinHotkeysReady = true
    UserInputService.InputBegan:Connect(function(input, gpe)
        if gpe or input.UserInputType ~= Enum.UserInputType.Keyboard then return end
        local k = input.KeyCode
        local ctrl = UserInputService:IsKeyDown(Enum.KeyCode.LeftControl) or UserInputService:IsKeyDown(Enum.KeyCode.RightControl)
        if not ctrl then return end
        local shift = UserInputService:IsKeyDown(Enum.KeyCode.LeftShift) or UserInputService:IsKeyDown(Enum.KeyCode.RightShift)

        if k == Enum.KeyCode.K then
            if BeautifulUI.IsCommandPaletteOpen and BeautifulUI:IsCommandPaletteOpen() then
                BeautifulUI:CloseCommandPalette()
            else
                BeautifulUI:OpenCommandPalette()
            end
        elseif k == Enum.KeyCode.Period and not shift then
            BeautifulUI:ShowHotkeys()
        elseif k == Enum.KeyCode.R and shift then
            BeautifulUI:ResetSettings()
        end
    end)

    -- Esc закрывает палитру
    UserInputService.InputBegan:Connect(function(input, gpe)
        if gpe then return end
        if input.KeyCode == Enum.KeyCode.Escape and BeautifulUI.IsCommandPaletteOpen and BeautifulUI:IsCommandPaletteOpen() then
            BeautifulUI:CloseCommandPalette()
        end
    end)
end

-- Нажатие без «дрожания»: не срабатывает, если пользователь скроллил список
local function OnTap(button, fn)
    local down, downTime
    button.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            down = input.Position
            downTime = os.clock()
        end
    end)
    button.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            if down and (input.Position - down).Magnitude <= 8 and (os.clock() - downTime) < 0.8 then
                Safe(fn)
            end
            down = nil
        end
    end)
end

local function MakeDraggable(frame, dragArea, onDragEnd, onDragStart)
    dragArea = dragArea or frame
    local dragging, dragStart, startPos, moved
    local connections = {}
    table.insert(connections, dragArea.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true; moved = false; dragStart = input.Position; startPos = frame.Position
            if onDragStart then Safe(onDragStart) end
            local c
            c = input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then
                    dragging = false
                    if onDragEnd then Safe(onDragEnd) end
                    if c then c:Disconnect() end
                end
            end)
        end
    end))
    table.insert(connections, UserInputService.InputChanged:Connect(function(input)
        if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
            local delta = input.Position - dragStart
            if math.abs(delta.X) + math.abs(delta.Y) > 2 then moved = true end
            frame.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
        end
    end))
    return function()
        for _, c in ipairs(connections) do pcall(function() c:Disconnect() end) end
    end
end

-- 🎬 ГЛОБАЛЬНАЯ ANIMATION SYSTEM
-- ═══════════════════════════════════════════════
local Animations = {}
local AnimationsByInstance = {}

local function RegisterAnimation(instance, prop, updateFn, cleanupFn)
    if AnimationsByInstance[instance] then
        local prev = AnimationsByInstance[instance][prop]
        if prev then
            prev.alive = false
            if prev.cleanup then pcall(prev.cleanup) end
            AnimationsByInstance[instance][prop] = nil
        end
    end

    local entry = {
        instance = instance, prop = prop, update = updateFn, cleanup = cleanupFn,
        alive = true, startTime = os.clock(),
    }
    table.insert(Animations, entry)
    AnimationsByInstance[instance] = AnimationsByInstance[instance] or {}
    AnimationsByInstance[instance][prop] = entry
    return entry
end

local function StopAnimation(instance, prop)
    if not AnimationsByInstance[instance] then return end
    if prop then
        local prev = AnimationsByInstance[instance][prop]
        if prev then
            prev.alive = false
            if prev.cleanup then pcall(prev.cleanup) end
            AnimationsByInstance[instance][prop] = nil
        end
    else
        for _, prev in pairs(AnimationsByInstance[instance]) do
            prev.alive = false
            if prev.cleanup then pcall(prev.cleanup) end
        end
        AnimationsByInstance[instance] = nil
    end
end

RunService.Heartbeat:Connect(function(dt)
    if AnimationsPaused then return end
    for i = #Animations, 1, -1 do
        local a = Animations[i]
        if not a.alive then
            table.remove(Animations, i)
        elseif not a.instance or not a.instance.Parent then
            if a.cleanup then pcall(a.cleanup) end
            a.alive = false
            if AnimationsByInstance[a.instance] then
                AnimationsByInstance[a.instance][a.prop] = nil
            end
            table.remove(Animations, i)
        elseif a.instance.Visible == false then
            -- ⚡ Оптимизация: не тратим кадры на невидимые элементы
            a.startTime = a.startTime + dt
        else
            local ok = pcall(a.update, dt, os.clock(), a)
            if not ok then
                if a.cleanup then pcall(a.cleanup) end
                a.alive = false
                if AnimationsByInstance[a.instance] then
                    AnimationsByInstance[a.instance][a.prop] = nil
                end
                table.remove(Animations, i)
            end
        end
    end
end)

BeautifulUI.StopAnimation = StopAnimation

-- 🌈 HEX

function BeautifulUI.Hex(hex)
    if type(hex) ~= "string" then return hex end
    hex = hex:gsub("#", ""):gsub("%s", "")
    if #hex ~= 6 then return Color3.new(1, 1, 1) end
    local r = tonumber(hex:sub(1, 2), 16) or 0
    local g = tonumber(hex:sub(3, 4), 16) or 0
    local b = tonumber(hex:sub(5, 6), 16) or 0
    return Color3.fromRGB(r, g, b)
end

function BeautifulUI.ToHex(color)
    return string.format("#%02X%02X%02X",
        math.floor(color.R * 255 + 0.5),
        math.floor(color.G * 255 + 0.5),
        math.floor(color.B * 255 + 0.5))
end

function C(value, fallback)
    if type(value) == "string" then return BeautifulUI.Hex(value) end
    if typeof(value) == "Color3" then return value end
    if type(value) == "table" and value.R ~= nil then
        return Color3.new(value.R, value.G, value.B)
    end
    return fallback or Color3.new(1, 1, 1)
end

-- 🅰️ CUSTOM FONT — свой шрифт для всего интерфейса
-- ═══════════════════════════════════════════════════════════════
-- Можно указать: имя Enum.Font (GothamBlack), ИД шрифта из
-- create.roblox.com/marketplace/fonts (например 12187365375) или строку
-- "rbxasset://fonts/families/GothamSSm.json".
FontConfig = {
    Family = nil,      -- Font / FontFace / имя Enum.Font / asset id
    Weight = nil,      -- Enum.FontWeight.Bold и т.п.
    Style = nil,       -- Enum.FontStyle.Italic
    Bold = nil,        -- true → GothamBold-аналог, false → обычный
    SizeScale = 1,     -- 1.1 = все тексты крупнее на 10%
    LineHeight = nil,  -- межстрочный интервал (для Paragraph/Input)
    Monospace = false, -- TrueType-хаки: не трогать "моно" надписи
    FamilyName = nil,  -- последнее применённое имя (для FontPicker)
    Active = false,    -- свой шрифт включён (новые элементы сразу через FontFace)
    Applied = nil,
}

local FONT_ALIASES = {
    Bold = "GothamBold", Black = "GothamBlack", Medium = "GothamMedium",
    SemiBold = "GothamSemibold", Semi = "GothamSemibold", Light = "GothamLight",
    Regular = "Gotham", Normal = "Gotham", Default = "Gotham",
    Code = "Code", Mono = "Code", MonoSpace = "Code",
}

local function isFontLike(v)
    if v == nil or type(v) == "boolean" then return false end
    local t = typeof(v)
    if t == "EnumItem" then return true end
    if t == "Font" or t == "FontFace" then return true end
    if type(v) == "string" then return true end
    if type(v) == "number" then return true end
    return false
end

-- Строка → {EnumFont = ..., Face = ...}; принимает имя, алиас, ассет-ID, json-путь
local function parseFontSpec(v)
    if type(v) == "number" then v = tostring(math.floor(v)) end
    if typeof(v) == "EnumItem" then
        local n = v.Name
        local e = Enum.Font[n]
        if e then return { EnumFont = e, Name = n } end
        return { EnumFont = v, Name = n }
    end
    if typeof(v) == "Font" or typeof(v) == "FontFace" then
        return { Face = Font.new(v, FontWeight and FontWeight.Regular or nil), Name = "custom" }
    end
    if type(v) ~= "string" then return nil end
    local s = v:gsub("^%s+", ""):gsub("%s+$", "")
    if s == "" then return nil end
    local alias = FONT_ALIASES[s] or FONT_ALIASES[s:sub(1, 1):upper() .. s:sub(2)]
    if alias then s = alias end
    local e = Enum.Font[s]
    if e and type(e) == "table" and e.Name == s then return { EnumFont = e, Name = s } end
    e = nil
    if s:match("^%d+$") then
        return { AssetId = s, Name = "font:" .. s }
    end
    if s:find("rbxasset", 1, true) then
        return { JsonPath = s, Name = "json:" .. s }
    end
    return nil
end

-- Создаёт Roblox FontFace из спецификации (уважает Weight/Style токена)
local function buildFontFace(spec, boldish)
    if not spec then return nil end
    if spec.Face then return spec.Face end
    local ok, face = pcall(function()
        local base
        local raw = spec.Face or spec.EnumFont or spec.JsonPath or (spec.AssetId and spec.AssetId)
        if spec.JsonPath then
            base = Font.new(spec.JsonPath)
        elseif spec.AssetId then
            base = Font.fromId(tonumber(spec.AssetId))
        elseif spec.EnumFont then
            -- Enum.Font → Font через fromEnum, иначе через имя json-файла
            local okEnum
            okEnum, base = pcall(function() return Font.fromEnum(spec.EnumFont) end)
            if not okEnum or not base then
                base = Font.new("rbxasset://fonts/families/GothamSSm.json")
            end
        else
            base = Font.new("rbxasset://fonts/families/GothamSSm.json")
        end
        local weight = FontConfig.Weight
        if not weight then
            local b = boldish
            if b == nil then b = FontConfig.Bold end
            if b == nil then
                if FontConfig.Applied then
                    -- сохраняем «жирность» исходного шрифта, когда свой не задан
                    b = FontConfig.Record and FontConfig.Record.bold
                end
            end
            weight = (b == true) and Enum.FontWeight.Bold or Enum.FontWeight.Regular
        end
        local f = Font.new(base, weight, FontConfig.Style or Enum.FontStyle.Normal)
        return f
    end)
    if ok then return face end
    return nil
end

local function fontSupportsFace()
    if HasFontFace ~= nil then return HasFontFace end
    HasFontFace = pcall(function()
        local l = Instance.new("TextLabel")
        l.FontFace = Font.new("rbxasset://fonts/families/GothamSSm.json")
        -- клиент молча игнорирует свойство — значит FontFace не поддержан
        assert(l.FontFace ~= nil, "FontFace not supported")
    end)
    return HasFontFace
end

local function scaled(size)
    local s = tonumber(FontConfig.SizeScale) or 1
    if s == 1 then return size end
    return math.max(7, math.floor((tonumber(size) or 14) * s + 0.5))
end

-- Имя Enum.Font «похожее» на исходный текст (для fallback-режима)
local function enumForBold(bold)
    local spec = FontConfig.Spec
    if spec and spec.EnumFont then
        if not bold then return spec.EnumFont end
        local n = tostring(spec.EnumFont.Name or "")
        local e = Enum.Font[n .. "Bold"] or Enum.Font[n]
        if e then return e end
        return spec.EnumFont
    end
    return bold and Enum.Font.GothamBold or Enum.Font.Gotham
end

-- Запись снимка (если объект зарегистрирован до появления снимка)
local function noteRecord(obj)
    local rec = FontRegistry[obj]
    if rec == true or rec == nil then
        local f = obj.Font
        local n = f and tostring(f.Name or "") or ""
        rec = {
            original = f,
            baseSize = obj.TextSize,
            bold = n:find("Bold") ~= nil or n:find("Black") ~= nil or n:find("Semibold") ~= nil or n:find("SemiBold") ~= nil,
            mono = n == "Code",
        }
        FontRegistry[obj] = rec
    end
    return rec
end

-- Применить шрифт к одному объекту
applyFontTo = function(obj, boldish)
    if not obj or not obj.Parent then return false end
    local rec = noteRecord(obj)
    if rec.mono and FontConfig.Monospace then return false end
    local size = obj.TextSize
    if rec.baseSize == nil then rec.baseSize = size end

    -- размер с учётом масштаба
    obj.TextSize = scaled(rec.baseSize)
    if FontConfig.LineHeight and (obj.LineHeight ~= nil) then
        pcall(function() obj.LineHeight = FontConfig.LineHeight end)
    end

    if fontSupportsFace() then
        local face = buildFontFace(FontConfig.Spec, boldish ~= nil and boldish or rec.bold)
        if face then
            pcall(function() obj.FontFace = face end)
            return true
        end
    end
    -- Fallback (TrueType / старые клиенты / мобилки без FontFace)
    local ok, err = pcall(function() obj.Font = enumForBold(boldish) end)
    if not ok and obj.Font then obj.Font = Enum.Font.Gotham end
    return ok
end

-- Все зарегистрированные тексты, которые сейчас существуют в игре.
-- Не полагаемся на IsDescendantOf/GetDescendants — работаем по реестру,
-- поэтому закрытые (невидимые) окна тоже получают шрифт.
local function eachTrackedText(callback)
    local n = 0
    for obj in pairs(FontRegistry) do
        if obj.Parent then
            if callback(obj) then n = n + 1 end
        end
    end
    return n
end

-- Применить шрифт ко всем известным текстам (включая закрытые окна)
local function restoreWindowFonts(win, boldish)
    if not win then return 0 end
    return eachTrackedText(function(obj) return applyFontTo(obj, boldish) end)
end

function BeautifulUI.AnimatedColor(instance, prop, color1, color2, duration, style)
    color1, color2 = C(color1), C(color2)
    duration = duration or 2
    style = style or Enum.EasingStyle.Sine

    local entry = RegisterAnimation(instance, prop, function(dt, now, a)
        local t = ((now - a.startTime) % duration) / duration
        local wave = (math.sin(t * math.pi * 2 - math.pi / 2) + 1) / 2
        if style == Enum.EasingStyle.Linear then
            wave = t
        elseif style == Enum.EasingStyle.Quad then
            wave = wave * wave
        elseif style == Enum.EasingStyle.Quart then
            wave = wave * wave * wave * wave
        end
        instance[prop] = color1:Lerp(color2, wave)
    end)

    return {Disconnect = function() StopAnimation(instance, prop) end}
end

function BeautifulUI.AnimatedGradient(instance, colors, speed)
    colors = colors or {Theme.Accent, Theme.AccentDark, Theme.Accent}
    speed = speed or 60

    local resolved = {}
    for _, c in ipairs(colors) do table.insert(resolved, C(c)) end

    local grad = instance:FindFirstChildOfClass("UIGradient")
    if not grad then
        grad = Instance.new("UIGradient")
        grad.Parent = instance
    end

    local keypoints = {}
    for i = 1, #resolved do
        local t = (i - 1) / math.max(#resolved - 1, 1)
        table.insert(keypoints, ColorSequenceKeypoint.new(t, resolved[i]))
    end
    grad.Color = ColorSequence.new(keypoints)

    local entry = RegisterAnimation(instance, "_gradient", function(dt, now)
        grad.Rotation = (now * speed) % 360
        local hueShift = (now * 0.05) % 1
        local kp = {}
        for i = 1, #resolved do
            local t = (i - 1) / math.max(#resolved - 1, 1)
            local h, s, v = Color3.toHSV(resolved[i])
            table.insert(kp, ColorSequenceKeypoint.new(t, Color3.fromHSV((h + hueShift) % 1, s, v)))
        end
        grad.Color = ColorSequence.new(kp)
    end)

    return {Disconnect = function() StopAnimation(instance, "_gradient") end}
end

function BeautifulUI.AnimatedImage(imageLabel, frames, fps)
    frames = frames or {}
    fps = fps or 10
    if #frames == 0 then return {Disconnect = function() end} end

    local lastIdx = -1
    local entry = RegisterAnimation(imageLabel, "_image", function(dt, now)
        local idx = math.floor(now * fps) % #frames + 1
        if idx ~= lastIdx then
            imageLabel.Image = frames[idx]
            lastIdx = idx
        end
    end)
    return {Disconnect = function() StopAnimation(imageLabel, "_image") end}
end

function BeautifulUI.Rainbow(instance, prop, speed, saturation, value)
    -- 🎯 ФИКС 1: защита от nil
    if prop == nil then
        prop = "BackgroundColor3"
    end
    -- 🎯 ФИКС 2: защита от Instance (если передали не prop а объект)
    if type(prop) ~= "string" then
        prop = "BackgroundColor3"
    end
    
    speed = speed or 0.1
    saturation = saturation or 1
    value = value or 1

    -- 🎯 ФИКС 3: используем безопасную проверку
    local propName = tostring(prop)
    
    local entry = RegisterAnimation(instance, "_rainbow_" .. propName, function(dt, now)
        local hue = (now * speed) % 1
        instance[prop] = Color3.fromHSV(hue, saturation, value)
    end)

    return {
        Disconnect = function()
            StopAnimation(instance, "_rainbow_" .. propName)
        end
    }
end

-- 🎨 CycleColors
function BeautifulUI.CycleColors(instance, prop, colors, duration)
    prop = prop or "BackgroundColor3"
    duration = duration or 3
    local hsv = {}
    for _, c in ipairs(colors) do
        c = C(c)
        local h, s, v = Color3.toHSV(c)
        table.insert(hsv, {h = h, s = s, v = v})
    end
    local n = #hsv
    if n < 2 then return {Disconnect = function() end} end
    local entry = RegisterAnimation(instance, "_cycle_" .. prop, function(dt, now)
        local t = (now / duration) % 1
        local pos = t * n
        local i1 = math.floor(pos) + 1
        local i2 = (i1 % n) + 1
        local frac = pos - math.floor(pos)
        local h1, s1, v1 = hsv[i1].h, hsv[i1].s, hsv[i1].v
        local h2, s2, v2 = hsv[i2].h, hsv[i2].s, hsv[i2].v
        local dh = h2 - h1
        if dh > 0.5 then dh = dh - 1 end
        if dh < -0.5 then dh = dh + 1 end
        local h = (h1 + dh * frac) % 1
        local s = s1 + (s2 - s1) * frac
        local v = v1 + (v2 - v1) * frac
        instance[prop] = Color3.fromHSV(h, s, v)
    end)
    return {Disconnect = function() StopAnimation(instance, "_cycle_" .. prop) end}
end

-- ✨ Sparkle
function BeautifulUI.Sparkle(instance, prop, baseColor, sparkleColor, chance)
    prop = prop or "BackgroundColor3"
    baseColor, sparkleColor = C(baseColor), C(sparkleColor)
    chance = chance or 0.05
    local current = baseColor
    local entry = RegisterAnimation(instance, "_sparkle_" .. prop, function(dt, now)

    end)

end

-- 🎬 CustomAnimatedColor
function BeautifulUI.CustomAnimatedColor(instance, prop, keyframes, duration, loopType)
    prop = prop or "BackgroundColor3"
    duration = duration or 3
    loopType = loopType or "loop"
    local sorted = {}
    for _, kf in ipairs(keyframes) do
        table.insert(sorted, {time = kf.time, color = C(kf.color)})
    end
    table.sort(sorted, function(a, b) return a.time < b.time end)
    local entry = RegisterAnimation(instance, "_custom_" .. prop, function(dt, now)
        local cycleTime = now / duration
        local t
        if loopType == "pingpong" then
            local cycle = cycleTime % 2
            t = cycle <= 1 and cycle or (2 - cycle)
        elseif loopType == "once" then
            t = math.min(cycleTime, 1)
        else
            t = cycleTime % 1
        end
        local c1, c2, segT
        for i = 1, #sorted - 1 do
            if t >= sorted[i].time and t <= sorted[i + 1].time then
                c1 = sorted[i].color
                c2 = sorted[i + 1].color
                segT = (t - sorted[i].time) / math.max(sorted[i + 1].time - sorted[i].time, 0.001)
                break
            end
        end
        if not c1 then
            c1 = sorted[#sorted].color
            c2 = c1
            segT = 0
        end
        instance[prop] = c1:Lerp(c2, segT)
    end)
    return {Disconnect = function() StopAnimation(instance, "_custom_" .. prop) end}
end

    local char = game.Players.LocalPlayer.Character

    local char = game.Players.LocalPlayer.Character
    if char then
        local hum = char:FindFirstChildOfClass("Humanoid")

            return string.format("%.1f", (hum.Health / hum.MaxHealth) * 100)
        end
    end
    return "0.0"
end
Placeholders["%time"] = function() return os.date("%H:%M:%S") end
Placeholders["%date"] = function() return os.date("%d.%m.%Y") end
Placeholders["%players"] = function() return tostring(#game.Players:GetPlayers()) end
Placeholders["%name"] = function() return game.Players.LocalPlayer.Name end
Placeholders["%displayname"] = function() return game.Players.LocalPlayer.DisplayName end
Placeholders["%userid"] = function() return tostring(game.Players.LocalPlayer.UserId) end
Placeholders["%memory"] = function()
    local ok, mem = pcall(function()
        return math.floor(game:GetService("Stats"):GetTotalMemoryUsageMb())
    end)
    return ok and tostring(mem) or "0"
end
Placeholders["%placeid"] = function() return tostring(game.PlaceId) end
Placeholders["%jobid"] = function() return game.JobId:sub(1, 8) .. "..." end
Placeholders["%uptime"] = function() return tostring(math.floor(tick() - (_G.LibraryLoadTime or tick()))) end

function BeautifulUI.SubstitutePlaceholders(text)
    if type(text) ~= "string" then return text end
    if not text:find("%%") then return text end
    for key, fn in pairs(Placeholders) do
        if text:find(key, 1, true) then
            local ok, val = pcall(fn)
            if ok then
                text = text:gsub(key:gsub("%%", "%%%%"), tostring(val))
            end
        end
    end
    return text
end

function BeautifulUI:Placeholder(instance, initialText, updateInterval)
    if not instance then return {Disconnect = function() end} end
    updateInterval = updateInterval or 0.5
    local template = initialText or instance.Text or ""
    local running = true
    task.spawn(function()
        while running and instance and instance.Parent do
            local newText = BeautifulUI.SubstitutePlaceholders(template)
            if instance.Text ~= newText then
                instance.Text = newText
            end
            task.wait(updateInterval)
        end
    end)
    return {
        SetTemplate = function(t) template = t end,
        Disconnect = function() running = false end
    }
end

function BeautifulUI:RegisterPlaceholder(name, fn)
    if type(name) == "string" and not name:find("^%%") then
        name = "%" .. name
    end
    Placeholders[name] = fn
end

function BeautifulUI:GetPlaceholders()
    local list = {}
    for k in pairs(Placeholders) do table.insert(list, k) end
    table.sort(list)

-- Универсальный разбор аргументов: методы работают и через `:` и через `.`
-- (в v5 вызов через точку «съедал» первый аргумент, т.к. self занимал его место)
local function methodArgs(...)
    local args = {...}
    if args[1] == BeautifulUI then table.remove(args, 1) end
    return args
end

function BeautifulUI.ApplyGradientPreset(...)
    local args = methodArgs(...)
    local instance, presetName, rotation = args[1], args[2], args[3]
    local preset = BeautifulUI.GradientPresets[presetName]
    if not preset then warn("[BeautifulUI] Пресет не найден:", presetName); return nil end
    local old = instance:FindFirstChildOfClass("UIGradient")
    if old then old:Destroy() end
    local keypoints = {}
    for i, hex in ipairs(preset) do
        local t = (i - 1) / math.max(#preset - 1, 1)
        table.insert(keypoints, ColorSequenceKeypoint.new(t, BeautifulUI.Hex(hex)))
    end
    local grad = Instance.new("UIGradient")
    grad.Color = ColorSequence.new(keypoints)
    grad.Rotation = rotation or 0
    grad.Parent = instance
    return grad
end

function BeautifulUI:GetGradientPresets()
    local list = {}
    for k in pairs(BeautifulUI.GradientPresets) do table.insert(list, k) end
    table.sort(list)
    return list
end

    BeautifulUI.GradientPresets[name] = colors
end

-- 💾 Настройки окна (позиция/размер/тема) — сохраняются сами
local function PrefsPath() return ConfigPath:gsub("%.json$", "") .. "_prefs.json" end

function BeautifulUI:SaveWindowPrefs(force)
    if not PrefsEnabled and not force then return end
    local data = {Theme = BeautifulUI.CurrentThemeName, Windows = {}}
    if FontConfig.Applied and FontConfig.FamilyName then
        data.Font = { Family = FontConfig.FamilyName, SizeScale = FontConfig.SizeScale,
                      Bold = FontConfig.Bold, Monospace = FontConfig.Monospace }
    end
    for _, w in ipairs(OpenWindows) do
        if w.SaveLayout then
            local ok, layout = pcall(w.SaveLayout, w)
            if ok and type(layout) == "table" then table.insert(data.Windows, layout) end
        end
    end
    local okJ, encoded = pcall(HttpService.JSONEncode, HttpService, data)
    if okJ then pcall(function() if writefile then writefile(PrefsPath(), encoded) end end) end
end

function BeautifulUI:LoadWindowPrefs()
    local content
    pcall(function()
        if readfile and isfile and isfile(PrefsPath()) then content = readfile(PrefsPath()) end
    end)
    if not content then return {} end
    local ok, data = pcall(HttpService.JSONDecode, HttpService, content)
    if not ok or type(data) ~= "table" then return {} end
    if data.Theme and Themes[data.Theme] then BeautifulUI:SetTheme(data.Theme) end
    if type(data.Font) == "table" and data.Font.Family then
        pcall(function()
            BeautifulUI:SetCustomFont(data.Font.Family, {
                SizeScale = data.Font.SizeScale, Bold = data.Font.Bold, Monospace = data.Font.Monospace,
            })
        end)
    end
    return data.Windows or {}
end

-- ♻️ Сброс пользовательских настроек (позиция/размер/тема → по умолчанию)
local WindowDefaults = {}

function BeautifulUI:ResetSettings()
    pcall(function() if delfile and isfile and isfile(PrefsPath()) then delfile(PrefsPath()) end end)
    self:SetTheme("Purple")
    pcall(function() self:ResetFont() end)
    for _, w in ipairs(OpenWindows) do
        local d = WindowDefaults[w]
        if d then
            pcall(function()
                w:SetSize(d.Width, d.Height)
                w:Center()
            end)
        end
    end
    self:Notify("Настройки", "Сброшены к значениям по умолчанию ♻️", 3, "Success")
end

-- Применяем сохранённую тему ещё до открытия окна
task.defer(function()
    pcall(function()
        local content
        if readfile and isfile and isfile(PrefsPath()) then content = readfile(PrefsPath()) end
        if content then
            local ok, data = pcall(HttpService.JSONDecode, HttpService, content)
            if ok and type(data) == "table" and data.Theme and Themes[data.Theme] then
                ApplyTheme(Themes[data.Theme])
                BeautifulUI.CurrentThemeName = data.Theme
            end
        end
    end)
end)

function BeautifulUI:SaveConfig()
    local data = {}
    for k, v in pairs(ConfigFlags) do
        if v.Get then data[k] = v:Get() end
    end
    local ok, encoded = pcall(HttpService.JSONEncode, HttpService, data)
    if ok then
        pcall(function() if writefile then writefile(ConfigPath, encoded) end end)
        BeautifulUI:Notify("Конфиг", "Сохранено ✅", 2, "Success")
    end
end

function BeautifulUI:LoadConfig()
    local ok, content = pcall(function()
        if readfile and isfile and isfile(ConfigPath) then return readfile(ConfigPath) end
    end)
    if not ok or not content then
        BeautifulUI:Notify("Конфиг", "Файл не найден", 2, "Warning"); return
    end
    local ok2, data = pcall(HttpService.JSONDecode, HttpService, content)
    if not ok2 then return end
    for k, v in pairs(data) do
        if ConfigFlags[k] and ConfigFlags[k].Set then pcall(function() ConfigFlags[k]:Set(v) end) end
    end
    BeautifulUI:Notify("Конфиг", "Загружено ✅", 2, "Success")
end

function BeautifulUI:DeleteConfig()
    pcall(function() if delfile and isfile and isfile(ConfigPath) then delfile(ConfigPath) end end)
end

-- 🔀 Миграция конфига: переименовать/удалить/переименовать-по-правилу флаги
-- BeautifulUI:MigrateConfig({rename = {OldFlag = "NewFlag"}, remove = {"DeadFlag"}})
function BeautifulUI:MigrateConfig(rules)
    if type(rules) ~= "table" then return false end
    for _, flag in ipairs(rules.remove or {}) do
        if ConfigFlags[flag] then ConfigFlags[flag] = nil end
    end
    for oldFlag, newFlag in pairs(rules.rename or {}) do
        local entry = ConfigFlags[oldFlag]
        if entry and not ConfigFlags[newFlag] then
            ConfigFlags[newFlag] = entry
            ConfigFlags[oldFlag] = nil
        end
    end
    return true
end

-- 📤 Экспорт конфига в читаемый текст (для «поделиться настройками»)
function BeautifulUI:ExportConfig()
    local lines = {}
    local labels = ConfigLabels
    for k, v in pairs(ConfigFlags) do
        if v.Get then
            local ok, val = pcall(v.Get, v)
            if ok then
                if type(val) == "table" then
                    local parts = {}
                    for i, item in ipairs(val) do parts[i] = tostring(item) end
                    val = table.concat(parts, ", ")
                end
                local meta = labels[k]
                local label = meta and meta.Label or k
                table.insert(lines, string.format("%s = %s", label, tostring(val) .. (meta and meta.Suffix or "")))
            end
        end
    end
    table.sort(lines)
    return table.concat(lines, "\n")
end

-- 📥 Импорт: показывает текущие значения и загружает конфиг по кнопке
function BeautifulUI:ShowConfigSummary()
    local text = self:ExportConfig()
    if text == "" then text = "Пока нечего сохранять" end
    return self:Notify("Текущий конфиг", text, 10, "Info", {
        {Text = "Сохранить", Color = Theme.Success, Callback = function() self:SaveConfig() end},
        {Text = "Закрыть", Color = Theme.Tertiary},
    })
end

local DebugConsole = {Logs = {}, Visible = false, Gui = nil}

    for _, v in ipairs(args) do table.insert(parts, tostring(v)) end

    table.insert(self.Logs, {level = level, msg = msg, time = os.date("%H:%M:%S")})
    print("[BeautifulUI " .. level .. "] " .. msg)
    if self.Visible and self.Gui then self:Refresh() end
end

        list.CanvasSize = UDim2.new(0, 0, 0, layout.AbsoluteContentSize.Y + 10)

    end

    local closeBtn = Create("TextButton", {Size = UDim2.new(0, 26, 0, 26), Position = UDim2.new(1, -32, 0, 5), BackgroundColor3 = Theme.Tertiary, BorderSizePixel = 0, Font = Enum.Font.GothamBold, Text = "✕", TextColor3 = Theme.SubText, TextSize = 12, Parent = topBar})
    Corner(6, closeBtn); closeBtn.MouseButton1Click:Connect(function() self:Toggle() end)
    local clearBtn = Create("TextButton", {Size = UDim2.new(0, 26, 0, 26), Position = UDim2.new(1, -64, 0, 5), BackgroundColor3 = Theme.Tertiary, BorderSizePixel = 0, Font = Enum.Font.GothamBold, Text = "🗑", TextColor3 = Theme.SubText, TextSize = 12, Parent = topBar})
    Corner(6, clearBtn); clearBtn.MouseButton1Click:Connect(function() self.Logs = {}; self:Refresh() end)
    local scroller = Create("ScrollingFrame", {Name = "List", Size = UDim2.new(1, -16, 1, -80), Position = UDim2.new(0, 8, 0, 44), BackgroundColor3 = Theme.Secondary, BorderSizePixel = 0, ScrollBarThickness = 4, ScrollBarImageColor3 = Theme.Accent, CanvasSize = UDim2.new(0, 0, 0, 0), Parent = frame})
    Corner(6, scroller)
    Create("UIListLayout", {Padding = UDim.new(0, 2), SortOrder = Enum.SortOrder.LayoutOrder, Parent = scroller})
    Create("UIPadding", {PaddingTop = UDim.new(0, 6), PaddingLeft = UDim.new(0, 6), Parent = scroller})
    local input = Create("TextBox", {Size = UDim2.new(1, -16, 0, 28), Position = UDim2.new(0, 8, 1, -36), BackgroundColor3 = Theme.Tertiary, BorderSizePixel = 0, Font = Enum.Font.Code, Text = "", PlaceholderText = "> введите Lua команду...", PlaceholderColor3 = Theme.SubText, TextColor3 = Theme.Text, TextSize = 12, TextXAlignment = Enum.TextXAlignment.Left, ClearTextOnFocus = false, Parent = frame})
    Corner(6, input); Create("UIPadding", {PaddingLeft = UDim.new(0, 8), Parent = input})
    input.FocusLost:Connect(function(enter)
        if enter and input.Text ~= "" then
            self:Info("> " .. input.Text)
            if not loadstring then self:Error("loadstring недоступен"); input.Text = ""; return end
            local fn, err = loadstring(input.Text)
            if not fn then self:Error("Compile error: " .. tostring(err)); input.Text = ""; return end
            local ok, runErr = pcall(fn)
            if not ok then self:Error("Runtime error: " .. tostring(runErr)) end
            input.Text = ""
        end
    end)
    MakeDraggable(frame, topBar)
end

    Log = function(...) if IsDebugUser() then DebugConsole:Info(...) end end,
    Info = function(...) if IsDebugUser() then DebugConsole:Info(...) end end,
    Warn = function(...) if IsDebugUser() then DebugConsole:Warn(...) end end,
    Error = function(...) if IsDebugUser() then DebugConsole:Error(...) end end,
    Toggle = function()
        if IsDebugUser() then DebugConsole:Toggle()
        else BeautifulUI:Notify("Доступ запрещён", "Debug только для разработчика", 2, "Danger") end
    end,
    IsAvailable = function() return IsDebugUser() end,
    Clear = function() DebugConsole.Logs = {}; DebugConsole:Refresh() end,
}
BeautifulUI.IsDebugUser = IsDebugUser
if IsDebugUser() then
    UserInputService.InputBegan:Connect(function(input, gpe)
        if gpe then return end
        if input.KeyCode == Enum.KeyCode.F10 then DebugConsole:Toggle() end
    end)
end

function BeautifulUI:DialogBox(config)
    config = config or {}
    local text = config.Text or "Hello!"
    local speaker = config.Speaker or nil
    local avatarId = config.Avatar or nil
    local duration = config.Duration or 0
    local accentColor = config.Accent and C(config.Accent) or Theme.Accent
    local iconChar = config.Icon or "💬"

    local gui = Create("ScreenGui", {Name = "BeautifulUIDialog_" .. tick(), ResetOnSpawn = false, ZIndexBehavior = Enum.ZIndexBehavior.Sibling, Parent = (gethui and gethui()) or CoreGui})
    local dlgWidth = IsMobile and 340 or 640
    local frame = Create("Frame", {Size = UDim2.new(0, dlgWidth, 0, 140), Position = UDim2.new(0.5, -dlgWidth / 2, 1, 200), BackgroundColor3 = Theme.Secondary, BorderSizePixel = 0, Parent = gui})
    Corner(16, frame); Stroke(accentColor, 2, 0.3, frame); Gradient(Theme.Gradient1, Theme.Secondary, 90, frame)

    local avFrame = Create("Frame", {Size = UDim2.new(0, 100, 0, 100), Position = UDim2.new(0, 16, 0.5, -50), BackgroundColor3 = Theme.Tertiary, BorderSizePixel = 0, Parent = frame})
    Corner(14, avFrame); Stroke(accentColor, 2, 0.2, avFrame)
    if avatarId then
        local img = Create("ImageLabel", {Size = UDim2.new(1, -6, 1, -6), Position = UDim2.new(0, 3, 0, 3), BackgroundTransparency = 1, Image = avatarId, ScaleType = Enum.ScaleType.Crop, Parent = avFrame})
        Corner(12, img)
        if config.Frames then BeautifulUI.AnimatedImage(img, config.Frames, config.FPS or 10) end
        if config.Icon then
            local badge = Create("TextLabel", {Size = UDim2.new(0, 26, 0, 26), Position = UDim2.new(1, -14, 1, -14), BackgroundColor3 = accentColor, BorderSizePixel = 0, Font = Enum.Font.GothamBold, Text = config.Icon, TextColor3 = Theme.Text, TextSize = 14, Parent = avFrame})
            Corner(13, badge); Stroke(Theme.Secondary, 2, 0, badge)
        end
    else
        Create("TextLabel", {Size = UDim2.new(1, 0, 1, 0), BackgroundTransparency = 1, Font = Enum.Font.GothamBold, Text = iconChar, TextColor3 = accentColor, TextSize = 42, Parent = avFrame})
    end
    if speaker then
        local nameLabel = Create("TextLabel", {Size = UDim2.new(0, 150, 0, 24), Position = UDim2.new(0, 130, 0, 12), BackgroundColor3 = accentColor, BorderSizePixel = 0, Font = Enum.Font.GothamBold, Text = speaker, TextColor3 = Theme.Text, TextSize = 13, Parent = frame})
        Corner(6, nameLabel)
    end
    local textLabel = Create("TextLabel", {Size = UDim2.new(1, -160, 1, -50), Position = UDim2.new(0, 130, speaker and 0.4 or 0.15, speaker and 4 or 0), BackgroundTransparency = 1, Font = Enum.Font.Gotham, Text = "", TextColor3 = Theme.Text, TextSize = 16, TextWrapped = true, TextXAlignment = Enum.TextXAlignment.Left, TextYAlignment = Enum.TextYAlignment.Top, RichText = true, Parent = frame})
    task.spawn(function()
        for i = 1, #text do
            if not textLabel.Parent then return end
            textLabel.Text = text:sub(1, i); task.wait(0.02)
        end
    end)
    local closeBtn = Create("TextButton", {Size = UDim2.new(0, 30, 0, 30), Position = UDim2.new(1, -40, 0, 12), BackgroundColor3 = Theme.Tertiary, BorderSizePixel = 0, Font = Enum.Font.GothamBold, Text = "✕", TextColor3 = Theme.SubText, TextSize = 12, Parent = frame})
    Corner(8, closeBtn)
    local closed = false
    local function close()
        if closed then return end
        closed = true
        Tween(frame, 0.35, {Position = UDim2.new(0.5, -dlgWidth / 2, 1, 200)})
        task.delay(0.4, function() gui:Destroy() end)
    end
    closeBtn.MouseButton1Click:Connect(close)
    Tween(frame, 0.5, {Position = UDim2.new(0.5, -dlgWidth / 2, 1, -170)})
    if duration > 0 then task.delay(duration + (#text * 0.02), close) end
    return {SetText = function(newText) textLabel.Text = newText end, Close = close}
end

function BeautifulUI:CreateWindow(config)
    config = config or {}
    local title = config.Title or "Beautiful UI"
    local subtitle = config.Subtitle or ("v" .. BeautifulUI.Version)
    local toggleKey = config.ToggleKey or Enum.KeyCode.RightShift
    local windowIcon = config.Icon or "⚡"
    local showConfig = config.ShowConfig == true
    local forceCompact = config.Compact
    local backdrop = config.Backdrop == true
    -- 🅰️ свой шрифт прямо в конфиге окна.
    -- Применяем ПОСЛЕ постройки элементов (task.defer), иначе первый шрифт
    -- «запечётся» только у тех текстов, что уже успели создаться.
    local configFont = config.Font
    local configFontOpts = {
        Weight = config.FontWeight, SizeScale = config.FontSizeScale,
        Bold = config.FontBold, Monospace = config.FontMonospace,
    }

    local baseW = (config.Size and config.Size.X.Offset > 0) and config.Size.X.Offset or (IsMobile and 400 or 780)
    local baseH = (config.Size and config.Size.Y.Offset > 0) and config.Size.Y.Offset or (IsMobile and 320 or 580)

    -- Discord = true → используем уже настроенный инвайт (SetDiscord),
    -- Discord = "код" или {Invite="код", ...} → настраиваем здесь
    if type(config.Discord) == "string" then
        BeautifulUI:SetDiscord({Invite = config.Discord})
    elseif type(config.Discord) == "table" then
        BeautifulUI:SetDiscord(config.Discord)
    end

    if config.Theme and Themes[config.Theme] then
        ApplyTheme(Themes[config.Theme])
        BeautifulUI.CurrentThemeName = config.Theme
    end

    local gui = Create("ScreenGui", {Name = "BeautifulUI_" .. tick(), ResetOnSpawn = false, DisplayOrder = 50, ZIndexBehavior = Enum.ZIndexBehavior.Sibling, Parent = (gethui and gethui()) or CoreGui})

    -- ═══ АДАПТИВНОСТЬ ═══
    local function viewport()
        local cam = workspace.CurrentCamera
        return cam and cam.ViewportSize or Vector2.new(1280, 720)
    end

    local function computeSize()
        local vp = viewport()
        local w = math.clamp(baseW, 280, math.max(vp.X - 24, 280))
        local h = math.clamp(baseH, 220, math.max(vp.Y - 24, 220))
        return Vector2.new(math.floor(w), math.floor(h))
    end

    local iconButton = Create("TextButton", {
        Name = "IconButton", Size = UDim2.new(0, 46, 0, 46), AnchorPoint = Vector2.new(0.5, 0.5),
        Position = UDim2.new(0.5, 0, 0, -60), BackgroundColor3 = Theme.Accent, BorderSizePixel = 0,
        Font = Enum.Font.GothamBold, Text = windowIcon, TextColor3 = Theme.Text, TextSize = 22,
        AutoButtonColor = false, Visible = false, ZIndex = 40, Parent = gui,
    })
    Corner(14, iconButton)
    local iconStroke = Stroke(Theme.Text, 1.5, 0.35, iconButton)
    local iconGrad = Gradient(Theme.Accent, Theme.AccentDark, 90, iconButton)
    local iconShadow = Shadow(UDim2.new(1, 30, 1, 30), iconButton, true)

    local iconToken = 0

    task.spawn(function()
        while iconButton.Parent do
            if iconButton.Visible then
                Tween(iconStroke, 0.7, {Transparency = 0.75}); task.wait(0.7)
                Tween(iconStroke, 0.7, {Transparency = 0.3}); task.wait(0.7)
            else
                task.wait(0.5)
            end

    end)

    local function ShowIcon()
        iconToken = iconToken + 1
        iconButton.Position = UDim2.new(0.5, 0, 0, -60)
        iconButton.Visible = true
        Tween(iconButton, 0.35, {Position = UDim2.new(0.5, 0, 0, 18)})
    end
    local function HideIcon()
        if not iconButton.Visible then return end
        iconToken = iconToken + 1
        local token = iconToken
        Tween(iconButton, 0.25, {Position = UDim2.new(0.5, 0, 0, -60)})
        task.delay(0.26, function()
            if token == iconToken then iconButton.Visible = false end
        end)
    end

    -- ═══ ПОИСК ═══
    local function applySearch()
        local q = (searchBox.Text or ""):lower()
        q = q:gsub("^%s+", ""):gsub("%s+$", "")
        searchClear.Visible = q ~= ""

        local matched, any, firstMatch = {}, false, nil
        local tabs = Window.Tabs
        for i = 1, #tabs do
            matched[tabs[i]] = (q == "" or tabs[i].Name:lower():find(q, 1, true) ~= nil)
        end
        if q ~= "" then
            -- один проход по индексу элементов вместо перебора для каждой вкладки
            for i = 1, #SearchRegistry do
                local e = SearchRegistry[i]
                if e.Tab and not matched[e.Tab] and e.Text:find(q, 1, true) then
                    matched[e.Tab] = true
                end
            end
        end
        for i = 1, #tabs do
            local t = tabs[i]
            local match = matched[t] or false
            t.Button.Visible = match
            if match then any = true; firstMatch = firstMatch or t end
        end
        emptyState.Visible = (not any) and q ~= ""
        if any and Window.ActiveTab and not Window.ActiveTab.Button.Visible then
            if Window.ActiveTab._activate and firstMatch then firstMatch._activate(true) end
        end
    end

    local searchDebounce
    searchBox:GetPropertyChangedSignal("Text"):Connect(function()
        if searchDebounce then task.cancel(searchDebounce) end
        searchDebounce = task.delay(0.05, applySearch)
    end)
    searchBox.FocusLost:Connect(function(enter)
        if enter then applySearch() end
    end)

        searchBox.Text = ""

    local function applyResponsive(animate)
        currentSize = computeSize()

        if forceCompact ~= nil then compact = forceCompact else compact = currentSize.X < 580 end
        isCompact = compact
        Window.Compact = compact

    local function clampToViewport()
        local vp = viewport()
        local halfW, halfH = currentSize.X / 2, currentSize.Y / 2
        if vp.X <= 0 or vp.Y <= 0 then return end
        local cx = vp.X / 2 + root.Position.X.Offset
        local cy = vp.Y / 2 + root.Position.Y.Offset
        local minX, maxX = halfW + 6, math.max(vp.X - halfW - 6, halfW + 6)
        local minY, maxY = halfH + 6, math.max(vp.Y - halfH - 6, halfH + 6)
        cx = math.clamp(cx, minX, maxX)
        cy = math.clamp(cy, minY, maxY)
        root.Position = UDim2.new(0.5, math.floor(cx - vp.X / 2), 0.5, math.floor(cy - vp.Y / 2))
    end

            end)
        end
    end)

    local visible = true
    local visibleToken = 0
    local syncBackdrop

    local function setVisible(v, animate)
        v = v and true or false
        if visible == v then return end
        visible = v
        visibleToken = visibleToken + 1
        Window.Visible = v
        if v then
            HideIcon()
            clampToViewport()
            root.Visible = true
            root.Size = UDim2.new(0, 0, 0, 0)
            if animate then
                Tween(root, 0.26, {Size = UDim2.new(0, currentSize.X, 0, currentSize.Y)})
            else
                root.Size = UDim2.new(0, currentSize.X, 0, currentSize.Y)
            end
        else
            local target = UDim2.new(0, 0, 0, 0)
            local token = visibleToken
            if animate then
                Tween(root, 0.2, {Size = target})
                task.delay(0.21, function()
                    if token == visibleToken then root.Visible = false; ShowIcon() end
                end)
            else
                root.Size = target
                root.Visible = false
                ShowIcon()
            end
        end
        syncBackdrop()
    end

        local target
        if type(nameOrIndex) == "number" then
            target = self.Tabs[nameOrIndex]
        else
            for _, t in ipairs(self.Tabs) do
                if t.Name == nameOrIndex then target = t; break end
            end
        end
        if target and target._activate then
            if searchBox.Text ~= "" then searchBox.Text = ""; applySearch() end
            target._activate()
            return true
        end
        return false
    end

        -- ─── BUTTON ───
        local BUTTON_STYLES = {
            Primary = {bg = "Accent", hover = "AccentDark"},
            Success = {bg = "Success", hover = "Success"},
            Danger = {bg = "Danger", hover = "Danger"},
            Warning = {bg = "Warning", hover = "Warning"},
            Ghost = {bg = "Tertiary", hover = "Tertiary"},
            Gradient = {bg = "Accent", hover = "AccentDark"},
            Outline = {bg = "Background", hover = "Tertiary"},
        }

        -- Поддерживаем оба стиля вызова:
        --   Tab:Button("Текст", cb, "Gradient", "🚀")
        --   Tab:Button{Text = "Текст", Callback = cb, Shape = "Pill", Size = 40}
        function TabAPI:_mkButton(parent, text, callback, style, icon, data)
            if type(text) == "table" then
                data = text
                text, callback, style, icon = data.Text or data.text or "Кнопка", data.Callback or data.callback, data.Style or data.style, data.Icon or data.icon
            end
            data = data or {}
            local btnData = data
            style = style or "Primary"
            local cfg = BUTTON_STYLES[style] or BUTTON_STYLES.Primary
            local darkText = (style == "Warning")
            local bg = Theme[cfg.bg]

            local btn = Create("TextButton", {
                Size = UDim2.new(1, 0, 0, 36), BackgroundColor3 = bg,
                BackgroundTransparency = (style == "Gradient") and 0 or 0.12,
                BorderSizePixel = 0, Font = Enum.Font.GothamMedium, AutoButtonColor = false,
                Text = (icon and icon ~= "" and (icon .. "  ") or "") .. text,
                TextColor3 = darkText and Theme.Background or Theme.Text, TextSize = 14, Parent = parent,
            })
            -- 🔘 ФОРМА КНОПКИ: Corner(9) по умолчанию, но можно Pill / Square / Sharp / Circle
            local shape = tostring(btnData.Shape or btnData.shape or "Round")
            local cornerRadius = 9
            if shape == "Pill" then cornerRadius = 18
            elseif shape == "Square" then cornerRadius = 6
            elseif shape == "Sharp" then cornerRadius = 0
            elseif shape == "Circle" then cornerRadius = 18 end
            local btnCorner = Corner(cornerRadius, btn)
            if shape == "Circle" or shape == "Icon" then
                btn.Size = UDim2.new(0, 40, 0, 40)
                btn.Text = icon or "●"
                btn.TextSize = 18
            elseif shape == "Wide" then
                btn.Size = UDim2.new(1, 0, 0, 44)
            end
            if btnData.Size or btnData.Height then
                local hh = tonumber(btnData.Height) or tonumber(btnData.Size) or 36
                btn.Size = UDim2.new(shape == "Circle" and 0 or 1, shape == "Circle" and hh or 0, 0, hh)
            end
            local btnStroke = Stroke(bg, 1, (style == "Outline") and 0.4 or 0.72, btn)
            local grad
            if style == "Gradient" then grad = Gradient(Theme.Accent, Theme.AccentDark, 0, btn) end
            local uiScale = Create("UIScale", {Scale = 1, Parent = btn})
            local enabled = true

            local function idle()
                Tween(btn, 0.14, {BackgroundTransparency = (style == "Gradient") and 0 or 0.12})
                Tween(btnStroke, 0.14, {Transparency = (style == "Outline") and 0.4 or 0.72, Thickness = 1})
            end
            local function hot()
                Tween(btn, 0.14, {BackgroundTransparency = 0})
                Tween(btnStroke, 0.14, {Transparency = 0.35, Thickness = 1.5})
            end

            btn.MouseEnter:Connect(function() if enabled then hot() end end)
            btn.MouseLeave:Connect(function() if enabled then idle() end end)
            OnTap(btn, function()
                if not enabled then return end
                Tween(uiScale, 0.07, {Scale = 0.97})
                task.wait(0.07)
                Tween(uiScale, 0.12, {Scale = 1})
                Safe(callback)
            end)

            registerThemeElement(btn, function()
                bg = Theme[cfg.bg]
                btn.BackgroundColor3 = bg
                btn.TextColor3 = darkText and Theme.Background or Theme.Text
                btnStroke.Color = bg
                if grad then grad.Color = ColorSequence.new(Theme.Accent, Theme.AccentDark) end
            end)

            RegisterSearchable(text, page, tab)
            BeautifulUI:RegisterCommand({Name = text, Keywords = {"кнопка", "button"}, Window = Window, Tab = tab.Name, Element = btn, Callback = callback})

            -- 🔧 Хелперы на самой кнопке
            btn.SetText = function(t) btn.Text = (icon and icon ~= "" and (icon .. "  ") or "") .. tostring(t) end
            btn.SetEnabled = function(state)
                enabled = state and true or false
                btn.TextTransparency = enabled and 0 or 0.5
                btn.BackgroundTransparency = enabled and ((style == "Gradient") and 0 or 0.12) or 0.6
                if not enabled then btnStroke.Transparency = 0.85 end
            end

            -- 🔘 смена формы на лету: btn:SetShape("Pill" | "Square" | "Sharp" | "Round")
            -- работает и как btn:SetShape("Pill"), и как btn.SetShape("Pill")
            btn.SetShape = function(a, b2)
                local name = type(a) == "table" and b2 or a
                local r = ({ Round = 9, Pill = 18, Square = 6, Sharp = 0 })[tostring(name)] or 9
                shape = tostring(name)
                btnCorner.CornerRadius = UDim.new(0, r)
                return btn
            end
            -- 📐 своя высота: btn:SetHeight(48)
            btn.SetHeight = function(a, b2)
                local h = type(a) == "table" and b2 or a
                local hh = tonumber(h) or 36
                btn.Size = UDim2.new(shape == "Circle" and 0 or 1, shape == "Circle" and hh or 0, 0, hh)
                return btn
            end
            return btn
        end

        -- 🔘 Кнопка-иконка: круглая, только эмодзи/символ
        function TabAPI:IconButton(icon, callback, opts)

        -- ─── SLIDER ───
        function TabAPI:_mkSlider(parent, text, min, max, default, callback, flag, opts)
            opts = opts or {}
            min, max = min or 0, max or 100
            if max == min then max = min + 1 end
            local step = tonumber(opts.step or opts.increment) or 1
            if step <= 0 then step = 1 end
            local suffix = opts.suffix or ""
            local decimals = 0
            if step < 1 then
                -- дробный шаг: показываем нужное число знаков после запятой
                local s = string.format("%.6f", step):gsub("0+$", "")
                local digits = s:match("%.(%d+)")
                decimals = digits and #digits or 0
            end
            local function roundToStep(v)
                v = math.clamp(v, min, max)
                local snapped = min + math.floor((v - min) / step + 0.5) * step
                snapped = math.clamp(snapped, min, max)
                if decimals > 0 then
                    local mult = 10 ^ decimals
                    return math.floor(snapped * mult + 0.5) / mult
                end
                return math.floor(snapped + 0.5)
            end
            local function fmt(v)
                if decimals > 0 then
                    return string.format("%." .. decimals .. "f", v) .. suffix
                end
                return tostring(math.floor(v + 0.5)) .. suffix
            end
            local value = roundToStep(default or min)
            local updating = false

            local frame = Create("Frame", {Size = UDim2.new(1, 0, 0, 50), BackgroundColor3 = Theme.Secondary, BorderSizePixel = 0, Parent = parent})
            Corner(8, frame)
            local lbl = Create("TextLabel", {Size = UDim2.new(0.7, 0, 0, 20), Position = UDim2.new(0, 12, 0, 6), BackgroundTransparency = 1, Font = Enum.Font.GothamMedium, Text = text, TextColor3 = Theme.Text, TextSize = 13, TextXAlignment = Enum.TextXAlignment.Left, Parent = frame})
            local valueLabel = Create("TextLabel", {Size = UDim2.new(0.3, -12, 0, 20), Position = UDim2.new(0.7, 0, 0, 6), BackgroundTransparency = 1, Font = Enum.Font.GothamBold, Text = fmt(value), TextColor3 = Theme.Accent, TextSize = 13, TextXAlignment = Enum.TextXAlignment.Right, Parent = frame})
            local barBg = Create("Frame", {Size = UDim2.new(1, -24, 0, 6), Position = UDim2.new(0, 12, 0, 34), BackgroundColor3 = Theme.Tertiary, BorderSizePixel = 0, Parent = frame})
            Corner(3, barBg)
            local fill = Create("Frame", {Size = UDim2.new((value - min) / (max - min), 0, 1, 0), BackgroundColor3 = Theme.Accent, BorderSizePixel = 0, Parent = barBg})
            Corner(3, fill)
            Gradient(Theme.Accent, Theme.AccentDark, 0, fill)
            local knob = Create("Frame", {Size = UDim2.new(0, 12, 0, 12), AnchorPoint = Vector2.new(0.5, 0.5), Position = UDim2.new((value - min) / (max - min), 0, 0.5, 0), BackgroundColor3 = Theme.Text, BorderSizePixel = 0, ZIndex = 3, Parent = barBg})
            Corner(6, knob)
            local knobScale = Create("UIScale", {Scale = 1, Parent = knob})
            barBg.MouseEnter:Connect(function() Tween(knobScale, 0.12, {Scale = 1.25}) end)
            barBg.MouseLeave:Connect(function() Tween(knobScale, 0.12, {Scale = 1}) end)

            local function setVal(x, fire)
                local rel = math.clamp((x - barBg.AbsolutePosition.X) / barBg.AbsoluteSize.X, 0, 1)
                local newVal = roundToStep(min + (max - min) * rel)
                if newVal == value and fire then return end
                value = newVal
                local newRel = (value - min) / (max - min)
                fill.Size = UDim2.new(newRel, 0, 1, 0)
                knob.Position = UDim2.new(newRel, 0, 0.5, 0)
                valueLabel.Text = fmt(value)
                if fire and not updating then Safe(callback, value) end
            end
            TrackDrag(barBg, function(x) setVal(x, true) end)

            local api
            api = {Set = function(a, b)
    local v = (type(a) == "table" and a == api) and b or a
    v = roundToStep(tonumber(v) or min)
    if v == value then return end
    updating = true
    value = v
    local rel = (value - min) / (max - min)
    fill.Size = UDim2.new(rel, 0, 1, 0)
    knob.Position = UDim2.new(rel, 0, 0.5, 0)
    valueLabel.Text = fmt(value)
    updating = false
end, Get = function() return value end}
            if flag then ConfigFlags[flag] = api end
            registerThemeElement(frame, function()
                frame.BackgroundColor3 = Theme.Secondary; lbl.TextColor3 = Theme.Text
                valueLabel.TextColor3 = Theme.Accent; barBg.BackgroundColor3 = Theme.Tertiary; fill.BackgroundColor3 = Theme.Accent
            end)
            return api
        end

        -- ─── INPUT SLIDER (слайдер с полем ввода) ───
        function TabAPI:InputSlider(text, min, max, default, callback, flag)
            RegisterSearchable(text, page, tab)
            BeautifulUI:RegisterCommand({Name = text, Keywords = {"слайдер", "slider", "ввод"}, Window = Window, Tab = tab.Name})
            min, max = min or 0, max or 100
            if max == min then max = min + 1 end
            local value = math.clamp(default or min, min, max)
            local updating = false

            local frame = Create("Frame", {Size = UDim2.new(1, 0, 0, 50), BackgroundColor3 = Theme.Secondary, BorderSizePixel = 0, Parent = page})
            Corner(8, frame)
            local lbl = Create("TextLabel", {Size = UDim2.new(0.5, 0, 0, 20), Position = UDim2.new(0, 12, 0, 6), BackgroundTransparency = 1, Font = Enum.Font.GothamMedium, Text = text, TextColor3 = Theme.Text, TextSize = 13, TextXAlignment = Enum.TextXAlignment.Left, Parent = frame})

            Gradient(Theme.Accent, Theme.AccentDark, 0, fill)
            local knob = Create("Frame", {Size = UDim2.new(0, 12, 0, 12), AnchorPoint = Vector2.new(0.5, 0.5), Position = UDim2.new((value - min) / (max - min), 0, 0.5, 0), BackgroundColor3 = Theme.Text, BorderSizePixel = 0, ZIndex = 3, Parent = barBg})

                local rel = (value - min) / (max - min)
                fill.Size = UDim2.new(rel, 0, 1, 0)

                v = tonumber(v); if not v then return end
                v = math.clamp(v, min, max)
                if v == value and not fire then return end
                value = v
                updateUI()

            valueBox.FocusLost:Connect(function() local p = tonumber(valueBox.Text); if p then setValue(p, true) else updateUI() end end)

            local api
            api = {Set = function(a, b)
    local v = (type(a) == "table" and a == api) and b or a
    updating = true
    setValue(v, false)
    updating = false
end, Get = function() return value end}
            if flag then ConfigFlags[flag] = api end
            registerThemeElement(frame, function()
                frame.BackgroundColor3 = Theme.Secondary; lbl.TextColor3 = Theme.Text
                valueBox.BackgroundColor3 = Theme.Tertiary; valueBox.TextColor3 = Theme.Accent

        -- ─── DROPDOWN СО СКРОЛЛОМ ───
        -- multi = true → мультивыбор; opts: {multi = true, maxSelected = nil}
        function TabAPI:Dropdown(text, options, callback, flag, opts)
            opts = opts or {}
            BeautifulUI:RegisterCommand({Name = text, Keywords = {"список", "dropdown"}, Window = Window, Tab = tab.Name})
            local multi = opts.multi and true or false
            options = options or {}
            RegisterSearchable(text .. " " .. table.concat(options, " "), page, tab)
            local frame = Create("Frame", {Size = UDim2.new(1, 0, 0, 36), BackgroundColor3 = Theme.Secondary, BorderSizePixel = 0, ClipsDescendants = true, Parent = page})
            Corner(8, frame)
            local btn = Create("TextButton", {Size = UDim2.new(1, 0, 0, 36), BackgroundTransparency = 1, Text = "", Parent = frame})
            local label = Create("TextLabel", {Size = UDim2.new(1, -40, 1, 0), Position = UDim2.new(0, 12, 0, 0), BackgroundTransparency = 1, Font = Enum.Font.GothamMedium, Text = text .. ": " .. (options[1] or "—"), TextColor3 = Theme.Text, TextSize = 13, TextXAlignment = Enum.TextXAlignment.Left, TextTruncate = Enum.TextTruncate.AtEnd, Parent = btn})
            local arrow = Create("TextLabel", {Size = UDim2.new(0, 30, 1, 0), Position = UDim2.new(1, -30, 0, 0), BackgroundTransparency = 1, Font = Enum.Font.GothamBold, Text = "▼", TextColor3 = Theme.SubText, TextSize = 11, Parent = btn})

            -- 🎯 SCROLLING FRAME для опций
            local list = Create("ScrollingFrame", {
                Size = UDim2.new(1, 0, 0, 0),
                Position = UDim2.new(0, 0, 0, 36),
                BackgroundTransparency = 1,
                BorderSizePixel = 0,
                ScrollBarThickness = 4,
                ScrollBarImageColor3 = Theme.Accent,
                CanvasSize = UDim2.new(0, 0, 0, 0),
                AutomaticCanvasSize = Enum.AutomaticSize.Y,
                ScrollingDirection = Enum.ScrollingDirection.Y,
                Parent = frame
            })
            Create("UIListLayout", {Padding = UDim.new(0, 4), SortOrder = Enum.SortOrder.LayoutOrder, Parent = list})

            local open = false
            local current = options[1]
            local optButtons = {}
            local frameStroke = Stroke(Theme.Tertiary, 1, 1, frame)

            -- 📌 MULTI-SELECT: храним выбранные опции (Set для быстрого поиска)
            local selected, selectedOrder = {}, {}
            if multi and current then selected[current] = true; table.insert(selectedOrder, current) end
            local function selectedList()
                local out = {}
                for i, v in ipairs(selectedOrder) do if selected[v] then out[#out + 1] = v end end
                return out
            end
            local function updateLabel()
                if not multi then return end
                local list2 = selectedList()
                if #list2 == 0 then
                    label.Text = text .. ": —"
                elseif #list2 <= 2 then
                    label.Text = text .. ": " .. table.concat(list2, ", ")
                else
                    label.Text = string.format("%s: %d выбрано", text, #list2)
                end
            end
            local checks = {}
            local function refreshMarks()
                if not multi then return end
                for opt, mark in pairs(checks) do
                    mark.Visible = selected[opt] and true or false
                    local ob = optButtons[opt]
                    if ob then
                        ob.BackgroundColor3 = selected[opt] and Theme.Accent or Theme.Tertiary
                        ob.BackgroundTransparency = selected[opt] and 0.15 or 0
                    end
                end
            end

            local function closeList()
                open = false
                frame.ZIndex = 1
                Tween(list, 0.18, {Size = UDim2.new(1, 0, 0, 0)})
                Tween(frame, 0.18, {Size = UDim2.new(1, 0, 0, 36)})
                arrow.Text = "▼"
            end

            for i, opt in ipairs(options) do
                local optBtn = Create("TextButton", {Size = UDim2.new(1, -8, 0, 28), Position = UDim2.new(0, 4, 0, 0), BackgroundColor3 = Theme.Tertiary, BorderSizePixel = 0, Font = Enum.Font.Gotham, Text = opt, TextColor3 = Theme.Text, TextSize = 12, TextXAlignment = Enum.TextXAlignment.Left, LayoutOrder = i, Parent = list})
                Corner(6, optBtn)
                local optPadding = Create("UIPadding", {PaddingLeft = UDim.new(0, 8), Parent = optBtn})
                -- галочка мультивыбора (в одиночном режиме остаётся скрытой)
                local check = Create("TextLabel", {Size = UDim2.new(0, 14, 1, 0), Position = UDim2.new(0, 4, 0, 0), BackgroundTransparency = 1, Font = Enum.Font.GothamBold, Text = "✓", TextColor3 = Theme.Accent, TextSize = 12, Visible = multi and selected[opt] or false, Parent = optBtn})
                if multi then optPadding.PaddingLeft = UDim.new(0, 22); checks[opt] = check end
                optButtons[opt] = optBtn
                optBtn.MouseEnter:Connect(function() Tween(optBtn, 0.1, {BackgroundColor3 = Theme.Accent}) end)
                optBtn.MouseLeave:Connect(function()
                    Tween(optBtn, 0.1, {BackgroundColor3 = selected[opt] and Theme.Accent or Theme.Tertiary})
                end)
                optBtn.MouseButton1Click:Connect(function()
                    if multi then
                        -- переключаем выбор, список остаётся открытым
                        if selected[opt] then
                            selected[opt] = nil
                        else
                            if opts.maxSelected and #selectedList() >= opts.maxSelected then return end
                            selected[opt] = true
                            table.insert(selectedOrder, opt)
                        end
                        check.Visible = selected[opt] and true or false
                        optBtn.TextColor3 = Theme.Text
                        optBtn.BackgroundColor3 = selected[opt] and Theme.Accent or Theme.Tertiary
                        optBtn.BackgroundTransparency = selected[opt] and 0.15 or 0
                        updateLabel()
                        Safe(callback, selectedList())
                    else
                        current = opt; label.Text = text .. ": " .. opt
                        closeList()
                        Safe(callback, opt)
                    end
                end)
            end
            updateLabel()

            btn.MouseButton1Click:Connect(function()
                open = not open
                local MAX_VISIBLE = 5
                local itemH = 32
                local visibleCount = math.min(#options, MAX_VISIBLE)
                local maxH = visibleCount * itemH + 8

                -- 🐞 ФИКС: выпадающий список больше не прячется за соседними элементами
                frame.ZIndex = open and 40 or 1
                if open then
                    Tween(list, 0.18, {Size = UDim2.new(1, 0, 0, maxH)})
                    Tween(frame, 0.18, {Size = UDim2.new(1, 0, 0, 36 + maxH)})
                    arrow.Text = "▲"
                else
                    Tween(list, 0.18, {Size = UDim2.new(1, 0, 0, 0)})
                    Tween(frame, 0.18, {Size = UDim2.new(1, 0, 0, 36)})
                    arrow.Text = "▼"
                end
            end)

            local api
            api = {}
            patchSet(api, function(v)
                if multi then
                    selected, selectedOrder = {}, {}
                    if type(v) == "table" then
                        for _, item in ipairs(v) do
                            selected[tostring(item)] = true
                            table.insert(selectedOrder, tostring(item))
                        end
                    elseif v ~= nil then
                        selected[tostring(v)] = true
                        selectedOrder = {tostring(v)}
                    end
                    refreshMarks()
                    updateLabel()
                    Safe(callback, selectedList())
                    return
                end
                current = v; label.Text = text .. ": " .. tostring(v)
                if open then closeList() end
                Safe(callback, v)
            end)
            api.Get = function()
                if multi then return selectedList() end
                return current
            end
            if flag then ConfigFlags[flag] = api end
            registerThemeElement(frame, function()
                frame.BackgroundColor3 = Theme.Secondary; label.TextColor3 = Theme.Text; arrow.TextColor3 = Theme.SubText
                frameStroke.Color = Theme.Tertiary
                for _, ob in pairs(optButtons) do
                    ob.BackgroundColor3 = Theme.Tertiary
                    ob.BackgroundTransparency = 0
                    ob.TextColor3 = Theme.Text
                end
                for _, mark in pairs(checks) do mark.TextColor3 = Theme.Text; mark.Visible = false end
            end)
            return api
        end

                currentColor = Color3.fromHSV(hue, sat, val)
                preview.BackgroundColor3 = currentColor
                svBox.BackgroundColor3 = Color3.fromHSV(hue, 1, 1)
                if hexBox and not hexBox:IsFocused() then hexBox.Text = BeautifulUI.ToHex(currentColor) end

                prevSt.Color = Theme.Tertiary; hexBox.BackgroundColor3 = Theme.Tertiary; hexBox.TextColor3 = Theme.Text
            end)
            return api
        end

        -- 🅰️ FONT PICKER — готовый экран настройки шрифта на вкладке
        -- Использование: local Fonts = Tab:FontPicker({Default = "GothamBlack", OnChange = function(name) end})
        -- Fonts.Apply("Code", {Bold = true}) применяет вручную, Fonts.Reset() возвращает исходный.
        function TabAPI:FontPicker(opts)
            opts = opts or {}
            RegisterSearchable("шрифт font text", page, tab)
            BeautifulUI:RegisterCommand({ Name = "Шрифт", Keywords = { "font", "шрифт", "текст" }, Window = Window, Tab = tab.Name })

            local scale = math.floor((FontConfig.SizeScale or 1) * 100 + 0.5)
            local families = opts.Families or {
                "Gotham", "GothamMedium", "GothamBold", "GothamSemibold", "GothamBlack",

            local card = Create("Frame", { Size = UDim2.new(1, 0, 0, 84), BackgroundColor3 = Theme.Secondary, BorderSizePixel = 0, Parent = page })

                    Weight = extra.Weight or opts.Weight,
                    Monospace = extra.Monospace ~= nil and extra.Monospace or opts.Monospace,

                end
            end, "FontCustom")

                apply(state.family, { Bold = on })

                apply(state.family, { Monospace = on })
            end, "FontMono")

            TabAPI.Button(page, "Сбросить шрифт", function()
                BeautifulUI:ResetFont()
                refreshSample()
                modeLbl.Text = fontSupportsFace() and "FontFace" or "Enum.Font"
            end, "Ghost", "↺")

            local api = {
                Apply = apply,
                Reset = function() BeautifulUI:ResetFont() refreshSample() end,
                GetFamily = function() return state.family end,
                IsFaceMode = fontSupportsFace,
            }
            if opts.Default then task.defer(function() apply(opts.Default) end) end
            return api
        end

    local function recalcAutoRange()
        if not autoRange or #dataBuffer == 0 then return end
        local mn, mx = dataBuffer[1], dataBuffer[1]
        for _, v in ipairs(dataBuffer) do
            if v < mn then mn = v end
            if v > mx then mx = v end
        end

        maxValue = mx
        if maxValue == minValue then maxValue = minValue + 1 end
    end

    local function norm(v)
        v = tonumber(v) or 0
        if maxValue == minValue then return 0.5 end
        return math.clamp((v - minValue) / (maxValue - minValue), 0, 1)

        poolIndex = 1

    local function drawLineSegment(x1, y1, x2, y2, color, thickness, zIndex)
        local ix1, iy1 = math.floor(x1 + 0.5), math.floor(y1 + 0.5)
        local ix2, iy2 = math.floor(x2 + 0.5), math.floor(y2 + 0.5)

        local dx = ix2 - ix1
        local dy = iy2 - iy1
        local len = math.sqrt(dx * dx + dy * dy)

        if len < 1 then return end

        local seg = acquire()
        seg.Size = UDim2.new(0, len, 0, thickness or 2)
        seg.Position = UDim2.new(0, ix1, 0, iy1)
        seg.AnchorPoint = Vector2.new(0, 0.5)
        seg.BackgroundColor3 = color
        seg.BackgroundTransparency = 0
        seg.Rotation = math.deg(math.atan2(dy, dx))
        seg.ZIndex = zIndex or 10
    end

    -- --- ОСНОВНАЯ ФУНКЦИЯ ПЕРЕРИСОВКИ ---
    local function redrawUnsafe()
        beginDraw()

        local w = canvas.AbsoluteSize.X
        local h = canvas.AbsoluteSize.Y
        if w <= 0 or h <= 0 then finishDraw(); return end

        local pad = 10
        local usableW = w - pad * 2
        local usableH = h - pad * 2
        if usableW < 10 or usableH < 10 then finishDraw(); return end

        local displayData = dataBuffer

        if graphType == "line" or graphType == "area" or graphType == "stacked" then
            
            -- Area Fill
            if graphType == "area" and #displayData >= 2 then
                for i = 1, #displayData - 1 do
                    local x = pad + (i - 1) / (#displayData - 1) * usableW
                    local y1 = pad + usableH - norm(displayData[i]) * usableH
                    local y2 = pad + usableH - norm(displayData[i+1]) * usableH
                    
                    local rect = acquire()
                    rect.Size = UDim2.new(0, usableW/(#displayData-1)+1, 0, math.abs(y2-y1))
                    rect.Position = UDim2.new(0, math.floor(x+0.5), 0, math.floor(math.min(y1,y2)+0.5))
                    rect.BackgroundColor3 = seriesMeta.color
                    rect.BackgroundTransparency = 0.6
                    rect.Rotation = 0
                    rect.AnchorPoint = Vector2.new(0, 0)
                    rect.ZIndex = 8
                end
            end

            -- Lines
            if #displayData >= 2 then
                for i = 1, #displayData - 1 do
                    local x1 = pad + (i - 1) / (#displayData - 1) * usableW
                    local x2 = pad + i / (#displayData - 1) * usableW
                    local y1 = pad + usableH - norm(displayData[i]) * usableH
                    local y2 = pad + usableH - norm(displayData[i+1]) * usableH
                    
                    drawLineSegment(x1, y1, x2, y2, seriesMeta.color, 2, 10)
                end
            end

            -- Interactive Dots (Рисуем ВСЕГДА, так как их макс 35)
            -- Но БЕЗ отдельных hitbox кнопок! Мы будем использовать MouseMove на канвасе.
            if #displayData > 0 then
                 for i, v in ipairs(displayData) do
                    local x = pad + (i - 1) / math.max(#displayData - 1, 1) * usableW
                    local y = pad + usableH - norm(v) * usableH
                    
                    local dot = acquire()
                    dot.Size = UDim2.new(0, 5, 0, 5)
                    dot.Position = UDim2.new(0, math.floor(x-2.5+0.5), 0, math.floor(y-2.5+0.5))
                    dot.AnchorPoint = Vector2.new(0, 0)
                    dot.Rotation = 0
                    dot.BackgroundColor3 = seriesMeta.color
                    dot.BackgroundTransparency = 0
                    dot.ZIndex = 12
                end
            end

        elseif graphType == "scatter" then
            -- ✨ SCATTER: точки без соединяющих линий (шум, разброс, корреляция)
            for i, v in ipairs(displayData) do
                local x = pad + (i - 1) / math.max(#displayData - 1, 1) * usableW
                local y = pad + usableH - norm(v) * usableH
                local dot = acquire()
                dot.Size = UDim2.new(0, 7, 0, 7)
                dot.Position = UDim2.new(0, math.floor(x - 3.5 + 0.5), 0, math.floor(y - 3.5 + 0.5))
                dot.AnchorPoint = Vector2.new(0, 0)
                dot.Rotation = 0
                dot.BackgroundColor3 = seriesMeta.color
                dot.BackgroundTransparency = 0.1
                dot.ZIndex = 10
            end

        elseif graphType == "pie" then
            -- 🥧 PIE / DOUGHNUT: доли от суммы, кольцо из сегментов
            local total = 0
            for _, v in ipairs(displayData) do total = total + math.max(tonumber(v) or 0, 0) end
            local cx, cy = w / 2, h / 2
            local radius = math.min(w, h) / 2 - 8
            local thickness = math.max(14, radius * 0.42)
            local segN = math.max(8, math.min(60, #displayData * 8))
            local cursor = -math.pi / 2

            if total > 0 and radius > 8 then
                for i, v in ipairs(displayData) do
                    local val = math.max(tonumber(v) or 0, 0)
                    local sweep = (val / total) * math.pi * 2
                    if sweep > 0 then
                    local col = palette[((i - 1) % #palette) + 1]
                    local steps = math.max(1, math.ceil(sweep / (math.pi * 2 / segN)))
                    for s = 0, steps - 1 do
                        local ang = cursor + sweep * (s / steps)
                        local dx = math.cos(ang) * radius
                        local dy = math.sin(ang) * radius
                        local dot = acquire()
                        local dotSize = math.max(3, math.ceil(math.pi * 2 * radius / segN) + 2)
                        dot.Size = UDim2.new(0, thickness, 0, dotSize)
                        dot.Position = UDim2.new(0, math.floor(cx + dx - thickness / 2 + 0.5), 0, math.floor(cy + dy - dotSize / 2 + 0.5))
                        dot.AnchorPoint = Vector2.new(0, 0)
                        dot.Rotation = math.deg(ang) + 90
                        dot.BackgroundColor3 = col
                        dot.BackgroundTransparency = 0
                        dot.ZIndex = 10
                    end
                    cursor = cursor + sweep
                    end
                end
                -- центр: дырка для doughnut-вида
                local hole = acquire()
                local holeR = math.max(6, radius - thickness - 4)
                hole.Size = UDim2.new(0, holeR * 2, 0, holeR * 2)
                hole.Position = UDim2.new(0.5, 0, 0.5, 0)
                hole.AnchorPoint = Vector2.new(0.5, 0.5)
                hole.Rotation = 0
                hole.BackgroundColor3 = Theme.Background
                hole.BackgroundTransparency = 0
                hole.ZIndex = 11
            end

        elseif graphType == "radial" then
            -- 🕸 RADIAL / RADAR: лучи из центра, длина = значение
            local cx, cy = w / 2, h / 2
            local radius = math.min(w, h) / 2 - 10
            local n = #displayData
            if n > 0 and radius > 8 then
                for i, v in ipairs(displayData) do
                    local ang = (i - 1) / n * math.pi * 2 - math.pi / 2
                    local len = norm(v) * radius
                    if len >= 2 then
                    local col = palette[((i - 1) % #palette) + 1]
                    local ray = acquire()
                    ray.Size = UDim2.new(0, len, 0, 3)
                    ray.Position = UDim2.new(0, math.floor(cx + 0.5), 0, math.floor(cy + 0.5))
                    ray.AnchorPoint = Vector2.new(0, 0.5)
                    ray.Rotation = math.deg(ang)
                    ray.BackgroundColor3 = col
                    ray.BackgroundTransparency = 0.15
                    ray.ZIndex = 10

                    -- кончик луча
                    local tip = acquire()
                    tip.Size = UDim2.new(0, 6, 0, 6)
                    tip.Position = UDim2.new(0, math.floor(cx + math.cos(ang) * len - 3 + 0.5),
                                             0, math.floor(cy + math.sin(ang) * len - 3 + 0.5))
                    tip.AnchorPoint = Vector2.new(0, 0)
                    tip.Rotation = 0
                    tip.BackgroundColor3 = col
                    tip.BackgroundTransparency = 0
                    tip.ZIndex = 12
                    end
                end
                -- соединяем кончики (многоугольник)
                for i = 1, n do
                    local a1 = (i - 1) / n * math.pi * 2 - math.pi / 2
                    local a2 = (i % n) / n * math.pi * 2 - math.pi / 2
                    local r1, r2 = norm(displayData[i]) * radius, norm(displayData[i % n + 1] or displayData[1]) * radius
                    local x1, y1 = cx + math.cos(a1) * r1, cy + math.sin(a1) * r1
                    local x2, y2 = cx + math.cos(a2) * r2, cy + math.sin(a2) * r2
                    drawLineSegment(x1, y1, x2, y2, seriesMeta.color, 1, 9)
                    -- ось
                    drawLineSegment(cx, cy, cx + math.cos(a1) * radius, cy + math.sin(a1) * radius, Theme.Tertiary, 1, 8)
                end
            end

        elseif graphType == "bar" then
             local n = #displayData
             if n == 0 then finishDraw(); return end
             
             local gap = 2
             local barW = math.max((usableW - gap * (n - 1)) / n, 2)
             
             for i, v in ipairs(displayData) do
                local x = pad + (i - 1) * (barW + gap)
                local barH = norm(v) * usableH
                
                local bar = acquire()
                bar.Size = UDim2.new(0, barW, 0, barH)
                bar.Position = UDim2.new(0, math.floor(x+0.5), 1, -pad - math.floor(barH+0.5))
                bar.AnchorPoint = Vector2.new(0, 0)
                bar.Rotation = 0
                bar.BackgroundColor3 = seriesMeta.color
                bar.BackgroundTransparency = 0
                bar.ZIndex = 10
             end
        end

        -- 📋 легенда: статичная (opts.Legend) или авто-доли для pie
        if opts.Legend ~= nil and #displayData > 0 then
            renderLegend(opts.Legend)
        elseif opts.LegendPie and graphType == "pie" then
            legendFromPie()
        end

        -- 📏 обновляем подписи осей
        if opts.ShowAxis ~= false and yLabels[1] and yLabels[1].Visible then
            local function fmt(v)
                v = tonumber(v) or 0
                if math.abs(v) >= 1000 then return string.format("%.1fk", v / 1000) end
                if math.abs(v) >= 100 then return string.format("%d", math.floor(v + 0.5)) end
                return string.format("%.1f", v)
            end
            yLabels[1].Text = fmt(maxValue)
            yLabels[2].Text = fmt((minValue + maxValue) / 2)
            yLabels[3].Text = fmt(minValue)
            if #displayData > 0 then
                xLabel.Text = string.format("последнее: %s · точек: %d/%d", fmt(displayData[#displayData]), #displayData, MAX_POINTS)
            else
                xLabel.Text = "нет данных"
            end
        end
        finishDraw()

        -- Обновление позиции курсора
        if viewOffset then
            local cursorX = pad + viewOffset * usableW
            cursorLine.Position = UDim2.new(0, math.floor(cursorX + 0.5), 0, 0)
            cursorLine.Visible = true
            
            local idx = math.floor(viewOffset * (#displayData - 1) + 0.5) + 1
            idx = math.clamp(idx, 1, #displayData)
            local val = displayData[idx]
            
            statText.Text = string.format("T:%d/%d | Val: %.2f | Range: [%.2f..%.2f]", 
                idx, #displayData, val or 0, minValue, maxValue)
            statText.TextColor3 = Theme.Success
        else
            cursorLine.Visible = false
            -- Если нет активного выбора, показываем общую сводку
            if #displayData > 0 then
                 local lastVal = displayData[#displayData]
                 statText.Text = string.format("Live | Last: %.2f | Points: %d", lastVal, #displayData)
                 statText.TextColor3 = Theme.SubText
            else
                 statText.Text = "No Data"
                 statText.TextColor3 = Theme.Danger
            end
        end
    end

    -- 🛡️ Ошибка в отрисовке больше не спамит в консоль каждый кадр
    local redrawWarned = false
    local function redraw()
        local ok, err = pcall(redrawUnsafe)
        if not ok and not redrawWarned then
            redrawWarned = true
            warn("[BeautifulUI Graph] draw error:", err)
        end
    end

    -- --- ОБРАБОТЧИКИ СОБЫТИЙ ---

    -- 1. Dragging Scrubber Knob
    local dragging = false
    local startMouseX = 0
    local startPos = 0

    scrubberKnob.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            startMouseX = input.Position.X
            startPos = scrubberKnob.Position.X.Scale
        end
    end)

    UserInputService.InputChanged:Connect(function(input)
        if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
            local delta = input.Position.X - startMouseX
            local newScale = math.clamp(startPos + delta / scrubberBg.AbsoluteSize.X, 0, 1)
            
            scrubberKnob.Position = UDim2.new(newScale, 0, 0.5, 0)
            scrubberFill.Size = UDim2.new(newScale, 0, 1, 0)
            
            viewOffset = newScale
            requestRedraw()
        end
    end)

    UserInputService.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = false
        end
    end)

    -- 2. HOVER LOGIC (Магия без лишних объектов!)
    -- Слушаем движение мыши ПО КАНАВУ. Вычисляем ближайшую точку математически.
    canvas.MouseMoved:Connect(function(x, y)
        -- Преобразуем глобальные координаты мыши в локальные координаты канваса
        local absPos = canvas.AbsolutePosition
        local relX = x - absPos.X
        local relY = y - absPos.Y
        
        local w = canvas.AbsoluteSize.X
        local h = canvas.AbsoluteSize.Y
        local pad = 10
        local usableW = w - pad * 2
        
        -- Проверяем, внутри ли рабочей области
        if relX >= pad and relX <= w - pad then
            -- Находим относительную позицию X (0.0 до 1.0)
            local ratio = (relX - pad) / usableW
            
            -- Устанавливаем viewOffset временно для подсветки
            viewOffset = ratio
            
            -- Перерисовываем график, чтобы показать курсор и статистику
            -- Важно: не делаем full redraw слишком часто, можно добавить throttle, 
            -- но для 35 точек это не критично.
            requestRedraw()
        end
    end)

    canvas.MouseLeave:Connect(function()
        -- Возвращаемся в режим LIVE, если мышь ушла с графика
        -- Но только если ползунок не был зажат вручную? 
        -- Для простоты: уход мыши сбрасывает выбор, если ползунок не трогали активно.
        -- Чтобы избежать конфликтов, давайте оставим возможность ручного управления ползунком.
        -- Здесь мы просто скрываем курсор, если мышь далеко.
         -- Лучше оставить поведение: Hover показывает инфо, клик фиксирует.
         -- Пока что: при уходе мыши ничего не меняем, чтобы пользователь мог читать данные.
         -- Или сбросим, если хотим чистый Live:
         -- viewOffset = nil; requestRedraw() 
    end)
    
    -- Клик по графику фиксирует точку (как раньше)
    canvas.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 then
            local absPos = canvas.AbsolutePosition
            local relX = input.Position.X - absPos.X
            local w = canvas.AbsoluteSize.X
            local pad = 10
            local usableW = w - pad * 2
            
            if relX >= pad and relX <= w - pad then
                local ratio = (relX - pad) / usableW
                viewOffset = ratio
                
                -- Синхронизируем ползунок визуально
                scrubberKnob.Position = UDim2.new(ratio, 0, 0.5, 0)
                scrubberFill.Size = UDim2.new(ratio, 0, 1, 0)
                
                requestRedraw()
            end
        end
    end)
    
    -- Кнопка LIVE
    local liveBtn = Create("TextButton", {
        Size = UDim2.new(0, 40, 0, 16),
        Position = UDim2.new(1, -50, 0, 6),
        BackgroundColor3 = Theme.Success,
        BorderSizePixel = 0,
        Font = Enum.Font.GothamBold,
        Text = "LIVE",
        TextColor3 = Theme.Text,
        TextSize = 10,
        Parent = frame
    })
    Corner(4, liveBtn)
    liveBtn.MouseButton1Click:Connect(function()
        viewOffset = nil
        scrubberKnob.Position = UDim2.new(1, 0, 0.5, 0)
        scrubberFill.Size = UDim2.new(1, 0, 1, 0)
        requestRedraw()
    end)


    -- --- API МЕТОДЫ ---

    local function requestRedraw()
        needsRedraw = true
    end

    local renderConn
    renderConn = RunService.Heartbeat:Connect(function()
        if not frame.Parent then
            if renderConn then renderConn:Disconnect() end
            return
        end
        if needsRedraw then
            if canvas.AbsoluteSize.X >= 20 then
                needsRedraw = false
                pcall(redraw)
            end
        end
    end)

    canvas:GetPropertyChangedSignal("AbsoluteSize"):Connect(function()
        if canvas.AbsoluteSize.X >= 20 then requestRedraw() end
    end)

    registerThemeElement(frame, function()
        frame.BackgroundColor3 = Theme.Secondary
        st.Color = Theme.Accent
        titleLbl.TextColor3 = Theme.Text
        canvas.BackgroundColor3 = Theme.Background
        scrubberBg.BackgroundColor3 = Theme.Tertiary
        scrubberKnob.BackgroundColor3 = Theme.Text
        seriesMeta.color = palette[1] 
        
        for _, gl in ipairs(gridLines) do gl.BackgroundColor3 = Theme.Tertiary end
        requestRedraw()
    end)

    api.AddPoint = function(first, second)
        local v = first == api and second or first
        v = tonumber(v) or 0
        
        table.insert(dataBuffer, v)
        
        -- 🔥 ОТСЕКАЕМ СТАРЫЕ ДАННЫЕ СТРОГО ПОСЛЕ 35 ТОЧКИ
        while #dataBuffer > MAX_POINTS do
            table.remove(dataBuffer, 1)
        end
        
        if autoRange then recalcAutoRange() end
        
        requestRedraw()
        return api
    end

    api.SetData = function(first, second)
        local arr = first == api and second or first
        dataBuffer = {}
        if type(arr) == "table" then
            for _, v in ipairs(arr) do 
                table.insert(dataBuffer, tonumber(v) or 0) 
            end
        end
        -- Обрезаем сразу после установки
        while #dataBuffer > MAX_POINTS do
            table.remove(dataBuffer, 1)
        end
        if autoRange then recalcAutoRange() end
        requestRedraw()
        return api
    end
    
    api.SetRange = function(mn, mx)
        minValue = tonumber(mn) or 0
        maxValue = tonumber(mx) or 100
        if maxValue == minValue then maxValue = minValue + 1 end
        autoRange = false
        requestRedraw()
        return api
    end

    api.SetAutoRange = function(enabled)
        autoRange = enabled ~= false
        recalcAutoRange()
        requestRedraw()
        return api
    end

    -- 🎨 Смена цвета/линии/лимита точек на лету
    api.SetColor = function(color)
        seriesMeta.color = C(color, seriesMeta.color)
        requestRedraw()
        return api
    end

    api.SetMaxPoints = function(n)
        MAX_POINTS = math.clamp(tonumber(n) or MAX_POINTS, 2, 500)
        while #dataBuffer > MAX_POINTS do table.remove(dataBuffer, 1) end
        requestRedraw()
        return api
    end

    api.Clear = function()
        dataBuffer = {}
        requestRedraw()
        return api
    end

    -- ─── 📋 ЛЕГЕНДА (для pie / radial / нескольких серий) ───
    local legendFrame, legendRows = nil, {}
    local legendHidden = false   -- SetLegend(nil) — до явного SetLegend(...)
    -- куда вставлять пункты: контейнер, если он есть, иначе сам кадр легенды
    local function legendHost()
        if not legendFrame then return nil end
        for _, c in ipairs(legendFrame:GetChildren()) do
            if c.__class == "UIListLayout" then return legendFrame end
        end
        return legendFrame
    end
    local legendItems = opts.Legend

    local function ensureLegend()
        if legendFrame then return legendFrame end
        legendFrame = Instance.new("Frame")
        legendFrame.Name = "GraphLegend"
        legendFrame.Size = UDim2.new(1, -24, 0, 0)
        legendFrame.Position = UDim2.new(0, 12, 0, height + 26)
        legendFrame.BackgroundTransparency = 1
        legendFrame.Parent = frame
        local layout = Create("UIListLayout", {
            FillDirection = Enum.FillDirection.Horizontal,
            Padding = UDim.new(0, 10),
            SortOrder = Enum.SortOrder.LayoutOrder,
            Parent = legendFrame,
        })
        local pad = Create("UIPadding", { PaddingTop = UDim.new(0, 2), Parent = legendFrame })
        -- высота кадра растёт под легенду
        frame.Size = UDim2.new(1, 0, 0, height + 70 + 22)
        return legendFrame
    end

    -- items: {{Name="A", Value=12, Color="#ff0"}, ...} или {Name=...}
    function renderLegend(items, autoFrom)
        if legendHidden then return end
        if not items and not autoFrom then return end
        local host = ensureLegend()
        -- чистим всё лишнее (и осиротевшие пункты от прошлых вызовов)
        for _, c in ipairs(host:GetChildren()) do
            if c.__class ~= "UIListLayout" and c.__class ~= "UIPadding" then c:Destroy() end
        end
        legendRows = {}
        legendItems = items
        local src = items
        if not src and autoFrom then
            src = {}
            for i, v in ipairs(autoFrom) do
                src[i] = { Name = opts.LegendLabels and opts.LegendLabels[i] or ("#" .. i), Value = v }
            end
        end
        for i, item in ipairs(src or {}) do
            -- один контейнер на пункт: чип + подпись (без дублей в иерархии)
            local wrap = Create("Frame", {
                Size = UDim2.new(0, 0, 1, 0),
                BackgroundTransparency = 1,
                AutomaticSize = Enum.AutomaticSize.X,
                LayoutOrder = i,
                Parent = host,
            })
            local chip = Create("Frame", {
                Size = UDim2.new(0, 10, 0, 10),
                Position = UDim2.new(0, 0, 0.5, -5),
                BackgroundColor3 = C(item.Color or palette[((i - 1) % #palette) + 1]),
                BorderSizePixel = 0,
                ZIndex = 3,
                Parent = wrap,
            })
            Corner(3, chip)
            Create("TextLabel", {
                Size = UDim2.new(0, 0, 1, 0),
                Position = UDim2.new(0, 15, 0, 0),
                BackgroundTransparency = 1,
                Font = Enum.Font.Gotham,
                Text = tostring(item.Name or "") .. (item.Value ~= nil and ("  " .. tostring(item.Value)) or ""),
                TextColor3 = Theme.SubText,
                TextSize = 11,
                TextXAlignment = Enum.TextXAlignment.Left,
                AutomaticSize = Enum.AutomaticSize.X,
                Parent = wrap,
            })
            table.insert(legendRows, wrap)
        end
    end

    -- автолегенда для pie: доли в процентах
    function legendFromPie()
        if #dataBuffer == 0 then return end
        local total = 0
        for _, v in ipairs(dataBuffer) do total = total + math.max(tonumber(v) or 0, 0) end
        if total <= 0 then return end
        local items = {}
        for i, v in ipairs(dataBuffer) do
            items[i] = {
                Name = (opts.LegendLabels and opts.LegendLabels[i]) or ("#" .. i),
                Value = string.format("%.0f%%", math.max(tonumber(v) or 0, 0) / total * 100),
                Color = palette[((i - 1) % #palette) + 1],
            }
        end
        renderLegend(items)
    end

    -- 🏷 подпись поверх графика (например «FPS: 60»)
    api.SetLabel = function(txt)
        titleLbl.Text = tostring(txt or text)
        requestRedraw()
        return api
    end

    -- 📊 добавить точку и сразу обновить подпись
    api.PushAndLabel = function(v, fmtStr)
        api.AddPoint(v)
        if fmtStr then titleLbl.Text = string.format(fmtStr, tonumber(v) or 0) end
        requestRedraw()
        return api
    end

    -- 🎨 своя палитра для pie/radial
    api.SetPalette = function(colors)
        if type(colors) ~= "table" then return api end
        local out = {}
        for i, c in ipairs(colors) do out[i] = C(c, palette[((i - 1) % #palette) + 1]) end
        if #out > 0 then palette = out end
        requestRedraw()
        return api
    end

    -- 📋 SetLegend: {{Name="Сервер", Value="12%", Color="#ff0"}, ...}  (nil — скрыть)
    api.SetLegend = function(first, second)
        -- поддержка и точки (g:SetLegend(items)), и вызова через точку
        local arg
        if first == api then arg = second else arg = first end
        local items = (type(arg) == "table") and arg or nil
        if items == nil then
            for _, row in ipairs(legendRows) do row:Destroy() end
            legendRows = {}
            legendItems = nil
            legendHidden = true
            if legendFrame then legendFrame.Visible = false end
            return api
        end
        legendHidden = false
        if legendFrame then legendFrame.Visible = true end
        renderLegend(items)
        return api
    end

    api.GetLegend = function() return legendItems end

    -- вернуть сам кадр легенды (для тонкой настройки)
    api.GetLegendFrame = function() return legendFrame end

    api.GetData = function()
        local copy = {}
        for i, v in ipairs(dataBuffer) do copy[i] = v end
        return copy
    end
    
    api.Destroy = function()
        if renderConn then renderConn:Disconnect() end
        clearContainer()
        if frame.Parent then frame:Destroy() end
    end

    RegisterSearchable(text or "graph", page, tab)

    task.defer(requestRedraw)
    return api
end
        return TabAPI
    end

    -- ─── ХЕЛПЕРЫ ОКНА ───
    function Window:Notify(t, txt, dur, tp, btns) BeautifulUI:Notify(t, txt, dur, tp, btns) end
    function Window:DialogBox(cfg) return BeautifulUI:DialogBox(cfg) end

    return Window
end

-- Конфиг: config.Discord = {Invite = "мойсервер", Remember = true, Reminder = 300}
--   Invite    — код приглашения (без discord.gg/)
--   Remember  — true: подсказка один раз, навсегда запомнить «уже вступил»
--   Reminder  — через сколько секунд напомнить снова (0 — не напоминать)
local DiscordConfig = {Invite = nil, Remember = true, Reminder = 300, Name = nil}
local DiscordLastReminder = 0
local DiscordPrompted = false
local DiscordInfoCache = {time = 0, data = nil}

function BeautifulUI:SetDiscord(config)
    if type(config) == "string" then config = {Invite = config} end
    if type(config) ~= "table" then return DiscordConfig end
    for k, v in pairs(config) do DiscordConfig[k] = v end
    DiscordInfoCache = {time = 0, data = nil}
end

function BeautifulUI:GetDiscordConfig() return DiscordConfig end

-- Полный URL инвайта
function BeautifulUI:GetDiscordURL()
    if not DiscordConfig.Invite or DiscordConfig.Invite == "" then return nil end
    return "https://discord.gg/" .. tostring(DiscordConfig.Invite)
end

-- 12345 → «12.3k», 1500000 → «1.5M»
function BeautifulUI:FormatCount(n)
    n = tonumber(n)
    if not n then return "—" end
    if n >= 1000000 then return string.format("%.1fM", n / 1000000) end
    if n >= 1000 then return string.format("%.1fk", n / 1000) end
    return tostring(math.floor(n))
end

-- Открывает приглашение в браузере
function BeautifulUI:OpenDiscord()
    local url = self:GetDiscordURL()
    if not url then
        return self:Notify("Discord", "Инвайт не настроен: BeautifulUI:SetDiscord({Invite = \"код\"})", 4, "Warning")
    end
    local ok = pcall(function() setclipboard(url) end)
    if not ok then pcall(function() game:GetService("Players"):SetClipboard(url) end) end
    self:Notify("Discord", "Ссылка скопирована:\n" .. url, 5, "Info",
        {{Text = "Вступить", Callback = function() pcall(function() setclipboard(url) end) end}})
    return url
end

-- Живая статистика сервера (кэш 5 минут)
-- 👉 Подставь свой endpoint, если он есть:
-- BeautifulUI.DiscordEndpoint = function(invite) return "https://мой-сайт/api/invite/" .. invite end
function BeautifulUI:GetDiscordInfo(inviteOverride)
    local invite = inviteOverride or DiscordConfig.Invite
    if not invite or invite == "" then return nil end
    local now = os.clock()
    if DiscordInfoCache.data and (now - DiscordInfoCache.time) < 300 then return DiscordInfoCache.data end

    local data = {invite = invite, name = DiscordConfig.Name}

    if BeautifulUI.DiscordEndpoint and HttpService.HttpEnabled ~= false then
        local ok, body = pcall(function()
            return HttpService:JSONDecode(game:HttpGet(BeautifulUI.DiscordEndpoint(invite)))
        end)
        if ok and type(body) == "table" then
            data.name = body.name or data.name
            data.members = body.members or body.member_count or body.approximate_member_count
            data.online = body.online or body.presence_count or body.approximate_presence_count
        else
            data.error = true
        end
    end

    DiscordInfoCache = {time = now, data = data}
    return data
end

-- Обновить виджеты во всех окнах
function BeautifulUI:RefreshDiscord()
    DiscordInfoCache = {time = 0, data = nil}
    for _, w in ipairs(OpenWindows) do
        if w._refreshDiscord then pcall(w._refreshDiscord) end
    end
end

-- Кнопка «вступить» с показом ссылки (используется в шапке и сайдбаре)
function BeautifulUI:ShowDiscord()
    local info = self:GetDiscordInfo()
    local title = (info and info.name) or "Discord"
    local text
    if info and info.members then
        text = string.format("%s онлайн · %s участников", self:FormatCount(info.online), self:FormatCount(info.members))
    else
        text = "Присоединяйтесь к сообществу"
    end
    if DiscordConfig.Invite then text = text .. "\ndiscord.gg/" .. tostring(DiscordConfig.Invite) end
    return self:Notify(title, text, 6, "Info", {
        {Text = "Копировать", Color = Theme.Accent, Callback = function() self:OpenDiscord() end},
        {Text = "Закрыть", Color = Theme.Tertiary},
    })
end

-- Напоминание о вступлении (вызывается из throttle-цикла внизу файла)
function BeautifulUI:DiscordTick()
    local cfg = DiscordConfig
    if not cfg.Invite or cfg.Invite == "" then return end
    if cfg.Remember and DiscordPrompted then return end
    local now = os.clock()
    if cfg.Reminder and cfg.Reminder > 0 and (now - DiscordLastReminder) < cfg.Reminder then return end
    DiscordLastReminder = now
    DiscordPrompted = true

    self:Notify("Наш Discord", "Там обновления, читы и поддержка\ndiscord.gg/" .. tostring(cfg.Invite), 8, "Info", {
        {Text = "Вступить", Color = Theme.Info, Callback = function() self:OpenDiscord() end},
        {Text = "Позже", Color = Theme.Tertiary, Callback = function()
            if not cfg.Remember then DiscordPrompted = false end
        end},
    })
end

local CommandList = {}

function BeautifulUI:RegisterCommand(cmd)
    if type(cmd) ~= "table" or not cmd.Name then return nil end
    table.insert(CommandList, cmd)
    return cmd
end

function BeautifulUI:GetCommands() return CommandList end

local CommandGui, CommandBox, CommandResults, CommandFrame

local function CloseCommandPalette()
    if CommandGui then CommandGui.Enabled = false end
end

local function RunCommand(cmd)
    if not cmd then return end
    local w = cmd.Window
    if w then
        if not (w.IsVisible and w:IsVisible()) then w:Show() end
        if cmd.Tab and w.SelectTab then pcall(function() w:SelectTab(cmd.Tab) end) end
        if cmd.Element then FlashElement(cmd.Element, Theme.Accent, 2) end
    end
    Safe(cmd.Callback)
end

local function RenderCommandResults()
    if not CommandResults then return end
    for _, c in ipairs(CommandResults:GetChildren()) do
        if c:IsA("TextButton") then c:Destroy() end
    end
    local q = (CommandBox.Text or ""):lower()
    local shown, firstMatch = 0, nil
    for _, cmd in ipairs(CommandList) do
        local keywords = cmd.Keywords or {}
        local haystack = (tostring(cmd.Name) .. " " .. table.concat(keywords, " ") .. " " .. tostring(cmd.Tab or "")):lower()
        if q == "" or haystack:find(q, 1, true) then
            shown = shown + 1
            firstMatch = firstMatch or cmd
            if shown <= 40 then
                local row = Create("TextButton", {Size = UDim2.new(1, -4, 0, 34), BackgroundColor3 = Theme.Tertiary, BackgroundTransparency = 0.35, BorderSizePixel = 0, Text = "", AutoButtonColor = false, LayoutOrder = shown, Parent = CommandResults})
                Corner(8, row)
                Create("TextLabel", {Size = UDim2.new(1, -96, 1, 0), Position = UDim2.new(0, 10, 0, 0), BackgroundTransparency = 1, Font = Enum.Font.GothamMedium, Text = tostring(cmd.Name), TextColor3 = Theme.Text, TextSize = 13, TextXAlignment = Enum.TextXAlignment.Left, TextTruncate = Enum.TextTruncate.AtEnd, Parent = row})
                Create("TextLabel", {Size = UDim2.new(0, 86, 1, 0), Position = UDim2.new(1, -90, 0, 0), BackgroundTransparency = 1, Font = Enum.Font.Gotham, Text = tostring(cmd.Tab or ""), TextColor3 = Theme.SubText, TextSize = 10, TextXAlignment = Enum.TextXAlignment.Right, TextTruncate = Enum.TextTruncate.AtEnd, Parent = row})
                row.MouseEnter:Connect(function() Tween(row, 0.1, {BackgroundTransparency = 0.1}) end)
                row.MouseLeave:Connect(function() Tween(row, 0.1, {BackgroundTransparency = 0.35}) end)
                OnTap(row, function()
                    CloseCommandPalette()
                    RunCommand(cmd)
                end)
            end
        end
    end
    if shown == 0 then
        Create("TextLabel", {Size = UDim2.new(1, 0, 0, 32), BackgroundTransparency = 1, Font = Enum.Font.Gotham, Text = "Ничего не найдено", TextColor3 = Theme.SubText, TextSize = 12, Parent = CommandResults})
    end
    CommandFrame.Size = UDim2.new(0, 480, 0, 54 + math.min(math.max(shown, 1), 8) * 37)
    return firstMatch
end

local function EnsureCommandPalette()
    if CommandGui and CommandGui.Parent then return end
    CommandGui = Create("ScreenGui", {Name = "BeautifulUICommandPalette", ResetOnSpawn = false, Enabled = false, DisplayOrder = 300, ZIndexBehavior = Enum.ZIndexBehavior.Sibling, Parent = (gethui and gethui()) or CoreGui})
    local backdrop = Create("TextButton", {Size = UDim2.new(1, 0, 1, 0), BackgroundColor3 = Color3.new(0, 0, 0), BackgroundTransparency = 0.45, Text = "", AutoButtonColor = false, Parent = CommandGui})
    CommandFrame = Create("Frame", {Size = UDim2.new(0, 480, 0, 54), Position = UDim2.new(0.5, -240, 0, 140), BackgroundColor3 = Theme.Secondary, BorderSizePixel = 0, Parent = CommandGui})
    Corner(12, CommandFrame)
    Stroke(Theme.Accent, 1.5, 0.4, CommandFrame)
    local searchWrap = Create("Frame", {Size = UDim2.new(1, -20, 0, 40), Position = UDim2.new(0, 10, 0, 8), BackgroundColor3 = Theme.Tertiary, BackgroundTransparency = 0.2, BorderSizePixel = 0, Parent = CommandFrame})
    Corner(9, searchWrap)
    Create("TextLabel", {Size = UDim2.new(0, 24, 1, 0), Position = UDim2.new(0, 8, 0, 0), BackgroundTransparency = 1, Font = Enum.Font.GothamBold, Text = "⌕", TextColor3 = Theme.SubText, TextSize = 16, Parent = searchWrap})
    CommandBox = Create("TextBox", {Size = UDim2.new(1, -44, 1, 0), Position = UDim2.new(0, 34, 0, 0), BackgroundTransparency = 1, Font = Enum.Font.Gotham, Text = "", PlaceholderText = "Поиск команды…  (Enter — перейти, Esc — закрыть)", PlaceholderColor3 = Theme.SubText, TextColor3 = Theme.Text, TextSize = 14, TextXAlignment = Enum.TextXAlignment.Left, ClearTextOnFocus = false, Parent = searchWrap})
    CommandResults = Create("ScrollingFrame", {Size = UDim2.new(1, -8, 0, 0), Position = UDim2.new(0, 4, 1, 6), BackgroundTransparency = 1, BorderSizePixel = 0, ScrollBarThickness = 3, ScrollBarImageColor3 = Theme.Accent, CanvasSize = UDim2.new(0, 0, 0, 0), AutomaticCanvasSize = Enum.AutomaticSize.Y, ScrollingDirection = Enum.ScrollingDirection.Y, Parent = CommandFrame})
    Create("UIListLayout", {Padding = UDim.new(0, 3), SortOrder = Enum.SortOrder.LayoutOrder, Parent = CommandResults})

    CommandBox:GetPropertyChangedSignal("Text"):Connect(function() RenderCommandResults() end)
    CommandBox.FocusLost:Connect(function(enter)
        if not enter then return end
        local q = (CommandBox.Text or ""):lower()
        for _, cmd in ipairs(CommandList) do
            local haystack = (tostring(cmd.Name) .. " " .. table.concat(cmd.Keywords or {}, " ")):lower()
            if q == "" or haystack:find(q, 1, true) then
                CloseCommandPalette()
                RunCommand(cmd)
                return
            end
        end
    end)
    backdrop.MouseButton1Click:Connect(CloseCommandPalette)
end

function BeautifulUI:OpenCommandPalette(initialText)
    EnsureCommandPalette()
    CommandGui.Enabled = true
    CommandBox.Text = initialText or ""
    RenderCommandResults()
    task.defer(function()
        if CommandBox and CommandBox.Parent then pcall(function() CommandBox:CaptureFocus() end) end
    end)
end

function BeautifulUI:CloseCommandPalette() CloseCommandPalette() end
function BeautifulUI:IsCommandPaletteOpen() return CommandGui and CommandGui.Enabled or false end

-- Скрыть/показать все окна библиотеки разом

    for _, w in ipairs(OpenWindows) do
        if w.Hide then w:Hide() end
    end
end

function BeautifulUI:ShowAll()
    for _, w in ipairs(OpenWindows) do
        if w.Show then w:Show() end
    end
end

function BeautifulUI:ToggleAll()
    local self_ = self
    if type(self_) ~= "table" then self_ = BeautifulUI end   -- поддержка вызова через точку
    local anyVisible = false
    for _, w in ipairs(OpenWindows) do
        if w.IsVisible and w:IsVisible() then anyVisible = true break end
    end
    if anyVisible then self_:HideAll() else self_:ShowAll() end
end

-- ─── Публичный API кастомного шрифта ───
-- BeautifulUI:SetCustomFont("GothamBlack")
-- BeautifulUI:SetCustomFont(12187365375, {Weight = "Bold", SizeScale = 1.1})
-- BeautifulUI:SetCustomFont({Family = "Code", Bold = false})
function BeautifulUI:SetCustomFont(family, opts)
    opts = type(opts) == "table" and opts or {}
    local spec
    if type(family) == "table" and family.Family ~= nil then
        -- вариант {Family = ..., Weight = ...}
        for k, v in pairs(family) do if k ~= "Family" and opts[k] == nil then opts[k] = v end end
        spec = parseFontSpec(family.Family)
    else
        spec = parseFontSpec(family)
    end
    if not spec then
        if DebugConsole then DebugConsole:Warn("SetCustomFont: неизвестный шрифт", tostring(family)) end
        return false, "unrecognized font: " .. tostring(family)
    end

    FontConfig.Spec = spec
    FontConfig.Family = family
    FontConfig.Weight = opts.Weight or opts.weight
    if type(FontConfig.Weight) == "string" then
        FontConfig.Weight = Enum.FontWeight[FontConfig.Weight] or Enum.FontWeight.Regular
    end
    FontConfig.Style = opts.Style or opts.style
    if type(FontConfig.Style) == "string" then
        FontConfig.Style = Enum.FontStyle[FontConfig.Style] or Enum.FontStyle.Normal
    end
    FontConfig.Bold = opts.Bold
    FontConfig.SizeScale = tonumber(opts.SizeScale or opts.sizeScale) or 1
    FontConfig.LineHeight = tonumber(opts.LineHeight or opts.lineHeight)
    FontConfig.Monospace = opts.Monospace == true
    FontConfig.Applied = true
    FontConfig.Active = true
    FontConfig.FamilyName = spec.Name

    -- применяем ко всем открытым окнам (и к спрятанным тоже — они в OpenWindows)
    local total = 0
    for _, win in ipairs(OpenWindows) do
        total = total + restoreWindowFonts(win)
    end
    FontConfig.AppliedCount = total
    return true, {
        Font = spec.Name,
        Applied = total,
        FaceMode = fontSupportsFace() and "FontFace" or "Enum.Font",
    }
end

function BeautifulUI:ResetFont()
    for obj, rec in pairs(FontRegistry) do
        if obj.Parent then
            if rec.baseSize then obj.TextSize = rec.baseSize end
            pcall(function() obj.FontFace = nil end)
            if rec.original then pcall(function() obj.Font = rec.original end) end
            if FontConfig.LineHeight and obj.LineHeight ~= nil then pcall(function() obj.LineHeight = 1 end) end
        end
    end
    FontConfig.Spec, FontConfig.Family, FontConfig.Applied = nil, nil, nil
    FontConfig.FamilyName, FontConfig.Active = nil, false
    FontConfig.SizeScale, FontConfig.Bold, FontConfig.Weight, FontConfig.Style = 1, nil, nil, nil
    FontConfig.AppliedCount = 0
    return true
end

function BeautifulUI:GetFont()
    return FontConfig.Spec and FontConfig.Spec.Name or nil
end

-- 📋 Все шрифты, которые можно передать в SetCustomFont
function BeautifulUI:GetAvailableFonts()
    local out = {}
    local ok = pcall(function()
        for _, item in ipairs(Enum.Font:GetEnumItems()) do
            table.insert(out, item.Name)
        end
    end)
    if not ok or #out == 0 then
        for _, n in ipairs({ "Gotham", "GothamBold", "GothamMedium", "GothamBlack", "GothamSemibold", "Code", "SourceSans", "SourceSansBold", "Roboto", "RobotoBold", "Legacy", "Arial", "Bodoni", "Cartoon", "Fantasy", "SciFi", "Antique", "Garamond", "Highway", "Bangers", "Creepster", "DenkOne", "Fondamento", "FredokaOne", "GrenzeGotisch", "IndieFlower", "JosefinSans", "Jura", "Kalam", "LuckiestGuy", "Merriweather", "Michroma", "Nunito", "Oswald", "PatrickHand", "PermanentMarker", "PressStart2P", "RobotoMono", "Sarpanch", "SpecialElite", "TitilliumWeb", "Ubuntu" }) do
            table.insert(out, n)
        end
    end
    table.sort(out)
    return out
end

-- 📐 Сколько текстовых объектов сейчас под управлением шрифта
function BeautifulUI:GetFontStats()
    local alive, total = 0, 0
    for obj in pairs(FontRegistry) do
        total = total + 1
        if obj.Parent then alive = alive + 1 end
    end
    return {
        Tracked = total,
        Alive = alive,
        Applied = FontConfig.Spec and FontConfig.Spec.Name or nil,
        SizeScale = FontConfig.SizeScale or 1,
        Mode = fontSupportsFace() and "FontFace" or "Enum.Font",
    }
end

-- 🧪 Живой тест: показать образец текста во всевозможных начертаниях
function BeautifulUI:PreviewFont(font, opts)
    local before = FontConfig.Applied
    if font ~= nil then
        local ok = self:SetCustomFont(font, opts)
        if not ok then return false end
    end
    local win = OpenWindows[1]
    if not win then
        return false, "нужно хотя бы одно открытое окно (или передай window вторым аргументом)"
    end
    local sample = "Съешь ещё этих мягких булок 0123456789 ABC abc — ƒłü"
    self:DialogBox({
        Title = "Тест шрифта: " .. tostring(self:GetFont() or "по умолчанию"),
        Content = sample .. "\n\nРежим: " .. (fontSupportsFace() and "FontFace (любой ассет)" or "Enum.Font (только встроенные)") ..
            "\nОбъектов под шрифтом: " .. tostring((self:GetFontStats()).Alive),
        Buttons = {
            { Text = "Применить", Callback = function() end },
            { Text = "Сбросить", Callback = function() self:ResetFont() end },
        },
    })
    return true, before
end

function BeautifulUI:CheckForUpdates(url, notify)
    if not url or url == "" then return nil end
    local ok, body = pcall(function()
        local res = game:HttpGet(url)
        return tostring(res):gsub("%s+$", "")
    end)
    if not ok or not body or body == "" then
        if notify then self:Notify("Обновление", "Не удалось проверить версию", 3, "Warning") end
        return nil
    end
    local current = tostring(BeautifulUI.Version)
    local isNewer = (body ~= current) and (body > current)
    if notify and isNewer then
        self:Notify("Доступно обновление", string.format("v%s → v%s", current, body), 8, "Info", {
            {Text = "Скачать", Color = Theme.Success, Callback = function()
                pcall(function() setclipboard(url) end)
            end},
        })
    elseif notify then
        self:Notify("Обновление", "У вас актуальная версия v" .. current, 3, "Success")
    end
    return body, isNewer
end
