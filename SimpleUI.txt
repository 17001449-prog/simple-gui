-- ═══════════════════════════════════════════════
local Animations = {}
local AnimationsByInstance = {}

local function RegisterAnimation(instance, prop, updateFn, cleanupFn)
    if AnimationsByInstance[instance] then
        local prev = AnimationsByInstance[instance][prop]
        if prev then
            prev.alive = false
            if prev.cleanup then pcall(prev.cleanup) end
            AnimationsByInstance[instance][prop] = nil
        end
    end

    local entry = {
        instance = instance, prop = prop, update = updateFn, cleanup = cleanupFn,
        alive = true, startTime = os.clock(),
    }
    table.insert(Animations, entry)
    AnimationsByInstance[instance] = AnimationsByInstance[instance] or {}
    AnimationsByInstance[instance][prop] = entry
    return entry
end

local function StopAnimation(instance, prop)
    if not AnimationsByInstance[instance] then return end
    if prop then
        local prev = AnimationsByInstance[instance][prop]
        if prev then
            prev.alive = false
            if prev.cleanup then pcall(prev.cleanup) end
            AnimationsByInstance[instance][prop] = nil
        end
    else
        for _, prev in pairs(AnimationsByInstance[instance]) do
            prev.alive = false
            if prev.cleanup then pcall(prev.cleanup) end
        end
        AnimationsByInstance[instance] = nil
    end
end

RunService.Heartbeat:Connect(function(dt)
    for i = #Animations, 1, -1 do
        local a = Animations[i]
        if not a.alive then
            table.remove(Animations, i)
        elseif not a.instance or not a.instance.Parent then
            if a.cleanup then pcall(a.cleanup) end
            a.alive = false
            if AnimationsByInstance[a.instance] then
                AnimationsByInstance[a.instance][a.prop] = nil
            end
            table.remove(Animations, i)
        else
            local ok = pcall(a.update, dt, os.clock(), a)
            if not ok then
                if a.cleanup then pcall(a.cleanup) end
                a.alive = false
                if AnimationsByInstance[a.instance] then
                    AnimationsByInstance[a.instance][a.prop] = nil
                end
                table.remove(Animations, i)
            end
        end
    end
end)

BeautifulUI.StopAnimation = StopAnimation

-- 🌈 HEX

function BeautifulUI.Hex(hex)
    if type(hex) ~= "string" then return hex end
    hex = hex:gsub("#", ""):gsub("%s", "")
    if #hex ~= 6 then return Color3.new(1, 1, 1) end
    local r = tonumber(hex:sub(1, 2), 16) or 0
    local g = tonumber(hex:sub(3, 4), 16) or 0
    local b = tonumber(hex:sub(5, 6), 16) or 0
    return Color3.fromRGB(r, g, b)
end

function BeautifulUI.ToHex(color)
    return string.format("#%02X%02X%02X",
        math.floor(color.R * 255 + 0.5),
        math.floor(color.G * 255 + 0.5),
        math.floor(color.B * 255 + 0.5))
end

local function C(value, fallback)
    if type(value) == "string" then return BeautifulUI.Hex(value) end
    if typeof(value) == "Color3" then return value end
    return fallback or Color3.new(1, 1, 1)
end

local Theme = {}
local function ApplyTheme(t)
    for k, v in pairs(t) do Theme[k] = C(v) end
end
ApplyTheme(Themes.Purple)

BeautifulUI.Themes = Themes
BeautifulUI.Theme = Theme
BeautifulUI.CurrentThemeName = "Purple"

local OpenWindows = {}
local function ReapplyThemeAllWindows()
    for _, win in ipairs(OpenWindows) do
        if win.Gui and win.Gui.Parent then
            pcall(function() win:_reapplyTheme() end)
        end
    end
end

function BeautifulUI:SetTheme(name)
    if Themes[name] then
        ApplyTheme(Themes[name])
        BeautifulUI.Theme = Theme
        BeautifulUI.CurrentThemeName = name
        ReapplyThemeAllWindows()
        return true
    end
    return false
end

function BeautifulUI:SetColor(key, value)
    if Theme[key] ~= nil then
        Theme[key] = C(value)
        ReapplyThemeAllWindows()
        return true
    end
    return false
end

function BeautifulUI:GetColor(key) return Theme[key] end

function BeautifulUI:GetThemeHex()
    local out = {}
    for k, v in pairs(Theme) do out[k] = BeautifulUI.ToHex(v) end
    return out
end

function BeautifulUI:CreateCustomTheme(name, colorsTable)
    local newTheme = {}
    for k, v in pairs(Themes.Purple) do newTheme[k] = v end
    for k, v in pairs(colorsTable) do newTheme[k] = v end
    Themes[name] = newTheme
    return newTheme
end

function BeautifulUI.AnimatedColor(instance, prop, color1, color2, duration, style)
    color1, color2 = C(color1), C(color2)
    duration = duration or 2
    style = style or Enum.EasingStyle.Sine

    local entry = RegisterAnimation(instance, prop, function(dt, now, a)
        local t = ((now - a.startTime) % duration) / duration
        local wave = (math.sin(t * math.pi * 2 - math.pi / 2) + 1) / 2
        if style == Enum.EasingStyle.Linear then
            wave = t
        elseif style == Enum.EasingStyle.Quad then
            wave = wave * wave
        elseif style == Enum.EasingStyle.Quart then
            wave = wave * wave * wave * wave
        end
        instance[prop] = color1:Lerp(color2, wave)
    end)

    return {Disconnect = function() StopAnimation(instance, prop) end}
end

function BeautifulUI.AnimatedGradient(instance, colors, speed)
    colors = colors or {Theme.Accent, Theme.AccentDark, Theme.Accent}
    speed = speed or 60

    local resolved = {}
    for _, c in ipairs(colors) do table.insert(resolved, C(c)) end

    local grad = instance:FindFirstChildOfClass("UIGradient")
    if not grad then
        grad = Instance.new("UIGradient")
        grad.Parent = instance
    end

    local keypoints = {}
    for i = 1, #resolved do
        local t = (i - 1) / math.max(#resolved - 1, 1)
        table.insert(keypoints, ColorSequenceKeypoint.new(t, resolved[i]))
    end
    grad.Color = ColorSequence.new(keypoints)

    local entry = RegisterAnimation(instance, "_gradient", function(dt, now)
        grad.Rotation = (now * speed) % 360
        local hueShift = (now * 0.05) % 1
        local kp = {}
        for i = 1, #resolved do
            local t = (i - 1) / math.max(#resolved - 1, 1)
            local h, s, v = Color3.toHSV(resolved[i])
            table.insert(kp, ColorSequenceKeypoint.new(t, Color3.fromHSV((h + hueShift) % 1, s, v)))
        end
        grad.Color = ColorSequence.new(kp)
    end)

    return {Disconnect = function() StopAnimation(instance, "_gradient") end}
end

function BeautifulUI.AnimatedImage(imageLabel, frames, fps)
    frames = frames or {}
    fps = fps or 10
    if #frames == 0 then return {Disconnect = function() end} end

    local lastIdx = -1
    local entry = RegisterAnimation(imageLabel, "_image", function(dt, now)
        local idx = math.floor(now * fps) % #frames + 1
        if idx ~= lastIdx then
            imageLabel.Image = frames[idx]
            lastIdx = idx
        end
    end)
    return {Disconnect = function() StopAnimation(imageLabel, "_image") end}
end

function BeautifulUI.Rainbow(instance, prop, speed, saturation, value)
    -- 🎯 ФИКС 1: защита от nil
    if prop == nil then
        prop = "BackgroundColor3"
    end
    -- 🎯 ФИКС 2: защита от Instance (если передали не prop а объект)
    if type(prop) ~= "string" then
        prop = "BackgroundColor3"
    end
    
    speed = speed or 0.1
    saturation = saturation or 1
    value = value or 1

    -- 🎯 ФИКС 3: используем безопасную проверку
    local propName = tostring(prop)
    
    local entry = RegisterAnimation(instance, "_rainbow_" .. propName, function(dt, now)
        local hue = (now * speed) % 1
        instance[prop] = Color3.fromHSV(hue, saturation, value)
    end)

    return {
        Disconnect = function()
            StopAnimation(instance, "_rainbow_" .. propName)
        end
    }
end

-- 💓 Breathe
function BeautifulUI.Breathe(instance, prop, color1, color2, duration, intensity)
    prop = prop or "BackgroundColor3"
    duration = duration or 2
    intensity = intensity or 1
    color1, color2 = C(color1), C(color2)
    local entry = RegisterAnimation(instance, "_breathe_" .. prop, function(dt, now)
        local t = ((now * (1 / duration)) % 1) * math.pi * 2
        local wave = (math.sin(t - math.pi / 2) + 1) / 2
        wave = wave * intensity
        instance[prop] = color1:Lerp(color2, wave)
    end)
    return {Disconnect = function() StopAnimation(instance, "_breathe_" .. prop) end}
end

-- 🎨 CycleColors
function BeautifulUI.CycleColors(instance, prop, colors, duration)
    prop = prop or "BackgroundColor3"
    duration = duration or 3
    local hsv = {}
    for _, c in ipairs(colors) do
        c = C(c)
        local h, s, v = Color3.toHSV(c)
        table.insert(hsv, {h = h, s = s, v = v})
    end
    local n = #hsv
    if n < 2 then return {Disconnect = function() end} end
    local entry = RegisterAnimation(instance, "_cycle_" .. prop, function(dt, now)
        local t = (now / duration) % 1
        local pos = t * n
        local i1 = math.floor(pos) + 1
        local i2 = (i1 % n) + 1
        local frac = pos - math.floor(pos)
        local h1, s1, v1 = hsv[i1].h, hsv[i1].s, hsv[i1].v
        local h2, s2, v2 = hsv[i2].h, hsv[i2].s, hsv[i2].v
        local dh = h2 - h1
        if dh > 0.5 then dh = dh - 1 end
        if dh < -0.5 then dh = dh + 1 end
        local h = (h1 + dh * frac) % 1
        local s = s1 + (s2 - s1) * frac
        local v = v1 + (v2 - v1) * frac
        instance[prop] = Color3.fromHSV(h, s, v)
    end)
    return {Disconnect = function() StopAnimation(instance, "_cycle_" .. prop) end}
end

-- ✨ Sparkle
function BeautifulUI.Sparkle(instance, prop, baseColor, sparkleColor, chance)
    prop = prop or "BackgroundColor3"
    baseColor, sparkleColor = C(baseColor), C(sparkleColor)
    chance = chance or 0.05
    local current = baseColor
    local entry = RegisterAnimation(instance, "_sparkle_" .. prop, function(dt, now)
        if math.random() < chance then
            current = sparkleColor
        else
            current = current:Lerp(baseColor, dt * 3)
        end
        instance[prop] = current
    end)
    return {Disconnect = function() StopAnimation(instance, "_sparkle_" .. prop) end}
end

-- 🌊 WaveColor
function BeautifulUI.WaveColor(instance, colors, speed, rotation)
    colors = colors or {Theme.Accent, Theme.AccentDark, Theme.Accent}
    speed = speed or 30
    local resolved = {}
    for _, c in ipairs(colors) do table.insert(resolved, C(c)) end
    local grad = instance:FindFirstChildOfClass("UIGradient")
    if not grad then
        grad = Instance.new("UIGradient")
        grad.Parent = instance
    end
    grad.Rotation = rotation or 0
    local entry = RegisterAnimation(instance, "_wave_" .. tostring(grad), function(dt, now)
        grad.Offset = Vector2.new((now * speed * 0.01) % 1, 0)
        local kp = {}
        for i = 1, #resolved do
            local t = (i - 1) / math.max(#resolved - 1, 1)
            table.insert(kp, ColorSequenceKeypoint.new(t, resolved[i]))
        end
        grad.Color = ColorSequence.new(kp)
    end)
    return {Disconnect = function() StopAnimation(instance, "_wave_" .. tostring(grad)) end}
end

-- 🎬 CustomAnimatedColor
function BeautifulUI.CustomAnimatedColor(instance, prop, keyframes, duration, loopType)
    prop = prop or "BackgroundColor3"
    duration = duration or 3
    loopType = loopType or "loop"
    local sorted = {}
    for _, kf in ipairs(keyframes) do
        table.insert(sorted, {time = kf.time, color = C(kf.color)})
    end
    table.sort(sorted, function(a, b) return a.time < b.time end)
    local entry = RegisterAnimation(instance, "_custom_" .. prop, function(dt, now)
        local cycleTime = now / duration
        local t
        if loopType == "pingpong" then
            local cycle = cycleTime % 2
            t = cycle <= 1 and cycle or (2 - cycle)
        elseif loopType == "once" then
            t = math.min(cycleTime, 1)
        else
            t = cycleTime % 1
        end
        local c1, c2, segT
        for i = 1, #sorted - 1 do
            if t >= sorted[i].time and t <= sorted[i + 1].time then
                c1 = sorted[i].color
                c2 = sorted[i + 1].color
                segT = (t - sorted[i].time) / math.max(sorted[i + 1].time - sorted[i].time, 0.001)
                break
            end
        end
        if not c1 then
            c1 = sorted[#sorted].color
            c2 = c1
            segT = 0
        end
        instance[prop] = c1:Lerp(c2, segT)
    end)
    return {Disconnect = function() StopAnimation(instance, "_custom_" .. prop) end}
end

local Placeholders = {}

Placeholders["%fps"] = function()
    if _G.FPSData and _G.FPSData.fps then return tostring(_G.FPSData.fps) end
    return "60"
end
Placeholders["%fps_min"] = function()
    if _G.FPSData and _G.FPSData.min and _G.FPSData.min < 9999 then return tostring(_G.FPSData.min) end
    return "60"
end
Placeholders["%fps_max"] = function()
    if _G.FPSData and _G.FPSData.max then return tostring(_G.FPSData.max) end
    return "60"
end
Placeholders["%ping"] = function()
    local ok, ping = pcall(function()
        return math.floor(game:GetService("Stats").Network.ServerStatsItem["Data Ping"]:GetValue())
    end)
    return ok and tostring(ping) or "0"
end
Placeholders["%hp"] = function()
    local char = game.Players.LocalPlayer.Character
    if char then
        local hum = char:FindFirstChildOfClass("Humanoid")
        if hum then return tostring(math.floor(hum.Health)) end
    end
    return "0"
end
Placeholders["%maxhp"] = function()
    local char = game.Players.LocalPlayer.Character
    if char then
        local hum = char:FindFirstChildOfClass("Humanoid")
        if hum then return tostring(math.floor(hum.MaxHealth)) end
    end
    return "100"
end
Placeholders["%hp_percent"] = function()
    local char = game.Players.LocalPlayer.Character
    if char then
        local hum = char:FindFirstChildOfClass("Humanoid")
        if hum and hum.MaxHealth > 0 then
            return string.format("%.1f", (hum.Health / hum.MaxHealth) * 100)
        end
    end
    return "0.0"
end
Placeholders["%time"] = function() return os.date("%H:%M:%S") end
Placeholders["%date"] = function() return os.date("%d.%m.%Y") end
Placeholders["%players"] = function() return tostring(#game.Players:GetPlayers()) end
Placeholders["%name"] = function() return game.Players.LocalPlayer.Name end
Placeholders["%displayname"] = function() return game.Players.LocalPlayer.DisplayName end
Placeholders["%userid"] = function() return tostring(game.Players.LocalPlayer.UserId) end
Placeholders["%memory"] = function()
    local ok, mem = pcall(function()
        return math.floor(game:GetService("Stats"):GetTotalMemoryUsageMb())
    end)
    return ok and tostring(mem) or "0"
end
Placeholders["%placeid"] = function() return tostring(game.PlaceId) end
Placeholders["%jobid"] = function() return game.JobId:sub(1, 8) .. "..." end
Placeholders["%uptime"] = function() return tostring(math.floor(tick() - (_G.LibraryLoadTime or tick()))) end

function BeautifulUI.SubstitutePlaceholders(text)
    if type(text) ~= "string" then return text end
    if not text:find("%%") then return text end
    for key, fn in pairs(Placeholders) do
        if text:find(key, 1, true) then
            local ok, val = pcall(fn)
            if ok then
                text = text:gsub(key:gsub("%%", "%%%%"), tostring(val))
            end
        end
    end
    return text
end

function BeautifulUI:Placeholder(instance, initialText, updateInterval)
    if not instance then return {Disconnect = function() end} end
    updateInterval = updateInterval or 0.5
    local template = initialText or instance.Text or ""
    local running = true
    task.spawn(function()
        while running and instance and instance.Parent do
            local newText = BeautifulUI.SubstitutePlaceholders(template)
            if instance.Text ~= newText then
                instance.Text = newText
            end
            task.wait(updateInterval)
        end
    end)
    return {
        SetTemplate = function(t) template = t end,
        Disconnect = function() running = false end
    }
end

function BeautifulUI:RegisterPlaceholder(name, fn)
    if type(name) == "string" and not name:find("^%%") then
        name = "%" .. name
    end
    Placeholders[name] = fn
end

function BeautifulUI:GetPlaceholders()
    local list = {}
    for k in pairs(Placeholders) do table.insert(list, k) end
    table.sort(list)
    return list
end

function BeautifulUI:ApplyGradientPreset(instance, presetName, rotation)
    local preset = BeautifulUI.GradientPresets[presetName]
    if not preset then warn("[BeautifulUI] Пресет не найден:", presetName); return nil end
    local old = instance:FindFirstChildOfClass("UIGradient")
    if old then old:Destroy() end
    local keypoints = {}
    for i, hex in ipairs(preset) do
        local t = (i - 1) / math.max(#preset - 1, 1)
        table.insert(keypoints, ColorSequenceKeypoint.new(t, BeautifulUI.Hex(hex)))
    end
    local grad = Instance.new("UIGradient")

local function Create(className, props)
    local inst = Instance.new(className)
    for k, v in pairs(props or {}) do inst[k] = v end
    return inst
end

local function Corner(radius, parent)
    return Create("UICorner", {CornerRadius = UDim.new(0, radius or 8), Parent = parent})
end

local function Stroke(color, thickness, transparency, parent)
    return Create("UIStroke", {Color = C(color, Theme.Accent), Thickness = thickness or 1, Transparency = transparency or 0.5, Parent = parent})
end

local function Tween(obj, time, props)
    local tw = TweenService:Create(obj, TweenInfo.new(time or 0.2, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), props)
    tw:Play()
    return tw
end

local function Gradient(c1, c2, rot, parent)
    return Create("UIGradient", {Color = ColorSequence.new(C(c1), C(c2)), Rotation = rot or 90, Parent = parent})
end

local function MakeDraggable(frame, dragArea)
    dragArea = dragArea or frame
    local dragging, dragStart, startPos
    local connections = {}
    table.insert(connections, dragArea.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true; dragStart = input.Position; startPos = frame.Position
            local c
            c = input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then
                    dragging = false
                    if c then c:Disconnect() end
                end
            end)
        end
    end))
    table.insert(connections, UserInputService.InputChanged:Connect(function(input)
        if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
            local delta = input.Position - dragStart
            frame.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
        end
    end))
    return function()
        for _, c in ipairs(connections) do pcall(function() c:Disconnect() end) end
    end
end

local ConfigFlags = {}
local ConfigPath = "BeautifulUI_Config.json"

function BeautifulUI:SetConfigPath(p) ConfigPath = p end

function BeautifulUI:SaveConfig()
    local data = {}
    for k, v in pairs(ConfigFlags) do
        if v.Get then data[k] = v:Get() end
    end
    local ok, encoded = pcall(HttpService.JSONEncode, HttpService, data)
    if ok then
        pcall(function() if writefile then writefile(ConfigPath, encoded) end end)
        BeautifulUI:Notify("Конфиг", "Сохранено ✅", 2, "Success")
    end
end

function BeautifulUI:LoadConfig()
    local ok, content = pcall(function()
        if readfile and isfile and isfile(ConfigPath) then return readfile(ConfigPath) end
    end)
    if not ok or not content then
        BeautifulUI:Notify("Конфиг", "Файл не найден", 2, "Warning"); return
    end
    local ok2, data = pcall(HttpService.JSONDecode, HttpService, content)
    if not ok2 then return end
    for k, v in pairs(data) do
        if ConfigFlags[k] and ConfigFlags[k].Set then pcall(function() ConfigFlags[k]:Set(v) end) end
    end
    BeautifulUI:Notify("Конфиг", "Загружено ✅", 2, "Success")
end

function BeautifulUI:DeleteConfig()
    pcall(function() if delfile and isfile and isfile(ConfigPath) then delfile(ConfigPath) end end)
end

function DebugConsole:Log(level, ...)
    local args = {...}
    local parts = {}
    for _, v in ipairs(args) do table.insert(parts, tostring(v)) end
    local msg = table.concat(parts, " ")
    table.insert(self.Logs, {level = level, msg = msg, time = os.date("%H:%M:%S")})
    print("[BeautifulUI " .. level .. "] " .. msg)
    if self.Visible and self.Gui then self:Refresh() end
end
function DebugConsole:Info(...) self:Log("INFO", ...) end
function DebugConsole:Warn(...) self:Log("WARN", ...) end
function DebugConsole:Error(...) self:Log("ERROR", ...) end

function DebugConsole:Refresh()
    if not self.Gui then return end
    local list = self.Gui:FindFirstChild("List")
    if not list then return end
    for _, c in ipairs(list:GetChildren()) do if c:IsA("TextLabel") then c:Destroy() end end
    for i, log in ipairs(self.Logs) do
        local colors = {INFO = Theme.Info, WARN = Theme.Warning, ERROR = Theme.Danger}
        Create("TextLabel", {Size = UDim2.new(1, -10, 0, 18), BackgroundTransparency = 1, Font = Enum.Font.Code, Text = "[" .. log.time .. "] [" .. log.level .. "] " .. log.msg, TextColor3 = colors[log.level] or Theme.Text, TextSize = 12, TextXAlignment = Enum.TextXAlignment.Left, TextWrapped = true, LayoutOrder = i, Parent = list})
    end
    task.wait()
    local layout = list:FindFirstChildOfClass("UIListLayout")

        list.CanvasSize = UDim2.new(0, 0, 0, layout.AbsoluteContentSize.Y + 10)
        list.CanvasPosition = Vector2.new(0, list.CanvasSize.Y.Offset)
    end
end

            table.insert(themeElements, {el = el, update = updateFn})

            for _, t in pairs(Window.Tabs) do
                t.Page.Visible = false

            end
            page.Visible = true

            if Window.ActiveTab ~= tab then Tween(tabButton, 0.15, {BackgroundColor3 = Theme.Tertiary, TextColor3 = Theme.Text}) end

        function TabAPI:_mkButton(parent, text, callback, style, icon)

            elseif style == "Danger" then hoverColor = Theme.Danger
            elseif style == "Warning" then hoverColor = Theme.Warning

            btn.MouseLeave:Connect(function() Tween(btn, 0.15, {BackgroundColor3 = bgColor}) end)
            btn.MouseButton1Click:Connect(function()
                Tween(btn, 0.08, {BackgroundColor3 = Theme.AccentDark})

                btn.BackgroundColor3 = bgColor; btn.TextColor3 = Theme.Text
                btnStroke.Color = Theme.Accent
                if grad then grad.Color = ColorSequence.new(Theme.Accent, Theme.AccentDark) end

        function TabAPI:_mkSlider(parent, text, min, max, default, callback, flag)
            min, max = min or 0, max or 100
            if max == min then max = min + 1 end
            local value = math.clamp(default or min, min, max)
            local updating = false

            local frame = Create("Frame", {Size = UDim2.new(1, 0, 0, 50), BackgroundColor3 = Theme.Secondary, BorderSizePixel = 0, Parent = parent})
            Corner(8, frame)
            local lbl = Create("TextLabel", {Size = UDim2.new(0.7, 0, 0, 20), Position = UDim2.new(0, 12, 0, 6), BackgroundTransparency = 1, Font = Enum.Font.GothamMedium, Text = text, TextColor3 = Theme.Text, TextSize = 13, TextXAlignment = Enum.TextXAlignment.Left, Parent = frame})
            local valueLabel = Create("TextLabel", {Size = UDim2.new(0.3, -12, 0, 20), Position = UDim2.new(0.7, 0, 0, 6), BackgroundTransparency = 1, Font = Enum.Font.GothamBold, Text = tostring(value), TextColor3 = Theme.Accent, TextSize = 13, TextXAlignment = Enum.TextXAlignment.Right, Parent = frame})
            local barBg = Create("Frame", {Size = UDim2.new(1, -24, 0, 6), Position = UDim2.new(0, 12, 0, 34), BackgroundColor3 = Theme.Tertiary, BorderSizePixel = 0, Parent = frame})
            Corner(3, barBg)
            local fill = Create("Frame", {Size = UDim2.new((value - min) / (max - min), 0, 1, 0), BackgroundColor3 = Theme.Accent, BorderSizePixel = 0, Parent = barBg})
            Corner(3, fill)

            local dragging = false
            local function setVal(x, fire)
                local rel = math.clamp((x - barBg.AbsolutePosition.X) / barBg.AbsoluteSize.X, 0, 1)
                local newVal = math.floor(min + (max - min) * rel + 0.5)
                if newVal == value and fire then return end
                value = newVal
                fill.Size = UDim2.new(rel, 0, 1, 0)
                valueLabel.Text = tostring(value)
                if fire and not updating and callback then pcall(callback, value) end
            end
            barBg.InputBegan:Connect(function(i) if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then dragging = true; setVal(i.Position.X, true) end end)
            UserInputService.InputChanged:Connect(function(i) if dragging and (i.UserInputType == Enum.UserInputType.MouseMovement or i.UserInputType == Enum.UserInputType.Touch) then setVal(i.Position.X, true) end end)
            UserInputService.InputEnded:Connect(function(i) if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then dragging = false end end)

            local api = {Set = function(a, b)
    local v = (type(a) == "table" and a == api) and b or a
    v = math.clamp(tonumber(v) or 0, min, max)
    if v == value then return end
    updating = true
    value = v
    local rel = (value - min) / (max - min)
    fill.Size = UDim2.new(rel, 0, 1, 0)
    valueLabel.Text = tostring(value)
    updating = false
end, Get = function() return value end}
            if flag then ConfigFlags[flag] = api end
            registerThemeElement(frame, function()
                frame.BackgroundColor3 = Theme.Secondary; lbl.TextColor3 = Theme.Text
                valueLabel.TextColor3 = Theme.Accent; barBg.BackgroundColor3 = Theme.Tertiary; fill.BackgroundColor3 = Theme.Accent
            end)
            return api
        end

        -- ─── INPUT SLIDER (слайдер с полем ввода) ───
        function TabAPI:InputSlider(text, min, max, default, callback, flag)
            min, max = min or 0, max or 100
            if max == min then max = min + 1 end
            local value = math.clamp(default or min, min, max)
            local updating = false

            local frame = Create("Frame", {Size = UDim2.new(1, 0, 0, 50), BackgroundColor3 = Theme.Secondary, BorderSizePixel = 0, Parent = page})
            Corner(8, frame)
            local lbl = Create("TextLabel", {Size = UDim2.new(0.5, 0, 0, 20), Position = UDim2.new(0, 12, 0, 6), BackgroundTransparency = 1, Font = Enum.Font.GothamMedium, Text = text, TextColor3 = Theme.Text, TextSize = 13, TextXAlignment = Enum.TextXAlignment.Left, Parent = frame})
            local valueBox = Create("TextBox", {Size = UDim2.new(0, 80, 0, 22), Position = UDim2.new(1, -92, 0, 5), BackgroundColor3 = Theme.Tertiary, BorderSizePixel = 0, Font = Enum.Font.GothamBold, Text = tostring(value), TextColor3 = Theme.Accent, TextSize = 12, ClearTextOnFocus = false, Parent = frame})
            Corner(4, valueBox)
            local barBg = Create("Frame", {Size = UDim2.new(1, -24, 0, 6), Position = UDim2.new(0, 12, 0, 34), BackgroundColor3 = Theme.Tertiary, BorderSizePixel = 0, Parent = frame})
            Corner(3, barBg)
            local fill = Create("Frame", {Size = UDim2.new((value - min) / (max - min), 0, 1, 0), BackgroundColor3 = Theme.Accent, BorderSizePixel = 0, Parent = barBg})
            Corner(3, fill)

            local function updateUI()
                local rel = (value - min) / (max - min)
                fill.Size = UDim2.new(rel, 0, 1, 0)
                if not valueBox:IsFocused() then valueBox.Text = tostring(value) end
            end

            local function setValue(v, fire)
                v = tonumber(v); if not v then return end
                v = math.clamp(v, min, max)
                if v == value and not fire then return end
                value = v
                updateUI()
                if fire and not updating and callback then pcall(callback, value) end
            end

            local dragging = false
            local function setFromX(x)
                local rel = math.clamp((x - barBg.AbsolutePosition.X) / barBg.AbsoluteSize.X, 0, 1)
                setValue(math.floor(min + (max - min) * rel + 0.5), true)
            end
            barBg.InputBegan:Connect(function(i) if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then dragging = true; setFromX(i.Position.X) end end)
            UserInputService.InputChanged:Connect(function(i) if dragging and (i.UserInputType == Enum.UserInputType.MouseMovement or i.UserInputType == Enum.UserInputType.Touch) then setFromX(i.Position.X) end end)
            UserInputService.InputEnded:Connect(function(i) if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then dragging = false end end)
            valueBox.FocusLost:Connect(function() local p = tonumber(valueBox.Text); if p then setValue(p, true) else updateUI() end end)

            local api = {Set = function(a, b)
    local v = (type(a) == "table" and a == api) and b or a
    updating = true
    setValue(v, false)
    updating = false
end, Get = function() return value end}
            if flag then ConfigFlags[flag] = api end
            registerThemeElement(frame, function()
                frame.BackgroundColor3 = Theme.Secondary; lbl.TextColor3 = Theme.Text
                valueBox.BackgroundColor3 = Theme.Tertiary; valueBox.TextColor3 = Theme.Accent
                barBg.BackgroundColor3 = Theme.Tertiary; fill.BackgroundColor3 = Theme.Accent
            end)
            return api
        end

                elseif not listening and input.KeyCode == currentKey then
                    if callback then pcall(callback) end
                end
            end)
            local api = {Set = function(k) if typeof(k) == "EnumItem" then currentKey = k elseif type(k) == "string" then currentKey = Enum.KeyCode[k] or currentKey end; keyBtn.Text = currentKey.Name end, Get = function() return currentKey.Name end}

                if not listening then keyBtn.TextColor3 = Theme.Accent end
            end)

        end

function TabAPI:Graph(text, graphType, height)
    -- Нормализация типов
    local function normalizeGraphType(t)
        t = tostring(t or "line"):lower():gsub("%s+", "")
        local map = {
            line = "line", lines = "line", polyline = "line",
            area = "area", fill = "area", histogram = "bar",
            bar = "bar", bars = "bar", column = "bar",
            scatter = "scatter", point = "scatter",
            pie = "pie", doughnut = "pie", donut = "pie",
            radial = "radial", radar = "radial", spider = "radial",
            stacked = "stacked", stack = "stacked",
        }
        return map[t] or t
    end

    if typeof(height) == "UDim2" then height = height.Offset end
    height = math.max(tonumber(height) or 160, 60)
    graphType = normalizeGraphType(graphType)

    -- Создание контейнера
    local frame = Create("Frame", {
        Size = UDim2.new(1, 0, 0, height + 40),
        BackgroundColor3 = Theme.Secondary,
        BorderSizePixel = 0,
        Parent = page
    })
    Corner(8, frame)
    local st = Stroke(Theme.Accent, 1, 0.7, frame)

    local titleLbl = Create("TextLabel", {
        Size = UDim2.new(1, -20, 0, 20),
        Position = UDim2.new(0, 12, 0, 6),
        BackgroundTransparency = 1,
        Font = Enum.Font.GothamBold,
        Text = text or "Graph",
        TextColor3 = Theme.Text,
        TextSize = 14,
        TextXAlignment = Enum.TextXAlignment.Left,
        Parent = frame
    })

    local canvas = Create("Frame", {
        Name = "GraphCanvas",
        Size = UDim2.new(1, -24, 0, height),
        Position = UDim2.new(0, 12, 0, 32),
        BackgroundColor3 = Theme.Tertiary,
        BorderSizePixel = 0,
        ClipsDescendants = true,
        ZIndex = 1,
        Parent = frame
    })
    Corner(6, canvas)

    -- Сетка (Grid Lines)
    local gridLines = {}
    for i = 1, 4 do
        local gl = Create("Frame", {
            Size = UDim2.new(1, 0, 0, 1),
            Position = UDim2.new(0, 0, i / 5, 0),
            BackgroundColor3 = Theme.Secondary,
            BackgroundTransparency = 0.5,
            BorderSizePixel = 0,
            ZIndex = 2,

    end

    -- Контейнер для данных
    local container = Create("Frame", {
        Name = "GraphContainer",
        Size = UDim2.new(1, 0, 1, 0),
        BackgroundTransparency = 1,
        ClipsDescendants = false,
        ZIndex = 5,
        Parent = canvas
    })

    -- Данные и настройки
    local data = {}
    local series = { {name = "Series 1", data = data} } 
    local minValue = 0
    local maxValue = 100
    local autoRange = false
    local maxPoints = 50
    
    local palette = {
        Color3.fromRGB(0, 255, 255), -- Cyan
        Color3.fromRGB(255, 200, 50), -- Yellow
        Color3.fromRGB(255, 100, 100), -- Red/Pink
        Color3.fromRGB(150, 100, 255), -- Purple
        Color3.fromRGB(100, 255, 100), -- Green
        Color3.fromRGB(255, 150, 50), -- Orange
        Color3.fromRGB(100, 150, 255), -- Blue
    }

    local needsRedraw = false
    local api = {}

    -- Вспомогательные функции
    local function recalcAutoRange()
        if not autoRange then return end
        local mn, mx
        for _, s in ipairs(series) do
            for _, v in ipairs(s.data) do
                v = tonumber(v)
                if v then
                    if not mn or v < mn then mn = v end
                    if not mx or v > mx then mx = v end
                end
            end
        end
        if mn then
            minValue = mn
            maxValue = mx
            if maxValue == minValue then maxValue = minValue + 1 end
        end
    end

    local function norm(v)
        v = tonumber(v) or 0
        if maxValue == minValue then return 0.5 end
        return math.clamp((v - minValue) / (maxValue - minValue), 0, 1)
    end

    local function clearContainer()
        for _, c in ipairs(container:GetChildren()) do
            c:Destroy()
        end
    end

    -- 🖱️ СИСТЕМА TOOLTIP (Подсказки)
    -- ВАЖНО: Здесь исправлен синтаксис Instance.new
    local tooltipGui = Instance.new("ScreenGui")
    tooltipGui.ResetOnSpawn = false
    tooltipGui.ZIndexBehavior = Enum.ZIndexBehavior.Sibling
    tooltipGui.Parent = (gethui and gethui()) or CoreGui

    local tooltipFrame = Instance.new("Frame")

    tooltipFrame.BackgroundColor3 = Theme.Background
    tooltipFrame.BorderSizePixel = 0
    tooltipFrame.Visible = false
    tooltipFrame.Active = true

    tooltipFrame.Parent = tooltipGui
    
    -- Используем наши хелперы вместо прямых вызовов с таблицами

    -- Функция создания интерактивной точки с тултипом
    local function createInteractiveDot(x, y, size, color, seriesName, index, value)
        local dot = Instance.new("Frame")

    -- 🔥 ГЛАВНОЕ ИСПРАВЛЕНИЕ: РИСОВАНИЕ ЛИНИЙ БЕЗ ROTATION
    local function drawStableLine(x1, y1, x2, y2, color, thickness, zIndex)
        local dx = x2 - x1
        local dy = y2 - y1
        local dist = math.sqrt(dx*dx + dy*dy)
        
        if dist < 1 then return end
        
        local steps = math.min(math.floor(dist / 2), 50) 
        if steps < 1 then steps = 1 end
        
        local stepX = dx / steps
        local stepY = dy / steps
        
        for i = 0, steps do
            local px = x1 + stepX * i
            local py = y1 + stepY * i
            
            local seg = Instance.new("Frame")
            seg.Size = UDim2.new(0, thickness, 0, thickness)
            seg.Position = UDim2.new(0, px - thickness/2, 0, py - thickness/2)
            seg.BackgroundColor3 = color
            seg.BorderSizePixel = 0
            seg.ZIndex = zIndex or 10
            seg.Parent = container
            
            Corner(thickness/2, seg) -- Делаем их круглыми
        end
    end

    local function redraw()
        clearContainer()
        if #series == 0 then return end

        local w = canvas.AbsoluteSize.X
        local h = canvas.AbsoluteSize.Y
        
        if w <= 0 or h <= 0 then return end
        if w < 20 or h < 20 then return end

        local pad = 12
        local usableW = w - pad * 2
        local usableH = h - pad * 2
        if usableW < 10 or usableH < 10 then return end

        -- Логика для разных типов
        if graphType == "line" or graphType == "area" or graphType == "stacked" then
            
            -- Рисуем области (для Area) сначала
            if graphType == "area" then
                for si, s in ipairs(series) do
                    local color = palette[((si - 1) % #palette) + 1]
                    local d = s.data
                    if #d >= 2 then
                        for i = 1, #d - 1 do
                            local x = pad + (i - 1) / (#d - 1) * usableW
                            local y1 = pad + usableH - norm(d[i]) * usableH
                            local y2 = pad + usableH - norm(d[i+1]) * usableH
                            
                            local rect = Instance.new("Frame")
                            rect.Size = UDim2.new(0, usableW/(#d-1)+1, 0, math.abs(y2-y1))
                            rect.Position = UDim2.new(0, x, 0, math.min(y1,y2))
                            rect.BackgroundColor3 = color
                            rect.BackgroundTransparency = 0.6
                            rect.BorderSizePixel = 0
                            rect.ZIndex = 8
                            rect.Parent = container
                        end
                    end
                end
            end

            -- Рисуем линии
            for si, s in ipairs(series) do
                local color = palette[((si - 1) % #palette) + 1]
                local d = s.data
                
                if #d >= 2 then
                    for i = 1, #d - 1 do
                        local x1 = pad + (i - 1) / (#d - 1) * usableW
                        local x2 = pad + i / (#d - 1) * usableW
                        local y1 = pad + usableH - norm(d[i]) * usableH
                        local y2 = pad + usableH - norm(d[i+1]) * usableH
                        
                        drawStableLine(x1, y1, x2, y2, color, 3, 10)
                    end
                    
                    -- Точки данных с тултипами
                    if graphType ~= "stacked" then
                         for i, v in ipairs(d) do
                            local x = pad + (i - 1) / math.max(#d - 1, 1) * usableW
                            local y = pad + usableH - norm(v) * usableH
                            createInteractiveDot(x, y, 6, color, s.name, i, v)
                        end
                    end
                elseif #d == 1 then
                     local x = pad + usableW / 2
                     local y = pad + usableH - norm(d[1]) * usableH
                     createInteractiveDot(x, y, 6, color, s.name, 1, d[1])
                end
            end

        elseif graphType == "bar" then
            local sData = series[1].data
            local n = #sData
            if n == 0 then return end
            
            local gap = 3
            local barW = math.max((usableW - gap * (n - 1)) / n, 3)
            
            for i, v in ipairs(sData) do
                local x = pad + (i - 1) * (barW + gap)
                local barH = norm(v) * usableH
                local color = palette[((i - 1) % #palette) + 1]
                
                local bar = Instance.new("Frame")
                bar.Size = UDim2.new(0, barW, 0, barH)
                bar.Position = UDim2.new(0, x, 1, -pad - barH)
                bar.BackgroundColor3 = color
                bar.BorderSizePixel = 0
                bar.ZIndex = 10
                bar.Parent = container
                
                Corner(3, bar)

                -- Тултип для бара
                local hitbox = Instance.new("TextButton")
                hitbox.Size = UDim2.new(0, barW, 0, barH)
                hitbox.Position = UDim2.new(0, x, 1, -pad - barH)
                hitbox.BackgroundTransparency = 1
                hitbox.Text = ""
                hitbox.ZIndex = 11
                hitbox.Parent = container
                
                hitbox.MouseEnter:Connect(function()
                    tooltipTitle.Text = "Item #" .. i
                    tooltipValue.Text = string.format("%.2f", v)
                    ttIndicator.BackgroundColor3 = color
                    tooltipFrame.Visible = true
                end)
                hitbox.MouseLeave:Connect(function()
                    tooltipFrame.Visible = false
                end)
            end

        elseif graphType == "scatter" then
             for si, s in ipairs(series) do
                local color = palette[((si - 1) % #palette) + 1]
                local d = s.data
                for i, v in ipairs(d) do
                    local x = pad + (i - 1) / math.max(#d - 1, 1) * usableW
                    local y = pad + usableH - norm(v) * usableH
                    createInteractiveDot(x, y, 8, color, s.name, i, v)
                end
            end

        elseif graphType == "pie" then
            local sData = series[1].data
            local total = 0
            for _, v in ipairs(sData) do total = total + math.max(tonumber(v) or 0, 0) end
            if total <= 0 then return end

            local size = math.min(w, h) - 24
            local cx, cy = w / 2, h / 2
            local radius = size / 2
            local startAngle = -90

            for i, v in ipairs(sData) do
                v = math.max(tonumber(v) or 0, 0)
                if v > 0 then
                    local sweep = (v / total) * 360
                    local steps = math.max(math.ceil(sweep / 4), 4)
                    local color = palette[((i - 1) % #palette) + 1]

                    for s = 0, steps - 1 do
                        local a1 = startAngle + (s / steps) * sweep
                        local a2 = startAngle + ((s + 1) / steps) * sweep
                        local midA = (a1 + a2) / 2
                        local rad = math.rad(midA)
                        
                        local wx = cx + math.cos(rad) * radius / 2
                        local wy = cy + math.sin(rad) * radius / 2

                        local wedge = Instance.new("Frame")
                        wedge.Size = UDim2.new(0, radius, 0, math.max((sweep/steps)*3, 6))
                        wedge.Position = UDim2.new(0, wx - radius/2, 0, wy - 3)
                        wedge.AnchorPoint = Vector2.new(0, 0.5)
                        wedge.BackgroundColor3 = color
                        wedge.BorderSizePixel = 0
                        wedge.Rotation = midA
                        wedge.ZIndex = 10
                        wedge.Parent = container
                    end
                end
                startAngle = startAngle + sweep
            end
            
            local inner = Instance.new("Frame")
            inner.Size = UDim2.new(0, radius, 0, radius)
            inner.Position = UDim2.new(0.5, -radius/2, 0.5, -radius/2)
            inner.BackgroundColor3 = Theme.Tertiary
            inner.BorderSizePixel = 0
            inner.ZIndex = 15
            inner.Parent = container
            Corner(radius/2, inner)

        elseif graphType == "radial" then
            local sData = series[1].data
            local n = #sData
            if n == 0 then return end
            
            local cx, cy = w / 2, h / 2
            local maxR = math.min(cx, cy) - 12
            if maxR < 5 then return end
            
            local angleStep = 360 / n
            
            -- Оси
            for i = 1, n do
                local spokeLen = maxR
                local ang = math.rad((i - 1) * angleStep - 90)
                local ex = cx + math.cos(ang) * spokeLen
                local ey = cy + math.sin(ang) * spokeLen
                drawStableLine(cx, cy, ex, ey, Theme.SubText, 1, 5)
            end
            
            -- Полигон данных
            local points = {}
            for i, v in ipairs(sData) do
                local ang = math.rad((i - 1) * angleStep - 90)
                local r = norm(v) * maxR
                points[i] = {x = cx + math.cos(ang) * r, y = cy + math.sin(ang) * r, val = v, idx = i}
            end
            
            for i = 1, n do
                local j = (i % n) + 1
                drawStableLine(points[i].x, points[i].y, points[j].x, points[j].y, Theme.Accent, 2, 10)
            end
            
            for _, p in ipairs(points) do
                createInteractiveDot(p.x, p.y, 7, Theme.Accent, "Axis", p.idx, p.val)
            end
        end
    end

    -- Управление перерисовкой
    local function requestRedraw()
        needsRedraw = true
    end

    local renderConn
    renderConn = RunService.Heartbeat:Connect(function()
        if not frame.Parent then
            if renderConn then renderConn:Disconnect() end
            return
        end
        if needsRedraw then
            if canvas.AbsoluteSize.X >= 20 then
                needsRedraw = false
                pcall(redraw)
            end
        end
    end)

    canvas:GetPropertyChangedSignal("AbsoluteSize"):Connect(function()
        if canvas.AbsoluteSize.X >= 20 then requestRedraw() end
    end)

        frame.BackgroundColor3 = Theme.Secondary
        st.Color = Theme.Accent
        titleLbl.TextColor3 = Theme.Text
        canvas.BackgroundColor3 = Theme.Tertiary
        for _, gl in ipairs(gridLines) do gl.BackgroundColor3 = Theme.Secondary end
        requestRedraw()
    end)

    -- API Методы
    api.AddPoint = function(first, second)
        local v = first == api and second or first
        v = tonumber(v) or 0
        table.insert(data, v)
        if #data > maxPoints then table.remove(data, 1) end
        if autoRange then recalcAutoRange() end
        requestRedraw()
        return api
    end

    api.SetData = function(first, second)
        local arr = first == api and second or first
        data = {}
        if type(arr) == "table" then
            for _, v in ipairs(arr) do table.insert(data, tonumber(v) or 0) end
        end
        series[1].data = data
        if autoRange then recalcAutoRange() end
        requestRedraw()
        return api
    end

    api.AddSeries = function(nameOrArr, maybeArr)
        local name, arr
        if type(nameOrArr) == "string" then
            name = nameOrArr
            arr = maybeArr
        else
            name = "Series " .. (#series + 1)
            arr = nameOrArr
        end
        
        local ns = {}
        if type(arr) == "table" then
            for _, v in ipairs(arr) do table.insert(ns, tonumber(v) or 0) end
        end
        
        table.insert(series, {name = name, data = ns})
        if autoRange then recalcAutoRange() end
        requestRedraw()
        return api
    end

    api.SetRange = function(mn, mx)
        minValue = tonumber(mn) or 0
        maxValue = tonumber(mx) or 100
        if maxValue == minValue then maxValue = minValue + 1 end
        autoRange = false
        requestRedraw()
        return api
    end

    api.SetAutoRange = function(enabled)
        autoRange = enabled ~= false
        recalcAutoRange()
        requestRedraw()
        return api
    end
    
    api.Destroy = function()
        if renderConn then renderConn:Disconnect() end
        if tooltipGui then tooltipGui:Destroy() end
        clearContainer()
        if frame.Parent then frame:Destroy() end
    end

    task.defer(requestRedraw)
    return api
end
        return TabAPI 
    end

    function Window:Notify(t, txt, dur, tp, btns) BeautifulUI:Notify(t, txt, dur, tp, btns) end
    function Window:DialogBox(cfg) return BeautifulUI:DialogBox(cfg) end

    local visible = true
    UserInputService.InputBegan:Connect(function(input, gpe)
        if gpe then return end
        if input.KeyCode == toggleKey then
            visible = not visible
            if not visible then
                Tween(main, 0.25, {Size = UDim2.new(0, 0, 0, 0), Position = UDim2.new(0.5, 0, 0.5, 0)})
                task.wait(0.26); main.Visible = false; ShowIcon()
            else
                HideIcon(); task.wait(0.05); main.Visible = true
                main.Position = UDim2.new(0.5, -size.X.Offset/2, 0.5, -size.Y.Offset/2)
                Tween(main, 0.3, {Size = size})
            end
        end
    end)

    return Window
end

return BeautifulUI
