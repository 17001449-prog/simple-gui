-- Один Heartbeat на всю библиотеку
-- ═══════════════════════════════════════════════
local Animations = {}           -- список активных анимаций
local AnimationsByInstance = {} -- {[instance] = {[prop] = animEntry}} для отмены дублей

local function RegisterAnimation(instance, prop, updateFn, cleanupFn)
    -- Fix 1: убить предыдущую анимацию на этом instance+prop
    if AnimationsByInstance[instance] then
        local prev = AnimationsByInstance[instance][prop]
        if prev then
            prev.alive = false
            if prev.cleanup then pcall(prev.cleanup) end
            AnimationsByInstance[instance][prop] = nil
        end
    end

    local entry = {
        instance = instance,
        prop = prop,
        update = updateFn,
        cleanup = cleanupFn,
        alive = true,
        startTime = os.clock(),  -- Fix 1: локальный сдвиг фазы
    }
    table.insert(Animations, entry)

    AnimationsByInstance[instance] = AnimationsByInstance[instance] or {}
    AnimationsByInstance[instance][prop] = entry

    return entry
end

local function StopAnimation(instance, prop)
    if not AnimationsByInstance[instance] then return end
    if prop then
        local prev = AnimationsByInstance[instance][prop]
        if prev then
            prev.alive = false
            if prev.cleanup then pcall(prev.cleanup) end
            AnimationsByInstance[instance][prop] = nil
        end
    else
        for _, prev in pairs(AnimationsByInstance[instance]) do
            prev.alive = false
            if prev.cleanup then pcall(prev.cleanup) end
        end
        AnimationsByInstance[instance] = nil
    end
end

-- 🔧 Единственный Heartbeat listener
RunService.Heartbeat:Connect(function(dt)
    for i = #Animations, 1, -1 do
        local a = Animations[i]
        if not a.alive then
            table.remove(Animations, i)
        elseif not a.instance or not a.instance.Parent then
            -- Fix 1: cleanup при удалении instance
            if a.cleanup then pcall(a.cleanup) end
            a.alive = false
            if AnimationsByInstance[a.instance] then
                AnimationsByInstance[a.instance][a.prop] = nil
            end
            table.remove(Animations, i)
        else
            local ok = pcall(a.update, dt, os.clock(), a)
            if not ok then
                if a.cleanup then pcall(a.cleanup) end
                a.alive = false
                if AnimationsByInstance[a.instance] then
                    AnimationsByInstance[a.instance][a.prop] = nil
                end
                table.remove(Animations, i)
            end
        end
    end
end)

function BeautifulUI.Hex(hex)
    if type(hex) ~= "string" then return hex end
    hex = hex:gsub("#", ""):gsub("%s", "")
    if #hex ~= 6 then return Color3.new(1, 1, 1) end
    local r = tonumber(hex:sub(1, 2), 16) or 0
    local g = tonumber(hex:sub(3, 4), 16) or 0
    local b = tonumber(hex:sub(5, 6), 16) or 0
    return Color3.fromRGB(r, g, b)
end

function BeautifulUI.ToHex(color)
    return string.format("#%02X%02X%02X",
        math.floor(color.R * 255 + 0.5),
        math.floor(color.G * 255 + 0.5),
        math.floor(color.B * 255 + 0.5))
end

local function C(value, fallback)
    if type(value) == "string" then return BeautifulUI.Hex(value) end
    if typeof(value) == "Color3" then return value end
    return fallback or Color3.new(1, 1, 1)
end

local Theme = {}
local function ApplyTheme(t)
    for k, v in pairs(t) do Theme[k] = C(v) end
end
ApplyTheme(Themes.Purple)

BeautifulUI.Themes = Themes
BeautifulUI.Theme = Theme
BeautifulUI.CurrentThemeName = "Purple"

local OpenWindows = {}
local function ReapplyThemeAllWindows()
    for _, win in ipairs(OpenWindows) do
        if win.Gui and win.Gui.Parent then
            pcall(function() win:_reapplyTheme() end)
        end
    end
end

function BeautifulUI:SetTheme(name)
    if Themes[name] then
        ApplyTheme(Themes[name])
        BeautifulUI.Theme = Theme
        BeautifulUI.CurrentThemeName = name
        ReapplyThemeAllWindows()
        return true
    end
    return false
end

function BeautifulUI:SetColor(key, value)
    if Theme[key] ~= nil then
        Theme[key] = C(value)
        ReapplyThemeAllWindows()
        return true
    end
    return false
end

function BeautifulUI:GetColor(key) return Theme[key] end

function BeautifulUI:GetThemeHex()
    local out = {}
    for k, v in pairs(Theme) do out[k] = BeautifulUI.ToHex(v) end
    return out
end

function BeautifulUI:CreateCustomTheme(name, colorsTable)
    local newTheme = {}
    for k, v in pairs(Themes.Purple) do newTheme[k] = v end
    for k, v in pairs(colorsTable) do newTheme[k] = v end
    Themes[name] = newTheme
    return newTheme
end

-- Fix 1: AnimatedColor с уникальной фазой + style + leak guard
function BeautifulUI.AnimatedColor(instance, prop, color1, color2, duration, style)
    color1, color2 = C(color1), C(color2)
    duration = duration or 2
    style = style or Enum.EasingStyle.Sine

    local entry = RegisterAnimation(instance, prop, function(dt, now, a)
        local t = ((now - a.startTime) % duration) / duration
        local wave = (math.sin(t * math.pi * 2 - math.pi / 2) + 1) / 2
        -- Применяем easing style
        if style == Enum.EasingStyle.Linear then
            wave = t
        elseif style == Enum.EasingStyle.Quad then
            wave = wave * wave
        elseif style == Enum.EasingStyle.Quart then
            wave = wave * wave * wave * wave
        end
        instance[prop] = color1:Lerp(color2, wave)
    end)

    return {
        Disconnect = function()
            StopAnimation(instance, prop)
        end
    }
end

-- Fix 2: AnimatedGradient — ОДИН update, rotation + hue shift
function BeautifulUI.AnimatedGradient(instance, colors, speed)
    colors = colors or {Theme.Accent, Theme.AccentDark, Theme.Accent}
    speed = speed or 60

    local resolved = {}
    for _, c in ipairs(colors) do table.insert(resolved, C(c)) end

    local grad = instance:FindFirstChildOfClass("UIGradient")
    if not grad then
        grad = Instance.new("UIGradient")
        grad.Parent = instance
    end

    -- Установить базовые цвета один раз
    local keypoints = {}
    for i = 1, #resolved do
        local t = (i - 1) / math.max(#resolved - 1, 1)
        table.insert(keypoints, ColorSequenceKeypoint.new(t, resolved[i]))
    end
    grad.Color = ColorSequence.new(keypoints)

    local entry = RegisterAnimation(instance, "_gradient", function(dt, now)
        grad.Rotation = (now * speed) % 360

        -- Реальный hue shift (не пересбор одного и того же!)
        local hueShift = (now * 0.05) % 1
        local kp = {}
        for i = 1, #resolved do
            local t = (i - 1) / math.max(#resolved - 1, 1)
            local h, s, v = Color3.toHSV(resolved[i])
            table.insert(kp, ColorSequenceKeypoint.new(t, Color3.fromHSV((h + hueShift) % 1, s, v)))
        end
        grad.Color = ColorSequence.new(kp)
    end)

    return {
        Disconnect = function()
            StopAnimation(instance, "_gradient")
        end
    }
end

-- Fix 3: AnimatedImage — index guard
function BeautifulUI.AnimatedImage(imageLabel, frames, fps)
    frames = frames or {}
    fps = fps or 10
    if #frames == 0 then
        return {Disconnect = function() end}
    end

    local lastIdx = -1
    local entry = RegisterAnimation(imageLabel, "_image", function(dt, now)
        local idx = math.floor(now * fps) % #frames + 1
        if idx ~= lastIdx then
            imageLabel.Image = frames[idx]
            lastIdx = idx
        end
    end)

    return {
        Disconnect = function()
            StopAnimation(imageLabel, "_image")
        end
    }
end

local function Create(className, props)
    local inst = Instance.new(className)
    for k, v in pairs(props or {}) do inst[k] = v end
    return inst
end

local function Corner(radius, parent)
    return Create("UICorner", {CornerRadius = UDim.new(0, radius or 8), Parent = parent})
end

local function Stroke(color, thickness, transparency, parent)
    return Create("UIStroke", {Color = C(color, Theme.Accent), Thickness = thickness or 1, Transparency = transparency or 0.5, Parent = parent})
end

local function Tween(obj, time, props)
    local tw = TweenService:Create(obj, TweenInfo.new(time or 0.2, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), props)
    tw:Play()
    return tw
end

local function Gradient(c1, c2, rot, parent)
    return Create("UIGradient", {Color = ColorSequence.new(C(c1), C(c2)), Rotation = rot or 90, Parent = parent})
end

local function MakeDraggable(frame, dragArea)
    dragArea = dragArea or frame
    local dragging, dragStart, startPos
    local connections = {}
    table.insert(connections, dragArea.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true; dragStart = input.Position; startPos = frame.Position
            local c
            c = input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then
                    dragging = false
                    if c then c:Disconnect() end
                end
            end)
        end
    end))
    table.insert(connections, UserInputService.InputChanged:Connect(function(input)
        if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
            local delta = input.Position - dragStart
            frame.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
        end
    end))
    return function()
        for _, c in ipairs(connections) do pcall(function() c:Disconnect() end) end
    end
end

local ConfigFlags = {}
local ConfigPath = "BeautifulUI_Config.json"

function BeautifulUI:SetConfigPath(p) ConfigPath = p end
function BeautifulUI:SaveConfig()
    local data = {}
    for k, v in pairs(ConfigFlags) do if v.Get then data[k] = v:Get() end end
    local ok, encoded = pcall(HttpService.JSONEncode, HttpService, data)
    if ok then
        pcall(function() if writefile then writefile(ConfigPath, encoded) end end)
        BeautifulUI:Notify("Конфиг", "Сохранено ✅", 2, "Success")
    end
end
function BeautifulUI:LoadConfig()
    local ok, content = pcall(function()
        if readfile and isfile and isfile(ConfigPath) then return readfile(ConfigPath) end
    end)
    if not ok or not content then
        BeautifulUI:Notify("Конфиг", "Файл не найден", 2, "Warning"); return
    end
    local ok2, data = pcall(HttpService.JSONDecode, HttpService, content)
    if not ok2 then return end
    for k, v in pairs(data) do
        if ConfigFlags[k] and ConfigFlags[k].Set then pcall(function() ConfigFlags[k]:Set(v) end) end
    end
    BeautifulUI:Notify("Конфиг", "Загружено ✅", 2, "Success")
end
function BeautifulUI:DeleteConfig()
    pcall(function() if delfile and isfile and isfile(ConfigPath) then delfile(ConfigPath) end end)
end

local DEBUG_ALLOWED = "Ck22d6"
local DebugConsole = {Logs = {}, Visible = false, Gui = nil}

function DebugConsole:Log(level, ...)
    local args = {...}
    local parts = {}
    for _, v in ipairs(args) do table.insert(parts, tostring(v)) end
    local msg = table.concat(parts, " ")
    table.insert(self.Logs, {level = level, msg = msg, time = os.date("%H:%M:%S")})
    print("[BeautifulUI " .. level .. "] " .. msg)
    if self.Visible and self.Gui then self:Refresh() end
end
function DebugConsole:Info(...) self:Log("INFO", ...) end
function DebugConsole:Warn(...) self:Log("WARN", ...) end
function DebugConsole:Error(...) self:Log("ERROR", ...) end

function DebugConsole:Refresh()
    if not self.Gui then return end
    local list = self.Gui:FindFirstChild("List")
    if not list then return end
    for _, c in ipairs(list:GetChildren()) do if c:IsA("TextLabel") then c:Destroy() end end
    for i, log in ipairs(self.Logs) do
        local colors = {INFO = Theme.Info, WARN = Theme.Warning, ERROR = Theme.Danger}
        Create("TextLabel", {Size = UDim2.new(1, -10, 0, 18), BackgroundTransparency = 1, Font = Enum.Font.Code, Text = "[" .. log.time .. "] [" .. log.level .. "] " .. log.msg, TextColor3 = colors[log.level] or Theme.Text, TextSize = 12, TextXAlignment = Enum.TextXAlignment.Left, TextWrapped = true, LayoutOrder = i, Parent = list})
    end
    task.wait()
    local layout = list:FindFirstChildOfClass("UIListLayout")
    if layout and list:IsA("ScrollingFrame") then
        list.CanvasSize = UDim2.new(0, 0, 0, layout.AbsoluteContentSize.Y + 10)
        list.CanvasPosition = Vector2.new(0, list.CanvasSize.Y.Offset)
    end
end

function DebugConsole:Toggle()
    if not self.Gui then self:Create() end
    self.Visible = not self.Visible
    self.Gui.Enabled = self.Visible
    if self.Visible then self:Refresh() end
end

    Corner(6, input); Create("UIPadding", {PaddingLeft = UDim.new(0, 8), Parent = input})

        if enter and input.Text ~= "" then
            self:Info("> " .. input.Text)
            if not loadstring then self:Error("loadstring недоступен"); input.Text = ""; return end
            local fn, err = loadstring(input.Text)
            if not fn then self:Error("Compile error: " .. tostring(err)); input.Text = ""; return end
            local ok, runErr = pcall(fn)
            if not ok then self:Error("Runtime error: " .. tostring(runErr)) end
            input.Text = ""
        end
    end)
    MakeDraggable(frame, topBar)
end

function BeautifulUI:DialogBox(config)
    config = config or {}
    local text = config.Text or "Hello!"
    local speaker = config.Speaker or nil
    local avatarId = config.Avatar or nil
    local duration = config.Duration or 0
    local accentColor = config.Accent and C(config.Accent) or Theme.Accent
    local iconChar = config.Icon or "💬"

    local gui = Create("ScreenGui", {Name = "BeautifulUIDialog_" .. tick(), ResetOnSpawn = false, ZIndexBehavior = Enum.ZIndexBehavior.Sibling, Parent = (gethui and gethui()) or CoreGui})
    local frame = Create("Frame", {Size = UDim2.new(0, 640, 0, 140), Position = UDim2.new(0.5, -320, 1, 200), BackgroundColor3 = Theme.Secondary, BorderSizePixel = 0, Parent = gui})
    Corner(16, frame); Stroke(accentColor, 2, 0.3, frame); Gradient(Theme.Gradient1, Theme.Secondary, 90, frame)

    local avFrame = Create("Frame", {Size = UDim2.new(0, 100, 0, 100), Position = UDim2.new(0, 20, 0.5, -50), BackgroundColor3 = Theme.Tertiary, BorderSizePixel = 0, Parent = frame})
    Corner(14, avFrame); Stroke(accentColor, 2, 0.2, avFrame)
    if avatarId then
        local img = Create("ImageLabel", {Size = UDim2.new(1, -6, 1, -6), Position = UDim2.new(0, 3, 0, 3), BackgroundTransparency = 1, Image = avatarId, ScaleType = Enum.ScaleType.Crop, Parent = avFrame})
        Corner(12, img)
        if config.Frames then BeautifulUI.AnimatedImage(img, config.Frames, config.FPS or 10) end
        if config.Icon then
            local badge = Create("TextLabel", {Size = UDim2.new(0, 26, 0, 26), Position = UDim2.new(1, -14, 1, -14), BackgroundColor3 = accentColor, BorderSizePixel = 0, Font = Enum.Font.GothamBold, Text = config.Icon, TextColor3 = Theme.Text, TextSize = 14, Parent = avFrame})
            Corner(13, badge); Stroke(Theme.Secondary, 2, 0, badge)
        end
    else
        Create("TextLabel", {Size = UDim2.new(1, 0, 1, 0), BackgroundTransparency = 1, Font = Enum.Font.GothamBold, Text = iconChar, TextColor3 = accentColor, TextSize = 42, Parent = avFrame})
    end
    if speaker then
        Create("TextLabel", {Size = UDim2.new(0, 150, 0, 24), Position = UDim2.new(0, 140, 0, 14), BackgroundColor3 = accentColor, BorderSizePixel = 0, Font = Enum.Font.GothamBold, Text = speaker, TextColor3 = Theme.Text, TextSize = 13, Parent = frame}):Clone()
        local nameLabel = Create("TextLabel", {Size = UDim2.new(0, 150, 0, 24), Position = UDim2.new(0, 140, 0, 14), BackgroundColor3 = accentColor, BorderSizePixel = 0, Font = Enum.Font.GothamBold, Text = speaker, TextColor3 = Theme.Text, TextSize = 13, Parent = frame})
        Corner(6, nameLabel)
    end
    local textLabel = Create("TextLabel", {Size = UDim2.new(1, -170, 1, -50), Position = UDim2.new(0, 140, speaker and 0.4 or 0.15, speaker and 4 or 0), BackgroundTransparency = 1, Font = Enum.Font.Gotham, Text = "", TextColor3 = Theme.Text, TextSize = 16, TextWrapped = true, TextXAlignment = Enum.TextXAlignment.Left, TextYAlignment = Enum.TextYAlignment.Top, RichText = true, Parent = frame})
    task.spawn(function()
        for i = 1, #text do
            if not textLabel.Parent then return end
            textLabel.Text = text:sub(1, i); task.wait(0.02)
        end
    end)
    local closeBtn = Create("TextButton", {Size = UDim2.new(0, 30, 0, 30), Position = UDim2.new(1, -40, 0, 12), BackgroundColor3 = Theme.Tertiary, BorderSizePixel = 0, Font = Enum.Font.GothamBold, Text = "✕", TextColor3 = Theme.SubText, TextSize = 12, Parent = frame})
    Corner(8, closeBtn)
    local closed = false
    local function close()
        if closed then return end
        closed = true
        Tween(frame, 0.4, {Position = UDim2.new(0.5, -320, 1, 200)})
        task.wait(0.45); gui:Destroy()
    end
    closeBtn.MouseButton1Click:Connect(close)
    Tween(frame, 0.5, {Position = UDim2.new(0.5, -320, 1, -170)})
    if duration > 0 then task.delay(duration + (#text * 0.02), close) end
    return {SetText = function(newText) textLabel.Text = newText end, Close = close}
end

    function Window:_reapplyTheme()
        -- Остановить активные анимации на всех элементах окна
        StopAnimation(main)
        StopAnimation(iconButton)

        main.BackgroundColor3 = Theme.Background
        mainStroke.Color = Theme.Accent
        if mainGradient then mainGradient.Color = ColorSequence.new(Theme.Gradient1, Theme.Gradient2) end
        titleLabel.TextColor3 = Theme.Text
        subtitleLabel.TextColor3 = Theme.SubText
        closeBtn.BackgroundColor3 = Theme.Tertiary; closeBtn.TextColor3 = Theme.SubText
        collapseBtn.BackgroundColor3 = Theme.Tertiary; collapseBtn.TextColor3 = Theme.SubText
        if debugBtn then debugBtn.BackgroundColor3 = Theme.Tertiary; debugBtn.TextColor3 = Theme.Warning end
        divider.BackgroundColor3 = Theme.Tertiary
        sidebar.BackgroundColor3 = Theme.Secondary
        searchBox.BackgroundColor3 = Theme.Tertiary
        searchBox.PlaceholderColor3 = Theme.SubText
        searchBox.TextColor3 = Theme.Text
        iconButton.BackgroundColor3 = Theme.Accent
        iconButton.TextColor3 = Theme.Text

        for _, tab in ipairs(Window.Tabs) do
            if Window.ActiveTab ~= tab then
                tab.Button.BackgroundColor3 = Theme.Secondary; tab.Button.TextColor3 = Theme.SubText
            else
                tab.Button.BackgroundColor3 = Theme.Accent; tab.Button.TextColor3 = Theme.Text
            end
            if tab._reapply then pcall(tab._reapply) end
        end
    end

    end)

    function Window:Tab(tabName, icon)
        local tabButton = Create("TextButton", {Name = tabName .. "Tab", Size = UDim2.new(1, 0, 0, 36), BackgroundColor3 = Theme.Secondary, BorderSizePixel = 0, Font = Enum.Font.GothamMedium, Text = "  " .. (icon and icon .. "  " or "") .. tabName, TextColor3 = Theme.SubText, TextSize = 13, TextXAlignment = Enum.TextXAlignment.Left, Parent = sidebar})
        Corner(8, tabButton)
        local page = Create("ScrollingFrame", {Name = tabName .. "Page", Size = UDim2.new(1, 0, 1, 0), BackgroundTransparency = 1, BorderSizePixel = 0, ScrollBarThickness = 4, ScrollBarImageColor3 = Theme.Accent, CanvasSize = UDim2.new(0, 0, 0, 0), AutomaticCanvasSize = Enum.AutomaticSize.Y, Visible = false, Parent = pagesContainer})
        Create("UIListLayout", {Padding = UDim.new(0, 8), SortOrder = Enum.SortOrder.LayoutOrder, Parent = page})
        Create("UIPadding", {PaddingRight = UDim.new(0, 8), Parent = page})

        local themeElements = {}
        local function registerThemeElement(el, updateFn)
            table.insert(themeElements, {el = el, update = updateFn})
        end

        local tab = {Button = tabButton, Page = page, Name = tabName}
        local function activate()
            for _, t in pairs(Window.Tabs) do
                t.Page.Visible = false
                Tween(t.Button, 0.15, {BackgroundColor3 = Theme.Secondary, TextColor3 = Theme.SubText})
            end
            page.Visible = true
            Tween(tabButton, 0.15, {BackgroundColor3 = Theme.Accent, TextColor3 = Theme.Text})
            Window.ActiveTab = tab
        end
        tabButton.MouseButton1Click:Connect(activate)
        tabButton.MouseEnter:Connect(function()
            if Window.ActiveTab ~= tab then Tween(tabButton, 0.15, {BackgroundColor3 = Theme.Tertiary, TextColor3 = Theme.Text}) end
        end)
        tabButton.MouseLeave:Connect(function()
            if Window.ActiveTab ~= tab then Tween(tabButton, 0.15, {BackgroundColor3 = Theme.Secondary, TextColor3 = Theme.SubText}) end
        end)
        table.insert(Window.Tabs, tab)
        if not Window.ActiveTab then activate() end

        tab._reapply = function()
            for _, entry in ipairs(themeElements) do
                if entry.el and entry.el.Parent then pcall(entry.update) end
            end
        end

        local TabAPI = {Window = Window, Page = page}

        -- ─── BUTTON ───
        function TabAPI:_mkButton(parent, text, callback, style, icon)
            style = style or "Primary"
            local bgColor, hoverColor = Theme.Secondary, Theme.Accent
            if style == "Success" then hoverColor = Theme.Success
            elseif style == "Danger" then hoverColor = Theme.Danger
            elseif style == "Warning" then hoverColor = Theme.Warning
            elseif style == "Ghost" then bgColor = Theme.Background; hoverColor = Theme.Tertiary
            elseif style == "Gradient" then hoverColor = Theme.AccentDark end

            local btn = Create("TextButton", {Size = UDim2.new(1, 0, 0, 36), BackgroundColor3 = bgColor, BorderSizePixel = 0, Font = Enum.Font.GothamMedium, Text = (icon and icon .. "  " or "") .. text, TextColor3 = Theme.Text, TextSize = 14, Parent = parent})
            Corner(8, btn)
            local btnStroke = Stroke(Theme.Accent, 1, 0.7, btn)
            local grad
            if style == "Gradient" then grad = Gradient(Theme.Accent, Theme.AccentDark, 0, btn) end
            btn.MouseEnter:Connect(function() Tween(btn, 0.15, {BackgroundColor3 = hoverColor}) end)
            btn.MouseLeave:Connect(function() Tween(btn, 0.15, {BackgroundColor3 = bgColor}) end)
            btn.MouseButton1Click:Connect(function()
                Tween(btn, 0.08, {BackgroundColor3 = Theme.AccentDark})
                task.wait(0.08); Tween(btn, 0.15, {BackgroundColor3 = hoverColor})
                if callback then pcall(callback) end
            end)
            registerThemeElement(btn, function()
                btn.BackgroundColor3 = bgColor; btn.TextColor3 = Theme.Text
                btnStroke.Color = Theme.Accent
                if grad then grad.Color = ColorSequence.new(Theme.Accent, Theme.AccentDark) end
            end)
            return btn
        end

        function TabAPI:Button(text, callback, style, icon)
            return TabAPI:_mkButton(page, text, callback, style, icon)
        end

            btn.MouseButton1Click:Connect(function() if callback then pcall(callback) end end)

        function TabAPI:_mkSlider(parent, text, min, max, default, callback, flag)
            min, max = min or 0, max or 100
            if max == min then max = min + 1 end
            local value = math.clamp(default or min, min, max)
            local updating = false

            local frame = Create("Frame", {Size = UDim2.new(1, 0, 0, 50), BackgroundColor3 = Theme.Secondary, BorderSizePixel = 0, Parent = parent})
            Corner(8, frame)
            local lbl = Create("TextLabel", {Size = UDim2.new(0.7, 0, 0, 20), Position = UDim2.new(0, 12, 0, 6), BackgroundTransparency = 1, Font = Enum.Font.GothamMedium, Text = text, TextColor3 = Theme.Text, TextSize = 13, TextXAlignment = Enum.TextXAlignment.Left, Parent = frame})
            local valueLabel = Create("TextLabel", {Size = UDim2.new(0.3, -12, 0, 20), Position = UDim2.new(0.7, 0, 0, 6), BackgroundTransparency = 1, Font = Enum.Font.GothamBold, Text = tostring(value), TextColor3 = Theme.Accent, TextSize = 13, TextXAlignment = Enum.TextXAlignment.Right, Parent = frame})
            local barBg = Create("Frame", {Size = UDim2.new(1, -24, 0, 6), Position = UDim2.new(0, 12, 0, 34), BackgroundColor3 = Theme.Tertiary, BorderSizePixel = 0, Parent = frame})
            Corner(3, barBg)
            local fill = Create("Frame", {Size = UDim2.new((value - min) / (max - min), 0, 1, 0), BackgroundColor3 = Theme.Accent, BorderSizePixel = 0, Parent = barBg})
            Corner(3, fill)

            local dragging = false
            local function setVal(x, fire)
                local rel = math.clamp((x - barBg.AbsolutePosition.X) / barBg.AbsoluteSize.X, 0, 1)
                local newVal = math.floor(min + (max - min) * rel + 0.5)
                if newVal == value and fire then return end
                value = newVal
                fill.Size = UDim2.new(rel, 0, 1, 0)
                valueLabel.Text = tostring(value)
                if fire and not updating and callback then pcall(callback, value) end
            end
            barBg.InputBegan:Connect(function(i) if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then dragging = true; setVal(i.Position.X, true) end end)
            UserInputService.InputChanged:Connect(function(i) if dragging and (i.UserInputType == Enum.UserInputType.MouseMovement or i.UserInputType == Enum.UserInputType.Touch) then setVal(i.Position.X, true) end end)
            UserInputService.InputEnded:Connect(function(i) if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then dragging = false end end)

            local api = {Set = function(v)
                v = math.clamp(v, min, max)
                if v == value then return end
                updating = true
                value = v
                local rel = (value - min) / (max - min)
                fill.Size = UDim2.new(rel, 0, 1, 0)
                valueLabel.Text = tostring(value)
                updating = false
            end, Get = function() return value end}
            if flag then ConfigFlags[flag] = api end
            registerThemeElement(frame, function()
                frame.BackgroundColor3 = Theme.Secondary; lbl.TextColor3 = Theme.Text
                valueLabel.TextColor3 = Theme.Accent; barBg.BackgroundColor3 = Theme.Tertiary; fill.BackgroundColor3 = Theme.Accent
            end)
            return api
        end

        function TabAPI:Graph(text, graphType, height)
            graphType = graphType or "Line"
            height = height or 160
            local frame = Create("Frame", {Size = UDim2.new(1, 0, 0, height + 40), BackgroundColor3 = Theme.Secondary, BorderSizePixel = 0, Parent = page})
            Corner(8, frame)
            local st = Stroke(Theme.Accent, 1, 0.7, frame)
            local titleLbl = Create("TextLabel", {Size = UDim2.new(1, -20, 0, 20), Position = UDim2.new(0, 12, 0, 6), BackgroundTransparency = 1, Font = Enum.Font.GothamBold, Text = text, TextColor3 = Theme.Text, TextSize = 14, TextXAlignment = Enum.TextXAlignment.Left, Parent = frame})

            local canvas = Create("Frame", {
                Name = "Canvas",
                Size = UDim2.new(1, -24, 0, height),
                Position = UDim2.new(0, 12, 0, 32),
                BackgroundColor3 = Theme.Tertiary,
                BorderSizePixel = 0,
                ClipsDescendants = true,
                Parent = frame
            })
            Corner(6, canvas)

            local gridLines = {}
            if graphType ~= "Pie" then
                for i = 1, 4 do
                    local gl = Create("Frame", {
                        Size = UDim2.new(1, 0, 0, 1),
                        Position = UDim2.new(0, 0, i/5, 0),
                        BackgroundColor3 = Theme.Secondary,
                        BackgroundTransparency = 0.5,
                        BorderSizePixel = 0,
                        Parent = canvas
                    })
                    table.insert(gridLines, gl)
                end
            end

            local container = Create("Frame", {
                Name = "Container",
                Size = UDim2.new(1, 0, 1, 0),
                BackgroundTransparency = 1,
                ClipsDescendants = false,
                Parent = canvas
            })

            local data = {}
            local series = {data}
            local maxPoints = 50
            local palette = {Theme.Accent, Theme.Success, Theme.Warning, Theme.Danger, Theme.Info, Color3.fromRGB(180,100,255), Color3.fromRGB(255,100,180)}

            local function clearContainer()
                for _, c in ipairs(container:GetChildren()) do
                    c:Destroy()
                end
            end

            local function getSafeSize()
                local w = canvas.AbsoluteSize.X
                local h = canvas.AbsoluteSize.Y
                if w < 30 or h < 30 then return nil, nil end
                return w, h
            end

            local function drawSegment(x1, y1, x2, y2, thickness, color, zIndex, transparency)
                local dx, dy = x2 - x1, y2 - y1
                local len = math.sqrt(dx * dx + dy * dy)
                if len < 0.1 then return end
                local ang = math.deg(math.atan2(dy, dx))
                Create("Frame", {
                    Size = UDim2.new(0, len, 0, thickness or 2),
                    Position = UDim2.new(0, x1, 0, y1),
                    AnchorPoint = Vector2.new(0, 0.5),
                    BackgroundColor3 = color,
                    BackgroundTransparency = transparency or 0,
                    BorderSizePixel = 0,
                    Rotation = ang,
                    ZIndex = zIndex or 2,
                    Parent = container
                })
            end

            local function drawDot(x, y, size, color, zIndex)
                local dot = Create("Frame", {
                    Size = UDim2.new(0, size, 0, size),
                    Position = UDim2.new(0, x - size/2, 0, y - size/2),
                    BackgroundColor3 = color,
                    BorderSizePixel = 0,
                    ZIndex = zIndex or 3,
                    Parent = container
                })
                Corner(size/2, dot)
                return dot
            end

            local function drawLine()
                clearContainer()
                if #data < 2 then return end
                local w, h = getSafeSize()
                if not w then return end
                local pad = 12
                local usableW = w - pad * 2
                local usableH = h - pad * 2
                for i = 1, #data - 1 do
                    local x1 = pad + (i - 1) / math.max(#data - 1, 1) * usableW
                    local x2 = pad + i / math.max(#data - 1, 1) * usableW
                    local y1 = pad + usableH - (data[i] / 100) * usableH
                    local y2 = pad + usableH - (data[i + 1] / 100) * usableH
                    drawSegment(x1, y1, x2, y2, 2, Theme.Accent, 2)
                end
                for i, v in ipairs(data) do
                    local x = pad + (i - 1) / math.max(#data - 1, 1) * usableW
                    local y = pad + usableH - (v / 100) * usableH
                    drawDot(x, y, 5, Theme.Accent, 3)
                end
            end

            local function drawArea()
                clearContainer()
                if #data < 2 then return end
                local w, h = getSafeSize()
                if not w then return end
                local pad = 12
                local usableW = w - pad * 2
                local usableH = h - pad * 2
                local bottomY = pad + usableH
                local barW = math.max(usableW / math.max(#data - 1, 1), 2)
                for i = 1, #data - 1 do
                    local x = pad + (i - 1) / math.max(#data - 1, 1) * usableW
                    local y1 = pad + usableH - (data[i] / 100) * usableH
                    local y2 = pad + usableH - (data[i + 1] / 100) * usableH
                    local minY = math.min(y1, y2)
                    local fillH = bottomY - minY
                    Create("Frame", {
                        Size = UDim2.new(0, barW + 1, 0, fillH),
                        Position = UDim2.new(0, x, 0, minY),
                        BackgroundColor3 = Theme.Accent,
                        BackgroundTransparency = 0.55,
                        BorderSizePixel = 0,
                        ZIndex = 1,
                        Parent = container
                    })
                end
                for i = 1, #data - 1 do
                    local x1 = pad + (i - 1) / math.max(#data - 1, 1) * usableW
                    local x2 = pad + i / math.max(#data - 1, 1) * usableW
                    local y1 = pad + usableH - (data[i] / 100) * usableH
                    local y2 = pad + usableH - (data[i + 1] / 100) * usableH
                    drawSegment(x1, y1, x2, y2, 2, Theme.Accent, 3)
                end
            end

            local function drawBar()
                clearContainer()
                if #data == 0 then return end
                local w, h = getSafeSize()
                if not w then return end
                local pad = 8
                local gap = 3
                local barW = math.max((w - pad * 2 - gap * (#data - 1)) / #data, 3)
                for i, v in ipairs(data) do
                    local x = pad + (i - 1) * (barW + gap)
                    local barH = (v / 100) * (h - pad * 2)
                    local bar = Create("Frame", {
                        Size = UDim2.new(0, barW, 0, barH),
                        Position = UDim2.new(0, x, 1, -pad - barH),
                        BackgroundColor3 = Theme.Accent,
                        BorderSizePixel = 0,
                        Parent = container
                    })
                    Corner(3, bar)
                end
            end

            local function drawScatter()
                clearContainer()
                if #data == 0 then return end
                local w, h = getSafeSize()
                if not w then return end
                local pad = 12
                local usableW = w - pad * 2
                local usableH = h - pad * 2
                for i, v in ipairs(data) do
                    local x = pad + (i - 1) / math.max(#data - 1, 1) * usableW
                    local y = pad + usableH - (v / 100) * usableH
                    drawDot(x, y, 8, palette[(i - 1) % #palette + 1], 3)
                end
            end

            local function drawPie()
                clearContainer()
                if #data == 0 then return end
                local w, h = getSafeSize()
                if not w then return end
                local total = 0
                for _, v in ipairs(data) do total = total + v end
                if total == 0 then return end
                local size = math.min(w, h) - 24
                local cx, cy = w / 2, h / 2
                local radius = size / 2
                local startAngle = -90
                for i, v in ipairs(data) do
                    local sweep = (v / total) * 360
                    local steps = math.max(math.ceil(sweep / 3), 3)
                    local color = palette[(i - 1) % #palette + 1]
                    for s = 0, steps - 1 do
                        local a1 = startAngle + (s / steps) * sweep
                        local a2 = startAngle + ((s + 1) / steps) * sweep
                        local midA = (a1 + a2) / 2
                        local rad = math.rad(midA)
                        local wx = cx + math.cos(rad) * radius / 2
                        local wy = cy + math.sin(rad) * radius / 2
                        Create("Frame", {
                            Size = UDim2.new(0, radius, 0, math.max(sweep / steps * 3, 5)),
                            Position = UDim2.new(0, wx - radius / 2, 0, wy - 2.5),
                            AnchorPoint = Vector2.new(0, 0.5),
                            BackgroundColor3 = color,
                            BorderSizePixel = 0,
                            Rotation = midA,
                            ZIndex = 2,
                            Parent = container
                        })
                    end
                    startAngle = startAngle + sweep
                end
                local inner = Create("Frame", {
                    Size = UDim2.new(0, radius, 0, radius),
                    Position = UDim2.new(0.5, -radius/2, 0.5, -radius/2),
                    BackgroundColor3 = canvas.BackgroundColor3,
                    BorderSizePixel = 0,
                    ZIndex = 5,
                    Parent = container
                })
                Corner(radius / 2, inner)
            end

            local function drawRadial()
                clearContainer()
                if #data == 0 then return end
                local w, h = getSafeSize()
                if not w then return end
                local cx, cy = w / 2, h / 2
                local maxR = math.min(cx, cy) - 12
                local n = #data
                local angleStep = 360 / n
                for i = 1, n do
                    Create("Frame", {
                        Size = UDim2.new(0, maxR, 0, 1),
                        Position = UDim2.new(0, cx, 0, cy),
                        AnchorPoint = Vector2.new(0, 0.5),
                        BackgroundColor3 = Theme.Tertiary,
                        BorderSizePixel = 0,
                        Rotation = (i - 1) * angleStep - 90,
                        ZIndex = 1,
                        Parent = container
                    })
                end
                for i, v in ipairs(data) do
                    local ang = math.rad((i - 1) * angleStep - 90)
                    local r = (v / 100) * maxR
                    local x = cx + math.cos(ang) * r
                    local y = cy + math.sin(ang) * r
                    drawDot(x, y, 8, Theme.Accent, 3)
                end
                for i = 1, n do
                    local j = (i % n) + 1
                    local ang1 = math.rad((i - 1) * angleStep - 90)
                    local ang2 = math.rad((j - 1) * angleStep - 90)
                    local r1 = (data[i] / 100) * maxR
                    local r2 = (data[j] / 100) * maxR
                    local x1 = cx + math.cos(ang1) * r1
                    local y1 = cy + math.sin(ang1) * r1
                    local x2 = cx + math.cos(ang2) * r2
                    local y2 = cy + math.sin(ang2) * r2
                    drawSegment(x1, y1, x2, y2, 2, Theme.Accent, 2)
                end
            end

            local function drawStacked()
                clearContainer()
                if #series == 0 then return end
                local w, h = getSafeSize()
                if not w then return end
                local pad = 12
                local usableW = w - pad * 2
                local usableH = h - pad * 2
                for sIdx, sData in ipairs(series) do
                    if #sData >= 2 then
                        local color = palette[((sIdx - 1) % #palette) + 1]
                        for i = 1, #sData - 1 do
                            local x1 = pad + (i - 1) / math.max(#sData - 1, 1) * usableW
                            local x2 = pad + i / math.max(#sData - 1, 1) * usableW
                            local y1 = pad + usableH - (sData[i] / 100) * usableH
                            local y2 = pad + usableH - (sData[i + 1] / 100) * usableH
                            drawSegment(x1, y1, x2, y2, 2, color, 2 + sIdx, sIdx == 1 and 0 or 0.2)
                        end
                    end
                end
            end

            local function redraw()
                if graphType == "Line" then drawLine()
                elseif graphType == "Bar" then drawBar()
                elseif graphType == "Pie" then drawPie()
                elseif graphType == "Area" then drawArea()
                elseif graphType == "Scatter" then drawScatter()
                elseif graphType == "Radial" then drawRadial()
                elseif graphType == "Stacked" then drawStacked()
                else drawLine() end
            end

            local function tryRedrawWithRetry(attempts)
                attempts = attempts or 10
                local count = 0
                local function attempt()
                    count = count + 1
                    local w = canvas.AbsoluteSize.X
                    if w >= 30 then
                        redraw()
                        return true
                    end
                    if count >= attempts then return false end
                    task.delay(0.1, attempt)
                    return false
                end
                attempt()
            end

            canvas:GetPropertyChangedSignal("AbsoluteSize"):Connect(function()
                task.wait(0.05)
                redraw()
            end)

            registerThemeElement(frame, function()
                frame.BackgroundColor3 = Theme.Secondary
                st.Color = Theme.Accent
                titleLbl.TextColor3 = Theme.Text
                canvas.BackgroundColor3 = Theme.Tertiary
                for _, gl in ipairs(gridLines) do
                    gl.BackgroundColor3 = Theme.Secondary
                end
                palette = {Theme.Accent, Theme.Success, Theme.Warning, Theme.Danger, Theme.Info, Color3.fromRGB(180,100,255), Color3.fromRGB(255,100,180)}
                redraw()
            end)

            return {
                AddPoint = function(v)
                    table.insert(data, math.clamp(v, 0, 100))
                    if #data > maxPoints then table.remove(data, 1) end
                    tryRedrawWithRetry(3)
                end,
                SetData = function(arr)
                    data = {}
                    for _, v in ipairs(arr) do table.insert(data, math.clamp(v, 0, 100)) end
                    series = {data}
                    tryRedrawWithRetry(10)
                end,
                GetData = function() return data end,
                Clear = function()
                    data = {}
                    series = {data}
                    clearContainer()
                end,
                AddSeries = function(arr)
                    local ns = {}
                    for _, v in ipairs(arr) do table.insert(ns, math.clamp(v, 0, 100)) end
                    table.insert(series, ns)
                    if graphType == "Stacked" then tryRedrawWithRetry(3) end
                end,
                ClearSeries = function()
                    series = {data}
                    if graphType == "Stacked" then tryRedrawWithRetry(3) end
                end,
                AutoUpdate = function(interval, mn, mx)
                    mn, mx = mn or 0, mx or 100
                    task.spawn(function()
                        while canvas.Parent do
                            table.insert(data, math.random(mn, mx))
                            if #data > maxPoints then table.remove(data, 1) end
                            tryRedrawWithRetry(3)
                            task.wait(interval or 1)
                        end
                    end)
                end,
                AnimateColor = function(c1, c2, duration)
                    return BeautifulUI.AnimatedColor(canvas, "BackgroundColor3", c1, c2, duration)
                end,
                AnimateGradient = function(colors, speed)
                    return BeautifulUI.AnimatedGradient(canvas, colors, speed)
                end
            }
        end

        function TabAPI:SubTab(name, icon)
            local header = Create("TextButton", {Size = UDim2.new(1, 0, 0, 32), BackgroundColor3 = Theme.Tertiary, BorderSizePixel = 0, Font = Enum.Font.GothamMedium, Text = "  " .. (icon and icon .. "  " or "") .. name, TextColor3 = Theme.Text, TextSize = 13, TextXAlignment = Enum.TextXAlignment.Left, Parent = page})
            Corner(6, header)
            local subPage = Create("Frame", {Size = UDim2.new(1, 0, 0, 0), BackgroundTransparency = 1, ClipsDescendants = true, Parent = page})
            Create("UIListLayout", {Padding = UDim.new(0, 6), SortOrder = Enum.SortOrder.LayoutOrder, Parent = subPage})
            Create("UIPadding", {PaddingLeft = UDim.new(0, 10), PaddingRight = UDim.new(0, 6), PaddingTop = UDim.new(0, 6), PaddingBottom = UDim.new(0, 6), Parent = subPage})
            local open = true
            local function updateHeight()
                local layout = subPage:FindFirstChildOfClass("UIListLayout")
                local h = layout.AbsoluteContentSize.Y + 12
                Tween(subPage, 0.25, {Size = UDim2.new(1, 0, 0, open and h or 0)})
            end
            header.MouseButton1Click:Connect(function() open = not open; updateHeight() end)
            subPage.ChildAdded:Connect(function() task.wait(0.05); updateHeight() end)
            subPage.ChildRemoved:Connect(function() task.wait(0.05); updateHeight() end)
            task.defer(updateHeight)
            registerThemeElement(header, function() header.BackgroundColor3 = Theme.Tertiary; header.TextColor3 = Theme.Text end)

            local SubAPI = {}
            SubAPI.Button = function(self, t, cb, style, ic) return TabAPI:_mkButton(subPage, t, cb, style, ic) end
            SubAPI.Toggle = function(self, t, d, cb, f) return TabAPI:_mkToggle(subPage, t, d, cb, f) end
            SubAPI.Slider = function(self, t, mi, ma, de, cb, f) return TabAPI:_mkSlider(subPage, t, mi, ma, de, cb, f) end
            SubAPI.Input = function(self, t, ph, cb, f)
                local frame = Create("Frame", {Size = UDim2.new(1, 0, 0, 36), BackgroundColor3 = Theme.Secondary, BorderSizePixel = 0, Parent = subPage})
                Corner(8, frame)
                local lbl = Create("TextLabel", {Size = UDim2.new(0, 100, 1, 0), Position = UDim2.new(0, 12, 0, 0), BackgroundTransparency = 1, Font = Enum.Font.GothamMedium, Text = t, TextColor3 = Theme.Text, TextSize = 13, TextXAlignment = Enum.TextXAlignment.Left, Parent = frame})
                local box = Create("TextBox", {Size = UDim2.new(1, -120, 1, 0), Position = UDim2.new(0, 110, 0, 0), BackgroundTransparency = 1, Font = Enum.Font.Gotham, Text = "", PlaceholderText = ph or "Введите...", PlaceholderColor3 = Theme.SubText, TextColor3 = Theme.Text, TextSize = 13, TextXAlignment = Enum.TextXAlignment.Left, ClearTextOnFocus = false, Parent = frame})
                box.FocusLost:Connect(function() if cb then pcall(cb, box.Text) end end)
                local api = {Set = function(v) box.Text = tostring(v) end, Get = function() return box.Text end}
                if f then ConfigFlags[f] = api end
                registerThemeElement(frame, function()
                    frame.BackgroundColor3 = Theme.Secondary; lbl.TextColor3 = Theme.Text
                    box.PlaceholderColor3 = Theme.SubText; box.TextColor3 = Theme.Text
                end)
                return api
            end
            SubAPI.Label = function(self, t, c)
                local lbl = Create("TextLabel", {Size = UDim2.new(1, 0, 0, 24), BackgroundTransparency = 1, Font = Enum.Font.GothamBold, Text = t, TextColor3 = c and C(c) or Theme.Text, TextSize = 14, TextXAlignment = Enum.TextXAlignment.Left, Parent = subPage})
                local useTheme = not c
                registerThemeElement(lbl, function() if useTheme then lbl.TextColor3 = Theme.Text end end)
                return lbl
            end
            SubAPI.Dropdown = function(self, t, opts, cb, f)
                local frame = Create("Frame", {Size = UDim2.new(1, 0, 0, 36), BackgroundColor3 = Theme.Secondary, BorderSizePixel = 0, ClipsDescendants = true, Parent = subPage})
                Corner(8, frame)
                local btn = Create("TextButton", {Size = UDim2.new(1, 0, 0, 36), BackgroundTransparency = 1, Text = "", Parent = frame})
                local label = Create("TextLabel", {Size = UDim2.new(1, -40, 1, 0), Position = UDim2.new(0, 12, 0, 0), BackgroundTransparency = 1, Font = Enum.Font.GothamMedium, Text = t .. ": " .. (opts[1] or "—"), TextColor3 = Theme.Text, TextSize = 13, TextXAlignment = Enum.TextXAlignment.Left, Parent = btn})
                local arrow = Create("TextLabel", {Size = UDim2.new(0, 30, 1, 0), Position = UDim2.new(1, -30, 0, 0), BackgroundTransparency = 1, Font = Enum.Font.GothamBold, Text = "▼", TextColor3 = Theme.SubText, TextSize = 11, Parent = btn})
                local list = Create("Frame", {Size = UDim2.new(1, 0, 0, 0), Position = UDim2.new(0, 0, 0, 36), BackgroundTransparency = 1, Parent = frame})
                Create("UIListLayout", {Padding = UDim.new(0, 4), SortOrder = Enum.SortOrder.LayoutOrder, Parent = list})
                local open2 = false; local current = opts[1]
                local optButtons = {}
                for i, opt in ipairs(opts) do
                    local optBtn = Create("TextButton", {Size = UDim2.new(1, -8, 0, 28), Position = UDim2.new(0, 4, 0, 0), BackgroundColor3 = Theme.Tertiary, BorderSizePixel = 0, Font = Enum.Font.Gotham, Text = opt, TextColor3 = Theme.Text, TextSize = 12, LayoutOrder = i, Parent = list})
                    Corner(6, optBtn); table.insert(optButtons, optBtn)
                    optBtn.MouseButton1Click:Connect(function() current = opt; label.Text = t .. ": " .. opt; if cb then pcall(cb, opt) end end)
                end
                btn.MouseButton1Click:Connect(function()
                    open2 = not open2
                    local h = open2 and (#opts * 32 + 8) or 0
                    Tween(list, 0.2, {Size = UDim2.new(1, 0, 0, h)})
                    Tween(frame, 0.2, {Size = UDim2.new(1, 0, 0, 36 + h)})
                    arrow.Text = open2 and "▲" or "▼"
                end)
                local api = {Set = function(v)
                    current = v; label.Text = t .. ": " .. tostring(v)
                    if open2 then
                        open2 = false
                        Tween(list, 0.2, {Size = UDim2.new(1, 0, 0, 0)})
                        Tween(frame, 0.2, {Size = UDim2.new(1, 0, 0, 36)})
                        arrow.Text = "▼"
                    end
                    if cb then pcall(cb, v) end
                end, Get = function() return current end}
                if f then ConfigFlags[f] = api end
                registerThemeElement(frame, function()
                    frame.BackgroundColor3 = Theme.Secondary; label.TextColor3 = Theme.Text; arrow.TextColor3 = Theme.SubText
                    for _, ob in ipairs(optButtons) do ob.BackgroundColor3 = Theme.Tertiary; ob.TextColor3 = Theme.Text end
                end)
                return api
            end
            return SubAPI
        end

        return TabAPI
    end

    function Window:Notify(t, txt, dur, tp, btns) BeautifulUI:Notify(t, txt, dur, tp, btns) end
    function Window:DialogBox(cfg) return BeautifulUI:DialogBox(cfg) end

    local visible = true
    UserInputService.InputBegan:Connect(function(input, gpe)
        if gpe then return end
        if input.KeyCode == toggleKey then
            visible = not visible
            if not visible then
                Tween(main, 0.25, {Size = UDim2.new(0, 0, 0, 0), Position = UDim2.new(0.5, 0, 0.5, 0)})
                task.wait(0.26); main.Visible = false; ShowIcon()
            else
                HideIcon(); task.wait(0.05); main.Visible = true
                main.Position = UDim2.new(0.5, -size.X.Offset/2, 0.5, -size.Y.Offset/2)
                Tween(main, 0.3, {Size = size})
            end
        end
    end)

    return Window
end

return BeautifulUI
